### 📁 `todo_app.py`
```python
#!/usr/bin/env python3
"""
CLI To-Do Application with Priority & Due Date support.
"""
import argparse
import json
import os
import sys
from dataclasses import dataclass, asdict
from typing import List, Optional
from datetime import datetime

TASKS_FILE = "tasks.json"

@dataclass
class Task:
    title: str
    completed: bool
    priority: str = "medium"
    due_date: str = ""

    def __post_init__(self):
        if self.priority not in ("low", "medium", "high"):
            raise ValueError("Priority must be 'low', 'medium', or 'high'")
        if self.due_date and not self._is_valid_date(self.due_date):
            raise ValueError("Due date must be in YYYY-MM-DD format")

    @staticmethod
    def _is_valid_date(date_str: str) -> bool:
        try:
            datetime.strptime(date_str, "%Y-%m-%d")
            return True
        except ValueError:
            return False

def load_tasks() -> List[Task]:
    """Load tasks from JSON file with automatic migration for backward compatibility."""
    if not os.path.exists(TASKS_FILE):
        return []
    
    with open(TASKS_FILE, "r") as f:
        data = json.load(f)
    
    tasks = []
    for item in data:
        # Migration: ensure new fields exist with sensible defaults
        task_data = {
            "title": item.get("title"),
            "completed": item.get("completed", False),
            "priority": item.get("priority", "medium"),
            "due_date": item.get("due_date", "")
        }
        tasks.append(Task(**task_data))
    return tasks

def save_tasks(tasks: List[Task]) -> None:
    with open(TASKS_FILE, "w") as f:
        json.dump([asdict(t) for t in tasks], f, indent=2)

def add_task(title: str, priority: str = "medium", due_date: str = "") -> None:
    tasks = load_tasks()
    tasks.append(Task(title=title, completed=False, priority=priority, due_date=due_date))
    save_tasks(tasks)
    print(f"✅ Added: {title}")

def complete_task(index: int) -> None:
    tasks = load_tasks()
    if 0 <= index < len(tasks):
        tasks[index].completed = True
        save_tasks(tasks)
        print(f"✅ Completed: {tasks[index].title}")
    else:
        print("❌ Invalid task index.")

def list_tasks(priority_filter: Optional[str] = None, due_before: Optional[str] = None, sort_by: Optional[str] = None) -> None:
    tasks = load_tasks()
    
    # 🔍 Filtering
    if priority_filter:
        tasks = [t for t in tasks if t.priority == priority_filter]
    if due_before:
        # String comparison works for YYYY-MM-DD format
        tasks = [t for t in tasks if t.due_date and t.due_date < due_before]
        
    # 📊 Sorting
    if sort_by == "due_date":
        # Sort by date; empty dates go to the end
        tasks.sort(key=lambda t: (0 if t.due_date else 1, t.due_date))
    elif sort_by == "priority":
        # High -> Medium -> Low
        priority_rank = {"high": 0, "medium": 1, "low": 2}
        tasks.sort(key=lambda t: priority_rank.get(t.priority, 1))
        
    # 🖨️ Display
    if not tasks:
        print("No tasks found.")
        return
        
    print(f"{'Title':<30} {'Completed':<12} {'Priority':<10} {'Due Date':<12}")
    print("-" * 64)
    for t in tasks:
        print(f"{t.title:<30} {str(t.completed):<12} {t.priority:<10} {t.due_date:<12}")

def parse_args():
    parser = argparse.ArgumentParser(description="A simple CLI To-Do application")
    subparsers = parser.add_subparsers(dest="command", help="Available commands")

    # add
    add_parser = subparsers.add_parser("add", help="Add a new task")
    add_parser.add_argument("title", help="Task title")
    add_parser.add_argument("--priority", choices=["low", "medium", "high"], default="medium", help="Task priority (default: medium)")
    add_parser.add_argument("--due-date", help="Due date (YYYY-MM-DD)")

    # list
    list_parser = subparsers.add_parser("list", help="List tasks")
    list_parser.add_argument("--priority", choices=["low", "medium", "high"], help="Filter by priority")
    list_parser.add_argument("--due-before", help="Filter tasks due before date (YYYY-MM-DD)")
    list_parser.add_argument("--sort", choices=["due_date", "priority"], help="Sort tasks by field")

    # complete
    complete_parser = subparsers.add_parser("complete", help="Mark a task as complete")
    complete_parser.add_argument("index", type=int, help="Task index (0-based)")

    return parser.parse_args()

def main():
    args = parse_args()
    
    if args.command == "add":
        add_task(args.title, args.priority, args.due_date)
    elif args.command == "list":
        list_tasks(priority_filter=args.priority, due_before=args.due_before, sort_by=args.sort)
    elif args.command == "complete":
        complete_task(args.index)
    else:
        parser.print_help()
        sys.exit(1)

if __name__ == "__main__":
    main()
```

### 📁 `test_todo_app.py`
```python
import pytest
import json
import os
import io
import sys
from todo_app import Task, load_tasks, save_tasks, list_tasks, add_task

# Override global file path for tests
import todo_app
todo_app.TASKS_FILE = "test_tasks.json"

@pytest.fixture
def clean_test_file():
    """Ensure a clean state before and after each test."""
    if os.path.exists(todo_app.TASKS_FILE):
        os.remove(todo_app.TASKS_FILE)
    yield
    if os.path.exists(todo_app.TASKS_FILE):
        os.remove(todo_app.TASKS_FILE)

def capture_list_output(priority_filter=None, due_before=None, sort_by=None):
    """Helper to capture stdout from list_tasks for assertions."""
    captured = io.StringIO()
    sys.stdout = captured
    list_tasks(priority_filter=priority_filter, due_before=due_before, sort_by=sort_by)
    sys.stdout = sys.__stdout__
    return captured.getvalue()

def test_migration_on_load(clean_test_file):
    """Verify backward compatibility: old files without priority/due_date get defaults."""
    with open(todo_app.TASKS_FILE, "w") as f:
        json.dump([{"title": "Legacy Task", "completed": False}], f)
        
    tasks = load_tasks()
    assert len(tasks) == 1
    assert tasks[0].priority == "medium"
    assert tasks[0].due_date == ""
    assert tasks[0].title == "Legacy Task"

def test_filter_by_priority(clean_test_file):
    """Verify --priority filtering."""
    save_tasks([
        Task("A", False, "high"),
        Task("B", False, "low"),
        Task("C", False, "high")
    ])
    output = capture_list_output(priority_filter="high")
    assert "A" in output
    assert "B" not in output
    assert "C" in output

def test_filter_due_before(clean_test_file):
    """Verify --due-before filtering."""
    save_tasks([
        Task("Soon", False, "medium", "2025-06-01"),
        Task("Later", False, "medium", "2026-12-01"),
        Task("No Date", False, "medium", "")
    ])
    output = capture_list_output(due_before="2025-07-01")
    assert "Soon" in output
    assert "Later" not in output
    assert "No Date" not in output

def test_sort_by_priority(clean_test_file):
    """Verify --sort priority orders high -> medium -> low."""
    save_tasks([
        Task("Low", False, "low"),
        Task("High", False, "high"),
        Task("Med", False, "medium")
    ])
    output = capture_list_output(sort_by="priority")
    lines = output.strip().split("\n")
    # Skip header line
    assert "High" in lines[1]
    assert "Med" in lines[2]
    assert "Low" in lines[3]

def test_sort_by_due_date(clean_test_file):
    """Verify --sort due_date orders chronologically, empty dates last."""
    save_tasks([
        Task("C", False, "medium", "2025-12-01"),
        Task("A", False, "medium", "2024-01-01"),
        Task("B", False, "medium", "")
    ])
    output = capture_list_output(sort_by="due_date")
    lines = output.strip().split("\n")
    assert "A" in lines[1]
    assert "C" in lines[2]
    assert "B" in lines[3]  # Empty date should be last

def test_combined_filter_and_sort(clean_test_file):
    """Verify filtering and sorting work together."""
    save_tasks([
        Task("X", False, "high", "2025-11-01"),
        Task("Y", False, "low", "2025-01-01"),
        Task("Z", False, "high", "2025-03-01"),
        Task("W", False, "medium", "2025-05-01")
    ])
    output = capture_list_output(priority_filter="high", sort_by="due_date")
    lines = output.strip().split("\n")
    assert "Z" in lines[1]
    assert "X" in lines[2]
    # W and Y should be filtered out
    assert "W" not in output
    assert "Y" not in output
```

### 🔑 Key Features Implemented
- **Migration on Read**: `load_tasks()` safely fills missing `priority` and `due_date` fields for legacy JSON files.
- **Filtering**: `--priority` and `--due-before` use efficient list comprehensions. Date comparison relies on lexicographical ordering of `YYYY-MM-DD`.
- **Sorting**: `due_date` sorts chronologically with empty dates pushed to the end. `priority` sorts `high → medium → low`.
- **CLI Help**: `argparse` automatically generates updated help text covering all new flags.
- **Backward Compatibility**: Existing `tasks.json` files without the new fields load without errors and are treated as `medium` priority / no due date.
