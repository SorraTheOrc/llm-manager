### 📁 `todo.py` (Main Implementation)
```python
#!/usr/bin/env python3
"""CLI To-Do application with priority and due date support."""

import argparse
import json
import os
import sys
from dataclasses import dataclass, asdict
from typing import List, Optional

# ──────────────────────────────────────────────────────────────
# Data Model
# ──────────────────────────────────────────────────────────────
@dataclass
class Task:
    title: str
    completed: bool = False
    priority: str = "medium"  # low, medium, high
    due_date: str = ""        # YYYY-MM-DD or empty

# ──────────────────────────────────────────────────────────────
# Storage & Migration
# ──────────────────────────────────────────────────────────────
TASKS_FILE = "tasks.json"

def load_tasks() -> List[Task]:
    """Load tasks from JSON file with backward compatibility migration."""
    if not os.path.exists(TASKS_FILE):
        return []
    
    with open(TASKS_FILE, "r") as f:
        raw_tasks = json.load(f)
    
    tasks = []
    for raw in raw_tasks:
        tasks.append(Task(
            title=raw.get("title", ""),
            completed=raw.get("completed", False),
            priority=raw.get("priority", "medium"),
            due_date=raw.get("due_date", "")
        ))
    return tasks

def save_tasks(tasks: List[Task]) -> None:
    """Save tasks to JSON file."""
    with open(TASKS_FILE, "w") as f:
        json.dump([asdict(t) for t in tasks], f, indent=2)

# ──────────────────────────────────────────────────────────────
# Core Logic
# ──────────────────────────────────────────────────────────────
def apply_filters(tasks: List[Task], priority: Optional[str] = None, due_before: Optional[str] = None) -> List[Task]:
    if priority:
        tasks = [t for t in tasks if t.priority == priority]
    if due_before:
        # Only filter tasks that have a due date, and keep those strictly before the target
        tasks = [t for t in tasks if t.due_date and t.due_date < due_before]
    return tasks

def apply_sort(tasks: List[Task], sort_by: Optional[str] = None) -> List[Task]:
    if sort_by == "due_date":
        # Empty dates go to the end, otherwise sort chronologically
        tasks.sort(key=lambda t: (t.due_date == "", t.due_date))
    elif sort_by == "priority":
        # high=0, medium=1, low=2 → high priority appears first
        rank = {"high": 0, "medium": 1, "low": 2}
        tasks.sort(key=lambda t: rank.get(t.priority, 1))
    return tasks

def list_tasks(args: argparse.Namespace) -> None:
    tasks = load_tasks()
    tasks = apply_filters(tasks, args.priority, args.due_before)
    tasks = apply_sort(tasks, args.sort)

    if not tasks:
        print("No tasks found.")
        return

    print(f"{'#':<4} {'Title':<20} {'Done':<6} {'Priority':<10} {'Due Date':<12}")
    print("-" * 52)
    for i, t in enumerate(tasks, 1):
        print(f"{i:<4} {t.title:<20} {str(t.completed):<6} {t.priority:<10} {t.due_date:<12}")

# ──────────────────────────────────────────────────────────────
# CLI Setup
# ──────────────────────────────────────────────────────────────
def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        description="To-Do CLI with priority & due date support",
        formatter_class=argparse.RawDescriptionHelpFormatter
    )
    subparsers = parser.add_subparsers(dest="command", help="Available commands")

    # add
    add_p = subparsers.add_parser("add", help="Add a new task")
    add_p.add_argument("title", help="Task title")
    add_p.add_argument("--priority", choices=["low", "medium", "high"], default="medium",
                       help="Task priority (default: medium)")
    add_p.add_argument("--due-date", help="Due date in YYYY-MM-DD format")

    # complete
    complete_p = subparsers.add_parser("complete", help="Mark a task as completed")
    complete_p.add_argument("index", type=int, help="Task index (1-based)")

    # list
    list_p = subparsers.add_parser("list", help="List tasks with optional filtering & sorting")
    list_p.add_argument("--priority", choices=["low", "medium", "high"],
                        help="Filter by priority level")
    list_p.add_argument("--due-before", help="Filter tasks due before YYYY-MM-DD")
    list_p.add_argument("--sort", choices=["due_date", "priority"],
                        help="Sort tasks by field")
    return parser

def main() -> None:
    parser = build_parser()
    args = parser.parse_args()

    if args.command == "add":
        tasks = load_tasks()
        tasks.append(Task(
            title=args.title,
            completed=False,
            priority=args.priority,
            due_date=args.due_date or ""
        ))
        save_tasks(tasks)
        print(f"✅ Added: {args.title}")

    elif args.command == "complete":
        tasks = load_tasks()
        if 1 <= args.index <= len(tasks):
            tasks[args.index - 1].completed = True
            save_tasks(tasks)
            print(f"✅ Completed: {tasks[args.index - 1].title}")
        else:
            print("❌ Invalid task index.")

    elif args.command == "list":
        list_tasks(args)

    else:
        parser.print_help()

if __name__ == "__main__":
    main()
```

### 📁 `test_todo.py` (Unit & Integration Tests)
```python
#!/usr/bin/env python3
import unittest
import os
import json
import argparse
from todo import Task, load_tasks, save_tasks, apply_filters, apply_sort, list_tasks

class TestMigration(unittest.TestCase):
    def setUp(self):
        self.test_file = "test_mig_tasks.json"
        # Simulate legacy format without priority/due_date
        with open(self.test_file, "w") as f:
            json.dump([{"title": "Legacy Task", "completed": False}], f)

    def tearDown(self):
        if os.path.exists(self.test_file):
            os.remove(self.test_file)

    def test_load_migrates_old_format(self):
        # Temporarily swap the global TASKS_FILE
        import todo
        old_file = todo.TASKS_FILE
        todo.TASKS_FILE = self.test_file
        try:
            tasks = load_tasks()
            self.assertEqual(len(tasks), 1)
            self.assertEqual(tasks[0].title, "Legacy Task")
            self.assertEqual(tasks[0].priority, "medium")
            self.assertEqual(tasks[0].due_date, "")
        finally:
            todo.TASKS_FILE = old_file

class TestFiltering(unittest.TestCase):
    def test_filter_by_priority(self):
        tasks = [
            Task("A", priority="high"),
            Task("B", priority="low"),
            Task("C", priority="high")
        ]
        filtered = apply_filters(tasks, priority="high")
        self.assertEqual(len(filtered), 2)
        self.assertTrue(all(t.priority == "high" for t in filtered))

    def test_filter_by_due_before(self):
        tasks = [
            Task("A", due_date="2025-01-01"),
            Task("B", due_date="2026-12-31"),
            Task("C", due_date="")
        ]
        filtered = apply_filters(tasks, due_before="2026-01-01")
        self.assertEqual(len(filtered), 1)
        self.assertEqual(filtered[0].title, "A")

    def test_combined_filters(self):
        tasks = [
            Task("A", priority="high", due_date="2025-06-01"),
            Task("B", priority="low", due_date="2025-06-01"),
            Task("C", priority="high", due_date="2025-07-01")
        ]
        filtered = apply_filters(tasks, priority="high", due_before="2025-06-15")
        self.assertEqual(len(filtered), 1)
        self.assertEqual(filtered[0].title, "A")

class TestSorting(unittest.TestCase):
    def test_sort_by_priority(self):
        tasks = [
            Task("Low", priority="low"),
            Task("High", priority="high"),
            Task("Med", priority="medium")
        ]
        sorted_tasks = apply_sort(tasks, sort_by="priority")
        self.assertEqual([t.title for t in sorted_tasks], ["High", "Med", "Low"])

    def test_sort_by_due_date(self):
        tasks = [
            Task("Late", due_date="2026-01-01"),
            Task("Early", due_date="2024-01-01"),
            Task("NoDue", due_date="")
        ]
        sorted_tasks = apply_sort(tasks, sort_by="due_date")
        self.assertEqual([t.title for t in sorted_tasks], ["Early", "Late", "NoDue"])

class TestCLIIntegration(unittest.TestCase):
    def setUp(self):
        self.test_file = "test_cli_tasks.json"
        import todo
        self.original_file = todo.TASKS_FILE
        todo.TASKS_FILE = self.test_file
        save_tasks([])

    def tearDown(self):
        if os.path.exists(self.test_file):
            os.remove(self.test_file)
        import todo
        todo.TASKS_FILE = self.original_file

    def test_add_and_list(self):
        # Simulate CLI args
        args = argparse.Namespace()
        args.title = "Test Task"
        args.priority = "high"
        args.due_date = "2025-12-25"
        
        import todo
        todo.save_tasks(todo.load_tasks() + [todo.Task(**vars(args))])
        
        list_args = argparse.Namespace(priority=None, due_before=None, sort=None)
        import io
        from contextlib import redirect_stdout
        f = io.StringIO()
        with redirect_stdout(f):
            todo.list_tasks(list_args)
        output = f.getvalue()
        self.assertIn("Test Task", output)
        self.assertIn("high", output)
        self.assertIn("2025-12-25", output)

if __name__ == "__main__":
    unittest.main()
```

### 🚀 Usage Examples
```bash
# Add tasks with new fields
python todo.py add "Buy groceries" --priority high --due-date 2025-08-15
python todo.py add "Read book" --priority low
python todo.py add "Fix bug" --priority medium --due-date 2025-07-01

# Filter by priority
python todo.py list --priority high

# Filter by due date (strictly before)
python todo.py list --due-before 2025-08-01

# Sort by priority (high → low)
python todo.py list --sort priority

# Sort by due date (empty dates move to bottom)
python todo.py list --sort due_date

# Combine filters & sorting
python todo.py list --priority medium --due-before 2025-09-01 --sort due_date
```

### ✅ Key Features & Design Decisions
1. **Backward Compatibility**: `load_tasks()` explicitly defaults missing fields to `priority="medium"` and `due_date=""`, ensuring legacy task files load without errors.
2. **Filtering Logic**: 
   - `--due-before` only applies to tasks that actually have a due date (empty strings are excluded).
   - Filters are composable and applied before sorting.
3. **Sorting Logic**:
   - `due_date`: Empty dates sink to the bottom using tuple sorting `(is_empty, date_string)`.
   - `priority`: Mapped to numeric ranks (`high=0, medium=1, low=2`) for intuitive ascending order (high priority first).
4. **CLI Design**: Uses `argparse` with `choices` for validation. Help text is auto-generated but enhanced with descriptive strings.
5. **Testing**: Unit tests cover migration, filtering, sorting, and CLI integration. Tests isolate the storage file to avoid polluting real data.
6. **Extensibility**: Core logic (`apply_filters`, `apply_sort`) is decoupled from CLI, making it easy to add web APIs, GUIs, or batch processors later.
