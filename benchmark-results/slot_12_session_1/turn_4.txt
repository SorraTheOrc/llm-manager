```python
"""Data models for the To-Do application."""

from __future__ import annotations

import uuid
from dataclasses import dataclass, field, asdict
from datetime import date, datetime
from enum import Enum
from typing import Optional


class TaskStatus(str, Enum):
    PENDING = "pending"
    DONE = "done"


@dataclass
class Task:
    """Represents a single to-do task."""

    id: str = field(default_factory=lambda: str(uuid.uuid4())[:8])
    title: str = ""
    description: str = ""
    status: TaskStatus = TaskStatus.PENDING
    created_at: datetime = field(default_factory=datetime.now)
    updated_at: datetime = field(default_factory=datetime.now)
    due_date: Optional[date] = None
    priority: int = 0  # 0 = normal, 1 = high, 2 = urgent

    def to_dict(self) -> dict:
        """Serialize task to a dictionary (JSON-safe)."""
        d = asdict(self)
        d["status"] = self.status.value
        if self.due_date:
            d["due_date"] = self.due_date.isoformat()
        return d

    @classmethod
    def from_dict(cls, data: dict) -> Task:
        """Deserialize task from a dictionary."""
        data = data.copy()
        data["status"] = TaskStatus(data["status"])
        if data.get("due_date"):
            data["due_date"] = date.fromisoformat(data["due_date"])
        return cls(**{k: v for k, v in data.items() if k in cls.__dataclass_fields__})

    @property
    def is_overdue(self) -> bool:
        """Check if task is overdue."""
        if not self.due_date or self.status != TaskStatus.PENDING:
            return False
        return self.due_date < date.today()

    def __repr__(self) -> str:
        status_icon = "✓" if self.status == TaskStatus.DONE else "○"
        overdue_marker = " ⚠" if self.is_overdue else ""
        return f"Task({status_icon} {self.id}: {self.title}{overdue_marker})"
```

### `todo/store.py`

```python
"""SQLite-backed persistent storage for tasks."""

from __future__ import annotations

import json
import sqlite3
import threading
from contextlib import contextmanager
from datetime import date, datetime
from pathlib import Path
from typing import Iterator, List, Optional

from .models import Task, TaskStatus


class StoreError(Exception):
    """Base exception for store operations."""


class TaskNotFoundError(StoreError):
    """Raised when a task is not found."""


class DuplicateTaskError(StoreError):
    """Raised when a task already exists."""


class TaskStore:
    """Thread-safe SQLite task store."""

    SCHEMA = """
    CREATE TABLE IF NOT EXISTS tasks (
        id TEXT PRIMARY KEY,
        title TEXT NOT NULL,
        description TEXT DEFAULT '',
        status TEXT DEFAULT 'pending',
        created_at TEXT NOT NULL,
        updated_at TEXT NOT NULL,
        due_date TEXT,
        priority INTEGER DEFAULT 0
    );
    CREATE INDEX IF NOT EXISTS idx_tasks_status ON tasks(status);
    CREATE INDEX IF NOT EXISTS idx_tasks_due_date ON tasks(due_date);
    CREATE INDEX IF NOT EXISTS idx_tasks_priority ON tasks(priority);
    """

    def __init__(self, db_path: Optional[Path] = None):
        if db_path is None:
            db_path = Path.home() / ".local" / "share" / "todo" / "tasks.db"
        self.db_path = Path(db_path)
        self.db_path.parent.mkdir(parents=True, exist_ok=True)
        self._local = threading.local()
        self._init_db()

    def _get_conn(self) -> sqlite3.Connection:
        """Get a thread-local database connection."""
        if not hasattr(self._local, "conn") or self._local.conn is None:
            self._local.conn = sqlite3.connect(
                str(self.db_path),
                check_same_thread=False,
            )
            self._local.conn.row_factory = sqlite3.Row
            self._local.conn.execute("PRAGMA journal_mode=WAL")
            self._local.conn.execute("PRAGMA foreign_keys=ON")
        return self._local.conn

    @contextmanager
    def _transaction(self) -> Iterator[sqlite3.Connection]:
        """Context manager for database transactions."""
        conn = self._get_conn()
        try:
            yield conn
            conn.commit()
        except Exception:
            conn.rollback()
            raise

    def _init_db(self) -> None:
        """Initialize database schema."""
        conn = self._get_conn()
        conn.executescript(self.SCHEMA)

    def close(self) -> None:
        """Close the thread-local connection."""
        if hasattr(self._local, "conn") and self._local.conn:
            self._local.conn.close()
            self._local.conn = None

    def add_task(
        self,
        title: str,
        description: str = "",
        due_date: Optional[date] = None,
        priority: int = 0,
    ) -> Task:
        """Add a new task to the store."""
        if not title.strip():
            raise ValueError("Task title cannot be empty")

        task = Task(
            title=title.strip(),
            description=description.strip(),
            due_date=due_date,
            priority=max(0, min(2, priority)),
        )

        with self._transaction() as conn:
            conn.execute(
                """
                INSERT INTO tasks (id, title, description, status, created_at, updated_at, due_date, priority)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?)
                """,
                (
                    task.id,
                    task.title,
                    task.description,
                    task.status.value,
                    task.created_at.isoformat(),
                    task.updated_at.isoformat(),
                    task.due_date.isoformat() if task.due_date else None,
                    task.priority,
                ),
            )

        return task

    def get_task(self, task_id: str) -> Task:
        """Get a task by ID."""
        conn = self._get_conn()
        row = conn.execute("SELECT * FROM tasks WHERE id = ?", (task_id,)).fetchone()
        if row is None:
            raise TaskNotFoundError(f"Task '{task_id}' not found")
        return self._row_to_task(row)

    def update_task_status(self, task_id: str, status: TaskStatus) -> Task:
        """Update the status of a task."""
        with self._transaction() as conn:
            conn.execute(
                "UPDATE tasks SET status = ?, updated_at = ? WHERE id = ?",
                (status.value, datetime.now().isoformat(), task_id),
            )
        return self.get_task(task_id)

    def delete_task(self, task_id: str) -> None:
        """Delete a task by ID."""
        with self._transaction() as conn:
            cursor = conn.execute("DELETE FROM tasks WHERE id = ?", (task_id,))
            if cursor.rowcount == 0:
                raise TaskNotFoundError(f"Task '{task_id}' not found")

    def list_tasks(
        self,
        status: Optional[TaskStatus] = None,
        due_date: Optional[date] = None,
        priority: Optional[int] = None,
        search: Optional[str] = None,
        limit: Optional[int] = None,
        offset: int = 0,
    ) -> List[Task]:
        """List tasks with optional filters."""
        conditions = []
        params = []

        if status is not None:
            conditions.append("status = ?")
            params.append(status.value)
        if due_date is not None:
            conditions.append("due_date = ?")
            params.append(due_date.isoformat())
        if priority is not None:
            conditions.append("priority = ?")
            params.append(priority)
        if search is not None:
            conditions.append("(title LIKE ? OR description LIKE ?)")
            search_term = f"%{search}%"
            params.extend([search_term, search_term])

        where_clause = ""
        if conditions:
            where_clause = "WHERE " + " AND ".join(conditions)

        query = f"SELECT * FROM tasks {where_clause} ORDER BY priority DESC, due_date ASC, created_at DESC"

        if limit is not None:
            query += " LIMIT ? OFFSET ?"
            params.extend([limit, offset])

        conn = self._get_conn()
        rows = conn.execute(query, params).fetchall()
        return [self._row_to_task(row) for row in rows]

    def get_daily_tasks(self, target_date: Optional[date] = None) -> dict:
        """Get tasks grouped by status for a given date.

        Returns a dict with keys: due_today, overdue, upcoming, completed_today.
        """
        if target_date is None:
            target_date = date.today()

        conn = self._get_conn()
        target_str = target_date.isoformat()

        # Tasks due today
        due_today = [
            self._row_to_task(r)
            for r in conn.execute(
                "SELECT * FROM tasks WHERE due_date = ? AND status != 'done' ORDER BY priority DESC, created_at ASC",
                (target_str,),
            ).fetchall()
        ]

        # Overdue tasks
        overdue = [
            self._row_to_task(r)
            for r in conn.execute(
                "SELECT * FROM tasks WHERE due_date < ? AND status != 'done' ORDER BY due_date ASC, priority DESC",
                (target_str,),
            ).fetchall()
        ]

        # Upcoming tasks (due in next 7 days)
        from datetime import timedelta

        week_later = (target_date + timedelta(days=7)).isoformat()
        upcoming = [
            self._row_to_task(r)
            for r in conn.execute(
                "SELECT * FROM tasks WHERE due_date BETWEEN ? AND ? AND status != 'done' ORDER BY due_date ASC",
                (target_str, week_later),
            ).fetchall()
        ]

        # Completed today
        completed_today = [
            self._row_to_task(r)
            for r in conn.execute(
                "SELECT * FROM tasks WHERE status = 'done' AND date(updated_at) = date(?) ORDER BY updated_at DESC",
                (target_str,),
            ).fetchall()
        ]

        return {
            "date": target_date,
            "due_today": due_today,
            "overdue": overdue,
            "upcoming": upcoming,
            "completed_today": completed_today,
            "stats": {
                "total": len(due_today) + len(overdue) + len(upcoming) + len(completed_today),
                "pending": len(due_today) + len(overdue) + len(upcoming),
                "done": len(completed_today),
                "overdue_count": len(overdue),
            },
        }

    def get_stats(self) -> dict:
        """Get overall task statistics."""
        conn = self._get_conn()
        total = conn.execute("SELECT COUNT(*) FROM tasks").fetchone()[0]
        done = conn.execute("SELECT COUNT(*) FROM tasks WHERE status = 'done'").fetchone()[0]
        pending = total - done
        overdue_count = conn.execute(
            "SELECT COUNT(*) FROM tasks WHERE due_date < date('now') AND status = 'pending'"
        ).fetchone()[0]

        return {
            "total": total,
            "pending": pending,
            "done": done,
            "overdue": overdue_count,
        }

    @staticmethod
    def _row_to_task(row: sqlite3.Row) -> Task:
        """Convert a database row to a Task object."""
        return Task(
            id=row["id"],
            title=row["title"],
            description=row["description"] or "",
            status=TaskStatus(row["status"]),
            created_at=datetime.fromisoformat(row["created_at"]),
            updated_at=datetime.fromisoformat(row["updated_at"]),
            due_date=date.fromisoformat(row["due_date"]) if row["due_date"] else None,
            priority=row["priority"],
        )
```

```python
"""Daily summary logic for the To-Do application."""

from __future__ import annotations

from datetime import date
from typing import Optional

from .store import TaskStore


def format_daily_summary(store: TaskStore, target_date: Optional[date] = None) -> str:
    """Format a daily summary as a human-readable string."""
    data = store.get_daily_tasks(target_date)
    today = data["date"]
    lines = []

    lines.append(f"\n{'=' * 50}")
    lines.append(f"  📅 Daily Summary — {today.isoformat()}")
    lines.append(f"{'=' * 50}")

    stats = data["stats"]
    lines.append(f"\n  📊 Stats: {stats['pending']} pending, {stats['done']} done, {stats['overdue_count']} overdue")

    if data["overdue"]:
        lines.append(f"\n  ⚠️  OVERDUE ({len(data['overdue'])}):")
        for task in data["overdue"]:
            lines.append(f"      {task.id}: {task.title} (due {task.due_date.isoformat() if task.due_date else 'no date'})")

    if data["due_today"]:
        lines.append(f"\n  🎯 DUE TODAY ({len(data['due_today'])}):")
        for task in data["due_today"]:
            priority_marker = "🔴" if task.priority == 2 else "🟡" if task.priority == 1 else "⚪"
            lines.append(f"      {priority_marker} {task.id}: {task.title}")

    if data["upcoming"]:
        lines.append(f"\n  🔜 UPCOMING ({len(data['upcoming'])}):")
        for task in data["upcoming"]:
            lines.append(f"      {task.id}: {task.title} (due {task.due_date.isoformat()})")

    if data["completed_today"]:
        lines.append(f"\n  ✅ COMPLETED TODAY ({len(data['completed_today'])}):")
        for task in data["completed_today"]:
            lines.append(f"      ✓ {task.id}: {task.title}")

    if not any([data["overdue"], data["due_today"], data["upcoming"], data["completed_today"]]):
        lines.append("\n  🎉 No tasks for today!")

    lines.append(f"\n{'=' * 50}\n")
    return "\n".join(lines)
```

```python
def render_tasks_table(tasks: list[Task], filters: dict = None) -> str:
    """Render the tasks table."""
    if not tasks:
        return '<div class="empty-state">No tasks found. Add one above!</div>'

    rows = []
    for task in tasks:
        status_class = "done" if task.status == TaskStatus.DONE else ""
        status_badge = (
            '<span class="badge badge-done">Done</span>'
            if task.status == TaskStatus.DONE
            else '<span class="badge badge-pending">Pending</span>'
        )
        if task.is_overdue:
            status_badge = '<span class="badge badge-overdue">Overdue</span>'

        priority_label = (
            '<span class="priority-high">🔴 Urgent</span>'
            if task.priority == 2
            else '<span style="color:#e67e22;font-weight:bold;">🟡 High</span>'
            if task.priority == 1
            else '<span class="priority-normal">Normal</span>'
        )

        due_str = task.due_date.isoformat() if task.due_date else "—"
        due_class = "priority-high" if task.is_overdue else ""

        rows.append(f"""
            <tr class="{status_class}">
                <td>{task.id}</td>
                <td>{priority_label}</td>
                <td><strong>{task.title}</strong><br>
                    <small style="color:#888;">{task.description[:60]}{'...' if len(task.description) > 60 else ''}</small>
                </td>
                <td>{status_badge}</td>
                <td class="{due_class}">{due_str}</td>
                <td>
                    <form method="POST" action="/toggle/{task.id}" style="display:inline;">
                        <button type="submit" class="btn btn-success btn-sm">
                            {('✓ Done' if task.status == TaskStatus.PENDING else '↩ Undo')}
                        </button>
                    </form>
                    <form method="POST" action="/delete/{task.id}" style="display:inline;" onsubmit="return confirm('Delete this task?');">
                        <button type="submit" class="btn btn-danger btn-sm">Delete</button>
                    </form>
                </td>
            </tr>""")

    return f"""
    <div class="filters">
        <form method="GET" action="/" style="display:flex;gap:8px;flex-wrap:wrap;align-items:center;width:100%;">
            <select name="status">
                <option value="">All Statuses</option>
                <option value="pending" {"selected" if filters and filters.get("status") == "pending" else ""}>Pending</option>
```

```python
"""CLI interface for the To-Do application."""

from __future__ import annotations

import argparse
import sys
from datetime import date, datetime
from pathlib import Path
from typing import Optional

from .models import TaskStatus
from .store import TaskStore
from .daily import format_daily_summary


def _get_store(args: argparse.Namespace) -> TaskStore:
    """Get a TaskStore from CLI args."""
    db_path = getattr(args, "db", None)
    if db_path:
        return TaskStore(Path(db_path))
    return TaskStore()


def cmd_add(args: argparse.Namespace) -> int:
    """Add a new task."""
    store = _get_store(args)
    try:
        due_date = None
        if args.due:
            due_date = date.fromisoformat(args.due)

        task = store.add_task(
            title=args.title,
            description=args.description or "",
            due_date=due_date,
            priority=args.priority or 0,
        )
        print(f"✅ Added task {task.id}: {task.title}")
        return 0
    except ValueError as e:
        print(f"❌ Error: {e}", file=sys.stderr)
        return 1
    finally:
        store.close()


def cmd_list(args: argparse.Namespace) -> int:
    """List tasks."""
    store = _get_store(args)
    try:
        status = None
        if args.status:
            status = TaskStatus(args.status)

        tasks = store.list_tasks(
            status=status,
            due_date=date.fromisoformat(args.due) if getattr(args, "due", None) else None,
            search=args.search,
            limit=args.limit,
        )

        if not tasks:
            print("No tasks found.")
            return 0

        print(f"\n{'ID':<10} {'Status':<8} {'Priority':<10} {'Title':<40} {'Due':<12}")
        print("-" * 80)
        for task in tasks:
            status_str = "DONE" if task.status == TaskStatus.DONE else "PENDING"
            due_str = task.due_date.isoformat() if task.due_date else "—"
            priority_str = ["Normal", "High", "Urgent"][task.priority]
            print(f"{task.id:<10} {status_str:<8} {priority_str:<10} {task.title:<40} {due_str:<12}")
        print()
        return 0
    finally:
        store.close()


def cmd_toggle(args: argparse.Namespace) -> int:
    """Toggle task completion."""
    store = _get_store(args)
    try:
        task = store.get_task(args.id)
        new_status = TaskStatus.DONE if task.status == TaskStatus.PENDING else TaskStatus.PENDING
        task = store.update_task_status(args.id, new_status)
        status_str = "completed" if new_status == TaskStatus.DONE else "reopened"
        print(f"✅ Task {task.id} {status_str}: {task.title}")
        return 0
    except Exception as e:
        print(f"❌ Error: {e}", file=sys.stderr)
        return 1
    finally:
        store.close()


def cmd_delete(args: argparse.Namespace) -> int:
    """Delete a task."""
    store = _get_store(args)
    try:
        store.delete_task(args.id)
        print(f"🗑️  Deleted task {args.id}")
        return 0
    except Exception as e:
        print(f"❌ Error: {e}", file=sys.stderr)
        return 1
    finally:
        store.close()


def cmd_daily(args: argparse.Namespace) -> int:
    """Show daily summary."""
    store = _get_store(args)
    try:
        target_date = None
        if hasattr(args, "date") and args.date:
            target_date = date.fromisoformat(args.date)

        print(summary)
        return 0
    finally:
        store.close()

    try:
        stats = store.get_stats()
        print(f"\n📊 To-Do Statistics")
        print(f"   Total tasks:  {stats['total']}")
        print(f"   Pending:      {stats['pending']}")
        print(f"   Completed:    {stats['done']}")

        return 0

        store.close()

    run_server(port=port)
    return 0


def main(argv: Optional[list[str]] = None) -> int:
    """Main entry point for the CLI."""
    parser = argparse.ArgumentParser(
        prog="todo",
        description="A simple to-do application with CLI and web UI.",
    )
    parser.add_argument(
        "--db",
        type=str,
        help="Path to the SQLite database file",
    )

    subparsers = parser.add_subparsers(dest="command", help="Available commands")

    # add
    add_parser = subparsers.add_parser("add", help="Add a new task")
    add_parser.add_argument("title", help="Task title")
    add_parser.add_argument("-d", "--description", help="Task description")
    add_parser.add_argument("--due", help="Due date (YYYY-MM-DD)")
    add_parser.add_argument("--priority", type=int, default=0, choices=[0, 1, 2], help="Priority (0=normal, 1=high, 2=urgent)")

    # list
    list_parser = subparsers.add_parser("list", help="List tasks")
    list_parser.add_argument("--status", choices=["pending", "done"], help="Filter by status")
    list_parser.add_argument("--due", help="Filter by due date (YYYY-MM-DD)")
    list_parser.add_argument("--search", help="Search in title/description")
    list_parser.add_argument("--limit", type=int, help="Limit results")

    # toggle
    toggle_parser = subparsers.add_parser("toggle", help="Toggle task completion")
    toggle_parser.add_argument("id", help="Task ID")

    delete_parser = subparsers.add_parser("delete", help="Delete a task")
    delete_parser.add_argument("id", help="Task ID")

    # stats
    subparsers.add_parser("stats", help="Show task statistics")

    # serve
    serve_parser = subparsers.add_parser("serve", help="Start web server")

if __name__ == "__main__":
```
