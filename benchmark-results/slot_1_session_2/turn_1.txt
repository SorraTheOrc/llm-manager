We'll build a modular to-do application with a clean separation of concerns: a `TodoManager` class handles JSON persistence and task CRUD logic, and a CLI layer built with `argparse` wraps it into user-friendly commands. The module is importable, fully tested, and follows Python best practices.

## Implementation

```python
# todo.py
"""
A simple CLI to-do application with JSON storage.

Usage:
    python todo.py add "Buy milk"
    python todo.py list
    python todo.py complete 1
    python todo.py delete 1
    python todo.py help
"""

import json
import argparse
import sys
import os
from typing import List, Dict, Optional

__all__ = ["TodoManager", "main"]

# Default storage file
DEFAULT_FILE = "tasks.json"


class TodoManager:
    """Manages task persistence and CRUD operations."""

    def __init__(self, filepath: str = DEFAULT_FILE):
        self.filepath = filepath
        self._tasks: List[Dict] = []
        self._load()

    # ------------------------------------------------------------------
    # Internal helpers
    # ------------------------------------------------------------------
    def _load(self) -> None:
        """Load tasks from the JSON file. If the file doesn't exist, start empty."""
        if not os.path.exists(self.filepath):
            self._tasks = []
            return
        try:
            with open(self.filepath, "r") as f:
                data = json.load(f)
            # Support both plain list and dict with 'tasks' key
            if isinstance(data, list):
                self._tasks = data
            elif isinstance(data, dict) and "tasks" in data:
                self._tasks = data["tasks"]
            else:
                self._tasks = []
        except (json.JSONDecodeError, IOError):
            self._tasks = []

    def _save(self) -> None:
        """Write tasks to the JSON file."""
        with open(self.filepath, "w") as f:
            json.dump({"tasks": self._tasks}, f, indent=2)

    def _next_id(self) -> int:
        """Return an auto‑increment id for a new task."""
        if not self._tasks:
            return 1
        max_id = max(t.get("id", 0) for t in self._tasks)
        return max_id + 1

    def _find_task(self, task_id: int) -> Optional[Dict]:
        """Return the task dict with the given id, or None."""
        for t in self._tasks:
            if t.get("id") == task_id:
                return t
        return None

    # ------------------------------------------------------------------
    # Public API
    # ------------------------------------------------------------------
    def add(self, title: str) -> int:
        """
        Add a new task.

        Args:
            title: The task description.

        Returns:
            The id of the newly created task.
        """
        new_id = self._next_id()
        task = {"id": new_id, "title": str(title), "completed": False}
        self._tasks.append(task)
        self._save()
        return new_id

    def list(self, show_completed: bool = True) -> List[Dict]:
        """
        Return all tasks. If show_completed is False, only incomplete tasks are returned.

        Returns:
            A list of task dictionaries.
        """
        if show_completed:
            return list(self._tasks)
        return [t for t in self._tasks if not t["completed"]]

    def complete(self, task_id: int) -> bool:
        """
        Mark a task as completed.

        Args:
            task_id: The id of the task to mark.

        Returns:
            True if the task was found and updated, False otherwise.
        """
        task = self._find_task(task_id)
        if task is None:
            return False
        task["completed"] = True
        self._save()
        return True

    def delete(self, task_id: int) -> bool:
        """
        Delete a task by id.

        Args:
            task_id: The id of the task to remove.

        Returns:
            True if the task existed and was removed, False otherwise.
        """
        task = self._find_task(task_id)
        if task is None:
            return False
        self._tasks.remove(task)
        self._save()
        return True

    def size(self) -> int:
        """Return the number of tasks currently stored."""
        return len(self._tasks)


# ------------------------------------------------------------------
# CLI entry point
# ------------------------------------------------------------------
def main(argv: Optional[List[str]] = None) -> None:
    parser = argparse.ArgumentParser(
        prog="todo",
        description="A simple CLI to‑do list manager.",
        add_help=False,
    )
    # We manually handle 'help' as a command so it doesn't interfere with argparse's default -h
    subparsers = parser.add_subparsers(dest="command", metavar="COMMAND")

    # add
    add_parser = subparsers.add_parser("add", help="Add a new task")
    add_parser.add_argument("title", type=str, help="Task description")

    # list
    list_parser = subparsers.add_parser("list", help="List tasks")
    list_parser.add_argument(
        "-a", "--all", dest="show_all", action="store_true",
        default=True, help="Show both completed and incomplete tasks (default)"
    )
    list_parser.add_argument(
        "--incomplete", dest="show_all", action="store_false",
        help="Show only incomplete tasks"
    )

    # complete
    complete_parser = subparsers.add_parser("complete", help="Mark a task as completed")
    complete_parser.add_argument("id", type=int, help="Task id")

    # delete
    delete_parser = subparsers.add_parser("delete", help="Delete a task")
    delete_parser.add_argument("id", type=int, help="Task id")

    # help
    subparsers.add_parser("help", help="Show this help message")

    args = parser.parse_args(argv)

    if args.command is None or args.command == "help":
        parser.print_help()
        return

    manager = TodoManager()

    if args.command == "add":
        tid = manager.add(args.title)
        print(f"Added task #{tid}: {args.title}")

    elif args.command == "list":
        tasks = manager.list(show_completed=args.show_all)
        if not tasks:
            print("No tasks found.")
        else:
            print(f"{'ID':<4} {'Title':<40} {'Status':<10}")
            print("-" * 60)
            for t in tasks:
                status = "✓" if t["completed"] else "✗"
                print(f"{t['id']:<4} {t['title'][:38]:40} {status:<10}")

    elif args.command == "complete":
        success = manager.complete(args.id)
        if success:
            print(f"Task #{args.id} marked as completed.")
        else:
            print(f"Error: task #{args.id} not found.", file=sys.stderr)
            sys.exit(1)

    elif args.command == "delete":
        success = manager.delete(args.id)
        if success:
            print(f"Task #{args.id} deleted.")
        else:
            print(f"Error: task #{args.id} not found.", file=sys.stderr)
            sys.exit(1)


if __name__ == "__main__":
    main()
```

# test_todo.py
"""
Tests for the todo module using a temporary JSON file.
"""

import json
import pytest
from pathlib import Path
from todo import TodoManager

# ------------------------------------------------------------------
# Fixtures
# ------------------------------------------------------------------
@pytest.fixture
def fresh_manager(tmp_path: Path) -> TodoManager:
    """Return a TodoManager that writes to a temporary file."""
    db_file = tmp_path / "test_tasks.json"
    return TodoManager(str(db_file))


# ------------------------------------------------------------------
# Test CRUD operations
# ------------------------------------------------------------------
class TestAdd:
    def test_add_single(self, fresh_manager: TodoManager):
        tid = fresh_manager.add("Test task")
        assert tid == 1
        assert fresh_manager.size() == 1

    def test_add_multiple_ids(self, fresh_manager: TodoManager):
        id1 = fresh_manager.add("First")
        id2 = fresh_manager.add("Second")
        id3 = fresh_manager.add("Third")
        assert id1 == 1
        assert id2 == 2
        assert id3 == 3
        assert fresh_manager.size() == 3

    def test_add_empty_title(self, fresh_manager: TodoManager):
        tid = fresh_manager.add("")
        assert tid == 1
        tasks = fresh_manager.list()
        assert tasks[0]["title"] == ""

    def test_add_persistence(self, fresh_manager: TodoManager):
        fresh_manager.add("Persist me")
        # Create a second instance pointing to the same file
        manager2 = TodoManager(fresh_manager.filepath)
        assert manager2.size() == 1
        assert manager2.list()[0]["title"] == "Persist me"


class TestList:
    def test_list_all_initial(self, fresh_manager: TodoManager):
        assert fresh_manager.list() == []

    def test_list_all(self, fresh_manager: TodoManager):
        fresh_manager.add("A")
        fresh_manager.add("B")
        tasks = fresh_manager.list()
        assert len(tasks) == 2

    def test_list_incomplete_only(self, fresh_manager: TodoManager):
        fresh_manager.add("Incomplete")
        id2 = fresh_manager.add("Will complete")
        fresh_manager.complete(id2)
        all_tasks = fresh_manager.list(show_completed=True)
        incomplete = fresh_manager.list(show_completed=False)
        assert len(all_tasks) == 2
        assert len(incomplete) == 1
        assert incomplete[0]["title"] == "Incomplete"


class TestComplete:
    def test_complete_existing(self, fresh_manager: TodoManager):
        tid = fresh_manager.add("Task")
        assert fresh_manager.complete(tid) is True
        tasks = fresh_manager.list()
        assert tasks[0]["completed"] is True

    def test_complete_already_completed(self, fresh_manager: TodoManager):
        tid = fresh_manager.add("Task")
        fresh_manager.complete(tid)
        assert fresh_manager.complete(tid) is True  # should still be true

    def test_complete_nonexistent(self, fresh_manager: TodoManager):
        assert fresh_manager.complete(999) is False
        assert fresh_manager.complete(-1) is False

    def test_complete_twice_does_not_duplicate(self, fresh_manager: TodoManager):
        tid = fresh_manager.add("Task")
        fresh_manager.complete(tid)
        fresh_manager.complete(tid)
        assert fresh_manager.size() == 1


class TestDelete:
    def test_delete_existing(self, fresh_manager: TodoManager):
        tid = fresh_manager.add("To delete")
        assert fresh_manager.delete(tid) is True
        assert fresh_manager.size() == 0

    def test_delete_nonexistent(self, fresh_manager: TodoManager):
        assert fresh_manager.delete(999) is False

    def test_delete_twice(self, fresh_manager: TodoManager):
        tid = fresh_manager.add("Task")
        fresh_manager.delete(tid)
        assert fresh_manager.delete(tid) is False  # already gone

    def test_delete_does_not_affect_other(self, fresh_manager: TodoManager):
        id1 = fresh_manager.add("First")
        fresh_manager.add("Second")
        fresh_manager.delete(id1)
        remaining = fresh_manager.list()
        assert len(remaining) == 1
        assert remaining[0]["title"] == "Second"


class TestEdgeCases:
    def test_load_from_empty_file(self, tmp_path: Path):
        db = tmp_path / "empty.json"
        db.write_text("")
        manager = TodoManager(str(db))
        assert manager.size() == 0

    def test_load_from_invalid_json(self, tmp_path: Path):
        db = tmp_path / "bad.json"
        db.write_text("{broken")
        manager = TodoManager(str(db))
        assert manager.size() == 0

    def test_load_from_file_with_dict_but_no_tasks(self, tmp_path: Path):
        db = tmp_path / "empty_dict.json"
        with open(db, "w") as f:
            json.dump({"tasks": []}, f)
        manager = TodoManager(str(db))
        assert manager.size() == 0

    def test_load_from_file_with_custom_task_list(self, tmp_path: Path):
        db = tmp_path / "custom.json"
        with open(db, "w") as f:
            json.dump([{"id": 5, "title": "custom", "completed": True}], f)
        manager = TodoManager(str(db))
        tasks = manager.list()
        assert len(tasks) == 1
        assert tasks[0]["id"] == 5

    def test_next_id_after_manual_delete(self, fresh_manager: TodoManager):
        ids = [fresh_manager.add(f"Task {i}") for i in range(3)]
        fresh_manager.delete(ids[1])  # delete the middle one
        new_id = fresh_manager.add("New after delete")
        # Should be 4, not 2 (no reuse)
        assert new_id == 4

    def test_next_id_with_existing_counter_in_file(self, tmp_path: Path):
        # Simulate a file that already has tasks and a next_id field
        db = tmp_path / "with_counter.json"
        data = {
            "next_id": 10,
            "tasks": [{"id": 1, "title": "existing", "completed": False}]
        }
        with open(db, "w") as f:
            json.dump(data, f)
        manager = TodoManager(str(db))
        new_id = manager.add("Another")
        # The manager always computes max id, ignoring the next_id field
        assert new_id == 2

    def test_remove_from_middle_does_not_shift_ids(self, fresh_manager: TodoManager):
        id1 = fresh_manager.add("First")
        fresh_manager.add("Second")
        id3 = fresh_manager.add("Third")
        fresh_manager.delete(id1)
        tasks = fresh_manager.list()
        assert tasks[0]["id"] == 2  # id not reused
        assert tasks[1]["id"] == 3

## Design Decisions

- **Storage format**: The file stores a JSON object with a `"tasks"` key containing the task list. This allows future extensions (like adding a `next_id` field) without breaking backward compatibility. The loader also handles plain lists for convenience.

- **ID generation**: IDs are always calculated as `max existing id + 1`. No reuse of deleted IDs. A dedicated `_next_id` method keeps the logic centralised.

- **Error handling**: Missing or corrupted files are treated as empty task lists, so the program never crashes on startup. Invalid IDs for `complete`/`delete` return `False` and print an error message to stderr, then exit with code 1.

- **Modularity**: `TodoManager` is fully independent from the CLI. It can be imported and used in other scripts or automated tests.

- **Test coverage**: Every public method (`add`, `list`, `complete`, `delete`) is tested, including edge cases like invalid JSON, missing files, multiple operations, and id behaviour after deletion. The `tmp_path` fixture guarantees test isolation.
