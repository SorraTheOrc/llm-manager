### `todo.py`

```python
#!/usr/bin/env python3
"""To-Do Application with Priority and Due Date Support."""

import argparse
import json
import os
import sys
from datetime import datetime

TASK_FILE = "tasks.json"
PRIORITY_ORDER = {"low": 0, "medium": 1, "high": 2}

def load_tasks():
    """Load tasks from JSON file with automatic migration for older formats."""
    if not os.path.exists(TASK_FILE):
        return []
    
    with open(TASK_FILE, "r", encoding="utf-8") as f:
        raw_tasks = json.load(f)
    
    tasks = []
    for idx, t in enumerate(raw_tasks, start=1):
        # Migration: ensure all fields exist with sensible defaults
        task = {
            "id": t.get("id", idx),
            "title": t.get("title", ""),
            "completed": bool(t.get("completed", False)),
            "priority": t.get("priority", "medium"),
            "due_date": t.get("due_date", "")
        }
        
        # Validate priority
        if task["priority"] not in PRIORITY_ORDER:
            task["priority"] = "medium"
            
        tasks.append(task)
    
    return tasks

def save_tasks(tasks):
    """Save tasks to JSON file."""
    with open(TASK_FILE, "w", encoding="utf-8") as f:
        json.dump(tasks, f, indent=2, ensure_ascii=False)

def add_task(title, priority="medium", due_date=""):
    """Add a new task to the list."""
    tasks = load_tasks()
    new_id = max((t["id"] for t in tasks), default=0) + 1
    
    task = {
        "id": new_id,
        "title": title,
        "completed": False,
        "priority": priority,
        "due_date": due_date
    }
    
    tasks.append(task)
    save_tasks(tasks)
    print(f"Added task #{new_id}: {title}")

def list_tasks(priority=None, due_before=None, sort_by=None):
    """List tasks with optional filtering and sorting."""
    tasks = load_tasks()
    
    # --- Filtering ---
    if priority:
        tasks = [t for t in tasks if t["priority"] == priority]
        
    if due_before:
        try:
            cutoff = datetime.strptime(due_before, "%Y-%m-%d").date()
            tasks = [
                t for t in tasks 
                if t["due_date"] and datetime.strptime(t["due_date"], "%Y-%m-%d").date() < cutoff
            ]
        except ValueError:
            print("Error: Invalid date format for --due-before. Use YYYY-MM-DD.")
            return

    # --- Sorting ---
    if sort_by == "due_date":
        # Empty due dates go to the end
        tasks.sort(key=lambda t: t["due_date"] if t["due_date"] else "9999-12-31")
    elif sort_by == "priority":
        tasks.sort(key=lambda t: PRIORITY_ORDER.get(t["priority"], 1))
        
    # --- Output ---
    if not tasks:
        print("No tasks found.")
        return
        
    print(f"{'ID':<4} {'Title':<20} {'Done':<4} {'Priority':<8} {'Due Date':<12}")
    print("-" * 54)
    for t in tasks:
        done = "Yes" if t["completed"] else "No"
        print(f"{t['id']:<4} {t['title']:<20} {done:<4} {t['priority']:<8} {t['due_date']:<12}")

def main():
    parser = argparse.ArgumentParser(
        description="To-Do Application with Priority and Due Date Support",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
Examples:
  python todo.py add "Buy groceries" --priority high --due-date 2024-12-25
  python todo.py list --priority high
  python todo.py list --due-before 2024-12-31 --sort due_date
  python todo.py list --sort priority
        """
    )
    
    subparsers = parser.add_subparsers(dest="command", help="Available commands")
    
    # Add command
    add_parser = subparsers.add_parser("add", help="Add a new task")
    add_parser.add_argument("title", help="Task title")
    add_parser.add_argument(
        "--priority", 
        choices=["low", "medium", "high"], 
        default="medium",
        help="Task priority (default: medium)"
    )
    add_parser.add_argument(
        "--due-date", 
        help="Due date in YYYY-MM-DD format"
    )
    
    # List command
    list_parser = subparsers.add_parser("list", help="List tasks")
    list_parser.add_argument("--priority", help="Filter by priority (low/medium/high)")
    list_parser.add_argument("--due-before", help="Filter tasks due before YYYY-MM-DD")
    list_parser.add_argument(
        "--sort", 
        choices=["due_date", "priority"], 
        help="Sort by field"
    )
    
    args = parser.parse_args()
    
    if args.command == "add":
        add_task(args.title, args.priority, args.due_date)
    elif args.command == "list":
        list_tasks(priority=args.priority, due_before=args.due_before, sort_by=args.sort)
    else:
        parser.print_help()

if __name__ == "__main__":
    main()
```

class TestMigration(unittest.TestCase):
    def setUp(self):
        self.temp_dir = tempfile.TemporaryDirectory()
        self.original_file = todo.TASK_FILE
        todo.TASK_FILE = os.path.join(self.temp_dir.name, "tasks.json")

        todo.TASK_FILE = self.original_file

        """Ensure tasks without priority or due_date get defaults."""
        with open(todo.TASK_FILE, "w") as f:
            json.dump([

        tasks = todo.load_tasks()
        self.assertEqual(tasks[0]["priority"], "medium")
        self.assertEqual(tasks[0]["due_date"], "")
        self.assertEqual(tasks[1]["priority"], "medium")  # Invalid priority defaults to medium

        """Ensure valid fields are preserved during migration."""
        with open(todo.TASK_FILE, "w") as f:

### Key Improvements & Design Decisions

1. **Data Model**: Tasks are stored as dictionaries with explicit `priority` and `due_date` fields. Defaults are enforced on read.
2. **Migration**: `load_tasks()` checks for missing fields and invalid priorities, ensuring old files work seamlessly without manual intervention.
3. **Filtering**:
   - `--priority` filters by exact match.
   - `--due-before` excludes tasks with empty due dates and parses dates safely.
4. **Sorting**:
   - `--sort due_date` uses string comparison (valid for ISO format) and pushes empty dates to the end.
   - `--sort priority` uses a numeric mapping for correct ordering.
5. **Help Text**: Updated `argparse` description and epilog with clear examples for new options.
6. **Tests**: Comprehensive unit tests cover migration, filtering, sorting, edge cases (invalid dates, empty fields), and backward compatibility.
7. **Robustness**: Handles missing files, invalid inputs, and uses UTF-8 encoding for safety.
