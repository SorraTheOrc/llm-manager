Below is a complete, refactored implementation of the CLI To-Do application with priority and due date support. It includes backward-compatible migration, filtering, sorting, updated help text, and a test suite.

### 📁 `todo.py`
```python
#!/usr/bin/env python3
"""
To-Do CLI Application
Supports priority, due dates, filtering, sorting, and backward-compatible migration.
"""

import json
import os
import sys
import argparse
from datetime import datetime

DATA_FILE = "tasks.json"

def load_tasks():
    """Load tasks from JSON file with migration for missing fields."""
    if not os.path.exists(DATA_FILE):
        return []
    with open(DATA_FILE, 'r') as f:
        try:
            tasks = json.load(f)
        except json.JSONDecodeError:
            print("Error: tasks.json is corrupted.")
            sys.exit(1)
    return migrate_tasks(tasks)

def migrate_tasks(tasks):
    """Ensure all tasks have priority and due_date fields."""
    for task in tasks:
        if 'priority' not in task:
            task['priority'] = 'medium'
        if 'due_date' not in task:
            task['due_date'] = ''
    return tasks

def save_tasks(tasks):
    """Save tasks to JSON file."""
    with open(DATA_FILE, 'w') as f:
        json.dump(tasks, f, indent=2)

def add_task(args):
    """Add a new task with optional priority and due date."""
    tasks = load_tasks()
    priority = args.priority if args.priority else 'medium'
    due_date = args.due_date if args.due_date else ''

    if priority not in ('low', 'medium', 'high'):
        print("Error: Priority must be low, medium, or high.")
        return
    if due_date:
        try:
            datetime.strptime(due_date, '%Y-%m-%d')
        except ValueError:
            print("Error: Due date format must be YYYY-MM-DD.")
            return

    new_id = max((t['id'] for t in tasks), default=0) + 1
    new_task = {
        'id': new_id,
        'title': args.title,
        'completed': False,
        'priority': priority,
        'due_date': due_date
    }
    tasks.append(new_task)
    save_tasks(tasks)
    print(f"Added task #{new_id}: {args.title} [{priority}] due {due_date or 'no date'}")

def list_tasks(args):
    """List tasks with filtering and sorting."""
    tasks = load_tasks()

    # Filter by priority
    if args.priority:
        tasks = [t for t in tasks if t['priority'] == args.priority]

    # Filter by due date
    if args.due_before:
        try:
            cutoff = datetime.strptime(args.due_before, '%Y-%m-%d').date()
            tasks = [t for t in tasks if t['due_date'] and datetime.strptime(t['due_date'], '%Y-%m-%d').date() < cutoff]
        except ValueError:
            print("Error: --due-before requires YYYY-MM-DD format.")
            return

    # Sort
    if args.sort:
        if args.sort == 'due_date':
            tasks.sort(key=lambda t: datetime.strptime(t['due_date'], '%Y-%m-%d').date() if t['due_date'] else datetime.max.date())
        elif args.sort == 'priority':
            priority_rank = {'high': 0, 'medium': 1, 'low': 2}
            tasks.sort(key=lambda t: priority_rank.get(t['priority'], 1))
        else:
            print(f"Error: --sort must be 'due_date' or 'priority'.")
            return

    # Display
    print(f"{'ID':<5} {'Priority':<10} {'Due Date':<12} {'Done':<5} {'Title'}")
    print("-" * 55)
    for t in tasks:
        done = "Yes" if t['completed'] else "No"
        due = t['due_date'] if t['due_date'] else "-"
        print(f"{t['id']:<5} {t['priority']:<10} {due:<12} {done:<5} {t['title']}")

def complete_task(args):
    """Mark a task as completed."""
    tasks = load_tasks()
    for t in tasks:
        if t['id'] == args.id:
            t['completed'] = True
            save_tasks(tasks)
            print(f"Task #{args.id} marked as completed.")
            return
    print(f"Error: Task #{args.id} not found.")

def remove_task(args):
    """Remove a task."""
    tasks = load_tasks()
    tasks = [t for t in tasks if t['id'] != args.id]
    save_tasks(tasks)
    print(f"Task #{args.id} removed.")

def build_parser():
    parser = argparse.ArgumentParser(description="To-Do List CLI")
    subparsers = parser.add_subparsers(dest='command', help='Available commands')

    # add
    add_p = subparsers.add_parser('add', help='Add a new task')
    add_p.add_argument('title', help='Task title')
    add_p.add_argument('--priority', choices=['low', 'medium', 'high'], default='medium', help='Task priority (default: medium)')
    add_p.add_argument('--due-date', help='Due date in YYYY-MM-DD format')

    # list
    list_p = subparsers.add_parser('list', help='List tasks')
    list_p.add_argument('--priority', choices=['low', 'medium', 'high'], help='Filter by priority')
    list_p.add_argument('--due-before', help='Filter tasks due before date (YYYY-MM-DD)')
    list_p.add_argument('--sort', choices=['due_date', 'priority'], help='Sort by due_date or priority')

    # complete
    complete_p = subparsers.add_parser('complete', help='Mark task as completed')
    complete_p.add_argument('id', type=int, help='Task ID')

    # remove
    remove_p = subparsers.add_parser('remove', help='Remove a task')
    remove_p.add_argument('id', type=int, help='Task ID')

    return parser

def main():
    parser = build_parser()
    args = parser.parse_args()

    if not args.command:
        parser.print_help()
        return

    commands = {
        'add': add_task,
        'list': list_tasks,
        'complete': complete_task,
        'remove': remove_task,
    }
    commands[args.command](args)

if __name__ == '__main__':
    main()
```

### 📁 `test_todo.py`
```python
import unittest
import json
import os
import tempfile
import shutil
from datetime import datetime
from todo import load_tasks, migrate_tasks, list_tasks, add_task, build_parser

class TestTodoApp(unittest.TestCase):
    def setUp(self):
        self.temp_dir = tempfile.mkdtemp()
        self.original_file = "tasks.json"
        self.test_file = os.path.join(self.temp_dir, "tasks.json")
        # Override DATA_FILE
        import todo
        todo.DATA_FILE = self.test_file

    def tearDown(self):
        shutil.rmtree(self.temp_dir)
        import todo
        todo.DATA_FILE = "tasks.json"

    def test_migration(self):
        old_data = [{"id": 1, "title": "Old Task", "completed": False}]
        with open(self.test_file, 'w') as f:
            json.dump(old_data, f)
        tasks = load_tasks()
        self.assertEqual(tasks[0]['priority'], 'medium')
        self.assertEqual(tasks[0]['due_date'], '')

    def test_filter_by_priority(self):
        add_task(type('Args', (), {'title': 'A', 'priority': 'high', 'due_date': ''})())
        add_task(type('Args', (), {'title': 'B', 'priority': 'low', 'due_date': ''})())
        out = list_tasks(type('Args', (), {'priority': 'high', 'due_before': None, 'sort': None})())
        self.assertIn('A', out)
        self.assertNotIn('B', out)

    def test_filter_by_due_before(self):
        add_task(type('Args', (), {'title': 'X', 'priority': 'medium', 'due_date': '2025-01-01'})())
        add_task(type('Args', (), {'title': 'Y', 'priority': 'medium', 'due_date': '2027-01-01'})())
        add_task(type('Args', (), {'title': 'Z', 'priority': 'medium', 'due_date': ''})())
        out = list_tasks(type('Args', (), {'priority': None, 'due_before': '2026-01-01', 'sort': None})())
        self.assertIn('X', out)
        self.assertNotIn('Y', out)
        self.assertNotIn('Z', out)

    def test_sort_by_priority(self):
        add_task(type('Args', (), {'title': 'L', 'priority': 'low', 'due_date': ''})())
        add_task(type('Args', (), {'title': 'H', 'priority': 'high', 'due_date': ''})())
        add_task(type('Args', (), {'title': 'M', 'priority': 'medium', 'due_date': ''})())
        out = list_tasks(type('Args', (), {'priority': None, 'due_before': None, 'sort': 'priority'})())
        lines = [l for l in out.split('\n') if l.startswith(('H', 'M', 'L'))]
        self.assertTrue(lines[0].startswith('H'))
        self.assertTrue(lines[1].startswith('M'))
        self.assertTrue(lines[2].startswith('L'))

    def test_sort_by_due_date(self):
        add_task(type('Args', (), {'title': 'C', 'priority': 'medium', 'due_date': '2026-05-01'})())
        add_task(type('Args', (), {'title': 'A', 'priority': 'medium', 'due_date': '2024-01-01'})())
        add_task(type('Args', (), {'title': 'B', 'priority': 'medium', 'due_date': ''})())
        out = list_tasks(type('Args', (), {'priority': None, 'due_before': None, 'sort': 'due_date'})())
        lines = [l for l in out.split('\n') if l.startswith(('A', 'B', 'C'))]
        self.assertTrue(lines[0].startswith('A'))
        self.assertTrue(lines[1].startswith('C'))
        self.assertTrue(lines[2].startswith('B'))

if __name__ == '__main__':
    unittest.main()
```

### ✅ Key Features & Design Decisions

| Feature | Implementation |
|--------|----------------|
| **Backward Compatibility** | `migrate_tasks()` injects `priority='medium'` and `due_date=''` for legacy tasks on every load. |
| **Filtering** | `--priority` filters by exact match; `--due-before` excludes empty due dates and compares dates strictly `< cutoff`. |
| **Sorting** | `--sort priority` uses rank `high=0, medium=1, low=2`; `--sort due_date` places empty dates last via `datetime.max`. |
| **Validation** | `add` validates priority choices and date format; errors prevent invalid data from entering storage. |
| **Help Text** | Updated argparse descriptions and choices for all new flags. |
| **Tests** | Unit tests cover migration, filtering, sorting, and edge cases (empty due dates, missing fields). |
