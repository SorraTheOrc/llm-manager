### 📦 `core.py`
Framework-agnostic task management logic.
```python
import json
import os
from dataclasses import dataclass, asdict, field
from datetime import datetime
from typing import List, Optional
from pathlib import Path

STORAGE_FILE = os.environ.get("TODO_STORAGE", "tasks.json")

@dataclass
class Task:
    id: int
    text: str
    completed: bool = False
    created: str = field(default_factory=lambda: datetime.now().isoformat())
    due: Optional[str] = None

def load_tasks() -> List[Task]:
    path = Path(STORAGE_FILE)
    if not path.exists():
        return []
    with open(path, "r") as f:
        data = json.load(f)
    return [Task(**t) for t in data]

def save_tasks(tasks: List[Task]) -> None:
    with open(STORAGE_FILE, "w") as f:
        json.dump([asdict(t) for t in tasks], f, indent=2)

def add_task(text: str, due: Optional[str] = None) -> Task:
    tasks = load_tasks()
    new_id = max((t.id for t in tasks), default=0) + 1
    task = Task(id=new_id, text=text, due=due)
    tasks.append(task)
    save_tasks(tasks)
    return task

def toggle_task(task_id: int) -> Optional[Task]:
    tasks = load_tasks()
    for t in tasks:
        if t.id == task_id:
            t.completed = not t.completed
            save_tasks(tasks)
            return t
    return None

def delete_task(task_id: int) -> bool:
    tasks = load_tasks()
    new_tasks = [t for t in tasks if t.id != task_id]
    if len(new_tasks) == len(tasks):
        return False
    save_tasks(new_tasks)
    return True

def get_tasks(status: Optional[str] = None) -> List[Task]:
    tasks = load_tasks()
    if status == "pending":
        return [t for t in tasks if not t.completed]
    elif status == "completed":
        return [t for t in tasks if t.completed]
    return tasks

def daily_summary() -> dict:
    tasks = load_tasks()
    today = datetime.now().strftime("%Y-%m-%d")
    total = len(tasks)
    completed = sum(1 for t in tasks if t.completed)
    pending = total - completed
    due_today = sum(1 for t in tasks if t.due and t.due == today)
    return {
        "date": today,
        "total": total,
        "completed": completed,
        "pending": pending,
        "due_today": due_today
    }
```

### 🖥️ `web.py`
FastAPI application with inline CSS, dual form/JSON support, and graceful error handling.
```python
from fastapi import FastAPI, Request, Form, HTTPException
from fastapi.responses import HTMLResponse, RedirectResponse
from core import add_task, toggle_task, delete_task, get_tasks, daily_summary
import uvicorn
import sys

app = FastAPI(title="To-Do Web UI")

HTML_TEMPLATE = """
<!DOCTYPE html>

        <p><strong>Total Tasks:</strong> {{ total }}</p>

    </div>

def render_html(template: str, **kwargs) -> str:
    html = template
    for k, v in kwargs.items():
        placeholder = "{{ " + k + " }}"
        if isinstance(v, bool):
            html = html.replace(placeholder, "True" if v else "False")
        elif v is not None:
            html = html.replace(placeholder, str(v))
    return html

@app.get("/", response_class=HTMLResponse)
async def index(request: Request):
    tasks = get_tasks(request.query_params.get("filter"))
    filter_val = request.query_params.get("filter")
    html = render_html(HTML_TEMPLATE, tasks=tasks, filter=filter_val)
    return HTMLResponse(content=html)

@app.post("/add")
async def add_route(request: Request, text: str = Form(None)):
    # Support both HTML form and JSON payloads
    if not text:
        if request.headers.get("content-type") == "application/json":
            data = await request.json()
            text = data.get("text")
    if not text:
        raise HTTPException(status_code=400, detail="Missing task text")
    add_task(text)
    return RedirectResponse(url="/", status_code=303)

@app.post("/toggle/{task_id}")
async def toggle_route(task_id: int):
    if not toggle_task(task_id):
        raise HTTPException(status_code=404, detail="Task not found")
    return RedirectResponse(url="/", status_code=303)

@app.post("/delete/{task_id}")
async def delete_route(task_id: int):
    if not delete_task(task_id):
        raise HTTPException(status_code=404, detail="Task not found")
    return RedirectResponse(url="/", status_code=303)

@app.get("/daily", response_class=HTMLResponse)
async def daily_route():
    summary = daily_summary()
    html = render_html(DAILY_TEMPLATE, **summary)
    return HTMLResponse(content=html)

def start_server(port: int = 8080):
    def handle_sigint(signum, frame):
        print("\nShutting down server gracefully...")
        sys.exit(0)
    
    import signal
    signal.signal(signal.SIGINT, handle_sigint)
    signal.signal(signal.SIGTERM, handle_sigint)

    try:
        uvicorn.run(app, host="127.0.0.1", port=port, log_level="warning")
    except OSError as e:
        if e.errno in (98, 48, 10048):  # Address already in use (Linux, macOS, Windows)
            print(f"Error: Port {port} is already in use. Try a different port.")
        else:
            print(f"Server error: {e}")
        sys.exit(1)
```

### 📟 `cli.py`
Keeps CLI commands fully functional alongside the web server.
```python
import argparse
import sys
from core import add_task, get_tasks, daily_summary
from web import start_server

def cli_add(args):
    task = add_task(args.text, args.due)
    print(f"Added task #{task.id}: {task.text}")

def cli_list(args):
    tasks = get_tasks(args.status)
    if not tasks:
        print("No tasks found.")
        return
    for t in tasks:
        status = "✅" if t.completed else "⬜"
        print(f"[{t.id}] {status} {t.text} (due: {t.due or 'none'})")

def cli_daily(args):
    summary = daily_summary()
    print(f"Daily Summary for {summary['date']}:")
    print(f"Total: {summary['total']}, Completed: {summary['completed']}, Pending: {summary['pending']}")
    print(f"Due Today: {summary['due_today']}")

def main():
    parser = argparse.ArgumentParser(prog="todo", description="CLI To-Do App")
    subparsers = parser.add_subparsers(dest="command")

    add_parser = subparsers.add_parser("add", help="Add a new task")
    add_parser.add_argument("text", help="Task description")
    add_parser.add_argument("--due", help="Due date (YYYY-MM-DD)")

    list_parser = subparsers.add_parser("list", help="List tasks")
    list_parser.add_argument("--status", choices=["pending", "completed"], help="Filter status")

    subparsers.add_parser("daily", help="Show daily summary")

    serve_parser = subparsers.add_parser("serve", help="Start web UI")
    serve_parser.add_argument("--port", type=int, default=8080, help="Port to run on")

    args = parser.parse_args()
    if not args.command:
        parser.print_help()
        sys.exit(1)

    if args.command == "add":
        cli_add(args)
    elif args.command == "list":
        cli_list(args)
    elif args.command == "daily":
        cli_daily(args)
    elif args.command == "serve":
        start_server(port=args.port)

if __name__ == "__main__":
    main()
```

import pytest

from core import STORAGE_FILE, Task, save_tasks, load_tasks

    assert "To-Do List" in response.text

    assert response.status_code == 303

### 🚀 Usage & Setup

1. **Install dependencies:**
   ```bash
   pip install fastapi uvicorn pytest httpx
   ```

2. **Run CLI commands:**
   ```bash
   python cli.py add "Buy groceries" --due 2024-12-01
   python cli.py list
   python cli.py daily
   ```

3. **Launch Web UI:**
   ```bash
   python cli.py serve --port 8080
   # Visit http://127.0.0.1:8080
   ```

4. **Run Tests:**
   ```bash
   pytest tests.py -v
   ```

### ✅ Key Features
- **Zero external CSS/JS**: All styling is inline, keeping it lightweight.
- **Dual Input Support**: `/add` accepts both `multipart/form-data` (HTML forms) and `application/json`.
- **Graceful Handling**: Catches port conflicts (`errno 98/48/10048`) and handles `Ctrl+C` cleanly.
- **CLI Preserved**: All original CLI commands work independently. Web server is purely additive.
- **Testable**: Uses FastAPI's `TestClient` with isolated file storage via `pytest` fixtures.
