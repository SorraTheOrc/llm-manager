```python
#!/usr/bin/env python3
"""
Todo Application with Priority & Due Date Support

Features:
- Task model: title, completed, priority (low/medium/high), due_date (YYYY-MM-DD or None)
- List command with filtering (--priority, --due-before) and sorting (--sort)
- Backward-compatible JSON storage with automatic migration on read
- Clean CLI with argparse help text
"""

import argparse
import json
import os
import sys
import unittest
from dataclasses import dataclass
from datetime import datetime, date
from typing import List, Optional

# Constants
TASKS_FILE = "tasks.json"
DATE_FMT = "%Y-%m-%d"
PRIORITY_ORDER = {"low": 1, "medium": 2, "high": 3}


@dataclass
class Task:
    title: str
    completed: bool
    priority: str
    due_date: Optional[str]


# ========================
# Storage & Migration
# ========================

def load_tasks() -> List[Task]:
    """Load tasks from JSON file. Automatically migrates older formats."""
    if not os.path.exists(TASKS_FILE):
        return []

    with open(TASKS_FILE, "r") as f:
        data = json.load(f)

    tasks = []
    for item in data:
        # Migration: default missing fields to safe values
        tasks.append(Task(
            title=item.get("title", ""),
            completed=item.get("completed", False),
            priority=item.get("priority", "medium"),
            due_date=item.get("due_date", None)
        ))
    return tasks


def save_tasks(tasks: List[Task]) -> None:
    """Save tasks to JSON file. Always writes the full schema."""
    data = [
        {
            "title": t.title,
            "completed": t.completed,
            "priority": t.priority,
            "due_date": t.due_date
        }
        for t in tasks
    ]
    with open(TASKS_FILE, "w") as f:
        json.dump(data, f, indent=2)


# ========================
# Filtering & Sorting
# ========================

def filter_tasks(
    tasks: List[Task],
    priority: Optional[str] = None,
    due_before: Optional[str] = None
) -> List[Task]:
    """Filter tasks by priority and/or due date."""
    filtered = tasks

    if priority:
        filtered = [t for t in filtered if t.priority == priority]

    if due_before:
        try:
            cutoff = datetime.strptime(due_before, DATE_FMT).date()
            # Include tasks without a due date (they're not "after" the cutoff)
            filtered = [
                t for t in filtered
                if not t.due_date or datetime.strptime(t.due_date, DATE_FMT).date() < cutoff
            ]
        except ValueError:
            print(f"Error: Invalid date format for --due-before. Expected YYYY-MM-DD, got '{due_before}'.", file=sys.stderr)
            sys.exit(1)

    return filtered


def sort_tasks(tasks: List[Task], sort_by: Optional[str] = None) -> List[Task]:
    """Sort tasks by priority or due_date."""
    if not sort_by:
        return tasks

    if sort_by == "priority":
        return sorted(tasks, key=lambda t: PRIORITY_ORDER.get(t.priority, 2))
    elif sort_by == "due_date":
        def due_date_key(t):
            if not t.due_date:
                return datetime.max.date()  # Push incomplete dates to the end
            return datetime.strptime(t.due_date, DATE_FMT).date()
        return sorted(tasks, key=due_date_key)
    else:
        print(f"Warning: Unknown sort field '{sort_by}'. Sorting by insertion order.", file=sys.stderr)
        return tasks


# ========================
# Display
# ========================

def print_tasks(tasks: List[Task]) -> None:
    """Render tasks in a formatted table."""
    if not tasks:
        print("No tasks found.")
        return

    header = f"{'ID':<4} {'Title':<20} {'Status':<8} {'Priority':<8} {'Due Date':<12}"
    print(header)
    print("-" * len(header))
    for i, t in enumerate(tasks, 1):
        status = "Done" if t.completed else "Todo"
        due = t.due_date if t.due_date else "-"
        print(f"{i:<4} {t.title:<20} {status:<8} {t.priority:<8} {due:<12}")


# ========================
# CLI Setup
# ========================

def setup_cli() -> argparse.ArgumentParser:
    """Configure argument parser with updated help text."""
    parser = argparse.ArgumentParser(
        description="To-Do List Application",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
Examples:
  python todo_app.py list --priority high
  python todo_app.py list --due-before 2026-08-01
  python todo_app.py list --sort priority --priority medium
        """
    )
    subparsers = parser.add_subparsers(dest="command", help="Available commands")

    list_parser = subparsers.add_parser("list", help="List tasks with optional filtering and sorting")
    list_parser.add_argument("--priority", choices=["low", "medium", "high"],
                             help="Filter tasks by priority (low, medium, high)")
    list_parser.add_argument("--due-before", metavar="YYYY-MM-DD",
                             help="Filter tasks due before the specified date. Tasks without a due date are included.")
    list_parser.add_argument("--sort", choices=["priority", "due_date"],
                             help="Sort tasks by field. Defaults to insertion order.")
    return parser


def main() -> None:
    """Entry point for CLI execution."""
    parser = setup_cli()
    args = parser.parse_args()

    if args.command == "list":
        tasks = load_tasks()
        tasks = filter_tasks(tasks, args.priority, args.due_before)
        tasks = sort_tasks(tasks, args.sort)
        print_tasks(tasks)
    else:
        parser.print_help()


# ========================
# Tests
# ========================

class TestFiltering(unittest.TestCase):
    def setUp(self):
        self.tasks = [
            Task("Buy milk", False, "medium", "2025-12-01"),
            Task("Finish report", False, "high", "2025-11-15"),
            Task("Call dentist", False, "low", "2025-12-20"),
            Task("No due date", False, "high", None),
            Task("Completed task", True, "low", "2025-10-01"),
        ]

    def test_filter_priority_high(self):
        result = filter_tasks(self.tasks, priority="high")
        self.assertEqual(len(result), 2)
        self.assertTrue(all(t.priority == "high" for t in result))

    def test_filter_due_before(self):
        result = filter_tasks(self.tasks, due_before="2025-12-10")
        self.assertEqual(len(result), 3)  # Includes "No due date"
        for t in result:
            if t.due_date:
                self.assertTrue(datetime.strptime(t.due_date, DATE_FMT).date() < date(2025, 12, 10))

    def test_filter_combined(self):
        result = filter_tasks(self.tasks, priority="high", due_before="2025-12-01")
        self.assertEqual(len(result), 1)
        self.assertEqual(result[0].title, "Finish report")

    def test_filter_no_match(self):
        result = filter_tasks(self.tasks, priority="high", due_before="2025-01-01")
        self.assertEqual(len(result), 0)


class TestSorting(unittest.TestCase):
    def setUp(self):
        self.tasks = [
            Task("A", False, "low", "2025-12-01"),
            Task("B", False, "high", "2025-11-01"),
            Task("C", False, "medium", None),
            Task("D", False, "high", "2025-12-15"),
            Task("E", False, "medium", "2025-11-10"),
        ]

    def test_sort_priority(self):
        result = sort_tasks(self.tasks, "priority")
        priorities = [t.priority for t in result]
        self.assertEqual(priorities, ["low", "medium", "medium", "high", "high"])

    def test_sort_due_date(self):
        result = sort_tasks(self.tasks, "due_date")
        dates = [t.due_date for t in result]
        # None should be last, dates ascending
        self.assertEqual(dates, ["2025-11-01", "2025-11-10", "2025-12-01", "2025-12-15", None])

    def test_sort_unknown(self):
        result = sort_tasks(self.tasks, "invalid")
        self.assertEqual(result, self.tasks)  # Returns original order


class TestBackwardCompatibility(unittest.TestCase):
    def test_migration_defaults(self):
        """Simulate reading old JSON format and verify defaults are applied."""
        old_data = [
            {"title": "Legacy task 1", "completed": False},
            {"title": "Legacy task 2", "completed": True, "priority": "low"}
        ]
        for item in old_data:
            # Simulate migration logic
            task = Task(
                title=item.get("title", ""),
                completed=item.get("completed", False),
                priority=item.get("priority", "medium"),
                due_date=item.get("due_date", None)
            )
            if "priority" not in item:
                self.assertEqual(task.priority, "medium")
            if "due_date" not in item:
                self.assertIsNone(task.due_date)


if __name__ == "__main__":
    if "--test" in sys.argv:
        unittest.main(argv=sys.argv[:1], exit=False)
    else:
        main()
```

### Key Design Decisions

1. **Backward Compatibility**: `load_tasks()` uses `.get()` with sensible defaults (`priority="medium"`, `due_date=None`). When `save_tasks()` runs, it always writes the full schema, so the file self-migrates after the first operation.
2. **Filtering Semantics**: `--due-before` includes tasks without a due date, as they aren't "after" the cutoff. This matches common user expectations.
3. **Sorting Stability**: Python's `sorted()` is stable. Tasks without `due_date` are sorted to the end using `datetime.max.date()` as a sentinel. Priority sorting uses a numeric mapping for correct alphabetical/semantic ordering.
4. **CLI Help**: `argparse` auto-generates clean help text. The `epilog` provides concrete usage examples.
5. **Test Isolation**: Tests run against pure functions without touching the filesystem, ensuring fast, deterministic unit tests. Run them with `python todo_app.py --test`.
