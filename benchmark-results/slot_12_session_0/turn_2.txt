### `todo.py`
```python
#!/usr/bin/env python3
"""
Priority & Due-Date Enhanced To-Do CLI
Supports filtering, sorting, and automatic migration of legacy task files.
"""

import argparse
import json
import os
from dataclasses import dataclass, field, asdict
from typing import List, Optional

DATA_FILE = "tasks.json"

# Priority ordering constants
PRIORITY_MAP = {"low": 0, "medium": 1, "high": 2}
PRIORITY_NAMES = {v: k for k, v in PRIORITY_MAP.items()}

@dataclass
class Task:
    title: str
    completed: bool = False
    priority: str = "medium"
    due_date: str = ""  # Format: YYYY-MM-DD or empty string


def load_tasks() -> List[Task]:
    """Load tasks from JSON file, migrating legacy formats automatically."""
    if not os.path.exists(DATA_FILE):
        return []
    
    with open(DATA_FILE, "r", encoding="utf-8") as f:
        raw_data = json.load(f)
    
    tasks = []
    for item in raw_data:
        # Backward compatibility: inject defaults if keys are missing
        priority = item.get("priority", "medium")
        due_date = item.get("due_date", "")
        completed = item.get("completed", False)
        
        tasks.append(Task(
            title=item["title"],
            completed=completed,
            priority=priority,
            due_date=due_date
        ))
    return tasks


def save_tasks(tasks: List[Task]) -> None:
    """Persist tasks to JSON file."""
    with open(DATA_FILE, "w", encoding="utf-8") as f:
        json.dump([asdict(t) for t in tasks], f, indent=2)


def add_task(title: str, priority: str, due_date: str) -> None:
    """Add a new task to the list."""
    tasks = load_tasks()
    tasks.append(Task(title=title, priority=priority, due_date=due_date))
    save_tasks(tasks)
    print(f"✅ Added task: '{title}'")


def complete_task(index: int) -> None:
    """Toggle completion status of a task by 1-based index."""
    tasks = load_tasks()
    if 0 <= index < len(tasks):
        tasks[index].completed = not tasks[index].completed
        save_tasks(tasks)
        status = "completed" if tasks[index].completed else "pending"
        print(f"✅ Task {index + 1} marked as '{status}'")
    else:
        print("❌ Invalid task index. Please provide a number between 1 and the total tasks.")


def list_tasks(priority_filter: Optional[str] = None, due_before: Optional[str] = None, sort_by: Optional[str] = None) -> None:
    """List tasks with optional filtering and sorting."""
    tasks = load_tasks()
    
    # --- Filtering ---
    if priority_filter:
        tasks = [t for t in tasks if t.priority == priority_filter]
        
    if due_before:
        # Only consider tasks that actually have a due date
        tasks = [t for t in tasks if t.due_date and t.due_date < due_before]
        
    # --- Sorting ---
    if sort_by == "due_date":
        # Sort ascending by date, push tasks with no due date to the end
        tasks.sort(key=lambda t: (t.due_date == "", t.due_date))
    elif sort_by == "priority":
        # Sort descending by priority (high first), then alphabetically by title
        tasks.sort(key=lambda t: (-PRIORITY_MAP.get(t.priority, 1), t.title))
        
    # --- Display ---
    if not tasks:
        print("📭 No tasks found matching your criteria.")
        return
        
    print(f"{'ID':<4} {'Title':<25} {'Priority':<10} {'Due Date':<12} {'Status'}")
    print("-" * 70)
    for i, t in enumerate(tasks):
        status = "✅ Done" if t.completed else "⏳ Pending"
        due_display = t.due_date if t.due_date else "None"
        print(f"{i + 1:<4} {t.title:<25} {t.priority:<10} {due_display:<12} {status}")


def main():
    parser = argparse.ArgumentParser(
        description="📋 To-Do List CLI with Priority & Due Date Support",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
Examples:
  python todo.py add "Buy groceries" --priority high --due 2024-12-31
  python todo.py list
  python todo.py list --priority high --sort priority
  python todo.py list --due-before 2025-01-01 --sort due_date
  python todo.py complete 1
        """
    )
    subparsers = parser.add_subparsers(dest="command", help="Available commands")
    
    # ADD command
    add_parser = subparsers.add_parser("add", help="Add a new task")
    add_parser.add_argument("title", help="Task title")
    add_parser.add_argument("--priority", choices=list(PRIORITY_MAP.keys()), default="medium",
                            help="Task priority (default: medium)")
    add_parser.add_argument("--due", dest="due_date", help="Due date in YYYY-MM-DD format")
    
    # LIST command
    list_parser = subparsers.add_parser("list", help="List tasks with filters & sorting")
    list_parser.add_argument("--priority", choices=list(PRIORITY_MAP.keys()),
                             help="Filter by priority level")
    list_parser.add_argument("--due-before", help="Filter tasks due before a date (YYYY-MM-DD)")
    list_parser.add_argument("--sort", choices=["due_date", "priority"],
                             help="Sort order")
                             
    # COMPLETE command
    done_parser = subparsers.add_parser("complete", help="Toggle task completion status")
    done_parser.add_argument("index", type=int, help="Task number (1-based)")
    
    args = parser.parse_args()
    
    if args.command == "add":
        add_task(args.title, args.priority, args.due_date or "")
    elif args.command == "list":
        list_tasks(args.priority, args.due_before, args.sort)
    elif args.command == "complete":
        complete_task(args.index)
    else:
        parser.print_help()


if __name__ == "__main__":
    main()
```

### Key Improvements & Design Decisions

1. **Data Model**: Used `dataclasses` for type safety and cleaner serialization. `priority` and `due_date` are explicit fields.
2. **Backward Compatibility**: `load_tasks()` automatically injects defaults (`medium`, `""`) when legacy JSON lacks these keys. No manual migration script required.
3. **Filtering Logic**:
   - `--priority`: Exact match.
   - `--due-before`: Excludes tasks with empty `due_date` to avoid ambiguous comparisons. Uses string comparison which works correctly for `YYYY-MM-DD` format.
4. **Sorting Logic**:
   - `--sort priority`: Descending order (High → Low). Ties broken alphabetically by title.
   - `--sort due_date`: Ascending order. Tasks without dates are pushed to the end using a tuple key `(has_no_date, date)`.
5. **CLI UX**: Updated help text with examples, clear column alignment, and emoji indicators for status.
6. **Testing**: Pytest fixtures isolate file state. Tests cover migration, filtering, and sorting edge cases. Mocked `sys.stdout` to verify output without cluttering test runs.
