### 1. `models.py` (Shared Storage & Logic)
```python
import json
import os
from dataclasses import dataclass, asdict, field
from typing import List, Optional
from datetime import datetime

DATA_FILE = "tasks.json"

@dataclass
class Task:
    id: int
    title: str
    completed: bool = False
    created_at: str = field(default_factory=lambda: datetime.now().isoformat())

def load_tasks() -> List[Task]:
    if not os.path.exists(DATA_FILE):
        return []
    with open(DATA_FILE, "r") as f:
        return [Task(**t) for t in json.load(f)]

def save_tasks(tasks: List[Task]) -> None:
    with open(DATA_FILE, "w") as f:
        json.dump([asdict(t) for t in tasks], f, indent=2)

def add_task(title: str) -> Task:
    tasks = load_tasks()
    new_id = max((t.id for t in tasks), default=0) + 1
    new_task = Task(id=new_id, title=title)
    tasks.append(new_task)
    save_tasks(tasks)
    return new_task

def toggle_task(task_id: int) -> Optional[Task]:
    tasks = load_tasks()
    for t in tasks:
        if t.id == task_id:
            t.completed = not t.completed
            save_tasks(tasks)
            return t
    return None

def delete_task(task_id: int) -> bool:
    tasks = load_tasks()
    original_len = len(tasks)
    tasks = [t for t in tasks if t.id != task_id]
    if len(tasks) < original_len:
        save_tasks(tasks)
        return True
    return False

def get_daily_summary() -> dict:
    tasks = load_tasks()
    today = datetime.now().strftime("%Y-%m-%d")
    total = len(tasks)
    completed = sum(1 for t in tasks if t.completed)
    return {"date": today, "total": total, "completed": completed, "pending": total - completed}
```

@app.get("/", response_class=HTMLResponse)
def index(filter: str = "all"):
    tasks = load_tasks()
    rows_html = []
    for t in tasks:
        if filter == "active" and t.completed: continue
        if filter == "completed" and not t.completed: continue
        status = "Done" if t.completed else "Pending"
        cls = "completed" if t.completed else ""
        rows_html.append(f"""<tr>
            <td>{t.id}</td><td class="{cls}">{t.title}</td><td>{status}</td>
            <td class="actions">
                <form method="POST" action="/toggle/{t.id}" style="display:inline"><button type="submit" class="action-btn">{'✓' if not t.completed else '↩'}</button></form>
                <form method="POST" action="/delete/{t.id}" style="display:inline" onsubmit="return confirm('Delete?');"><button type="submit" class="action-btn delete-btn">🗑</button></form>
            </td></tr>""")

    active_classes = {
        "all": "active" if filter == "all" else "",
        "active": "active" if filter == "active" else "",
        "completed": "active" if filter == "completed" else ""
    }

    html = INDEX_TEMPLATE.format(
        filter_all=active_classes["all"],
        filter_active=active_classes["active"],
        filter_completed=active_classes["completed"],

        empty_msg="<p style='text-align:center;color:var(--muted);margin-top:1rem;'>No tasks found. Add one above!</p>" if not rows_html else ""

    return HTMLResponse(html)

### 3. `main.py` (CLI + Server Launcher)
```python
import sys
import argparse
import signal
import uvicorn
from models import add_task, toggle_task, delete_task, load_tasks, get_daily_summary

def cli_main():
    parser = argparse.ArgumentParser(description="Todo CLI & Web App")
    subparsers = parser.add_subparsers(dest="command")

    # CLI Commands
    subparsers.add_parser("add", help="Add a new task").add_argument("title")
    p_complete = subparsers.add_parser("complete", help="Toggle task completion")
    p_complete.add_argument("id", type=int)
    p_delete = subparsers.add_parser("delete", help="Delete a task")
    p_delete.add_argument("id", type=int)
    subparsers.add_parser("list", help="List all tasks")
    subparsers.add_parser("daily", help="Show daily summary")

    # Web Server Command
    p_serve = subparsers.add_parser("serve", help="Start web UI")
    p_serve.add_argument("--port", type=int, default=8080, help="Port to run server on")

    args = parser.parse_args()

    if not args.command:
        parser.print_help()
        return

    # CLI Handlers
    if args.command == "add":
        task = add_task(args.title)
        print(f"Added task #{task.id}: {task.title}")
    elif args.command == "complete":
        task = toggle_task(args.id)
        if task:
            status = "completed" if task.completed else "pending"
            print(f"Task #{task.id} is now {status}")
        else:
            print(f"Task #{args.id} not found")
    elif args.command == "delete":
        if delete_task(args.id):
            print(f"Deleted task #{args.id}")
        else:
            print(f"Task #{args.id} not found")
    elif args.command == "list":
        tasks = load_tasks()
        if not tasks:
            print("No tasks found.")
        else:
            print(f"{'ID':<5} | {'Status':<10} | {'Title'}")
            print("-" * 50)
            for t in tasks:
                status = "✅" if t.completed else "⏳"
                print(f"{t.id:<5} | {status:<10} | {t.title}")
    elif args.command == "daily":
        summary = get_daily_summary()
        print(f"📅 Date: {summary['date']}")
        print(f"✅ Completed: {summary['completed']}")
        print(f"⏳ Pending: {summary['pending']}")
        print(f"📊 Total: {summary['total']}")
    elif args.command == "serve":
        def handle_signal(sig, frame):
            print("\n🛑 Shutting down server gracefully...")
            sys.exit(0)

        signal.signal(signal.SIGINT, handle_signal)
        signal.signal(signal.SIGTERM, handle_signal)

        try:
            print(f"🌐 Starting server on http://localhost:{args.port}")
            uvicorn.run("web:app", host="0.0.0.0", port=args.port, log_level="info")
        except OSError as e:
            if "Address already in use" in str(e) or "[Errno 98]" in str(e):
                print(f"❌ Port {args.port} is already in use. Try a different port or stop the conflicting process.")
                sys.exit(1)
            raise

if __name__ == "__main__":
    cli_main()
```

### 4. `tests/test_web.py`
```python
import httpx
import json
import os
import tempfile
from fastapi.testclient import TestClient
from web import app
import models

# Use a temporary file for storage isolation during tests
TEST_FILE = tempfile.mktemp(suffix=".json")
models.DATA_FILE = TEST_FILE

client = TestClient(app)

def setup_function():
    with open(TEST_FILE, "w") as f:
        json.dump([], f)

def teardown_function():
    if os.path.exists(TEST_FILE):
        os.remove(TEST_FILE)

def test_add_task():
    res = client.post("/add", data={"title": "Test Task"})
    assert res.status_code == 303
    tasks = models.load_tasks()
    assert len(tasks) == 1
    assert tasks[0].title == "Test Task"
    assert tasks[0].id == 1

def test_toggle_task():
    client.post("/add", data={"title": "Toggle Task"})
    res = client.post("/toggle/1")
    assert res.status_code == 303
    tasks = models.load_tasks()
    assert tasks[0].completed is True

def test_delete_task():
    client.post("/add", data={"title": "Delete Task"})
    res = client.post("/delete/1")
    assert res.status_code == 303
    tasks = models.load_tasks()
    assert len(tasks) == 0

def test_index_filter():
    client.post("/add", data={"title": "Task 1"})
    client.post("/add", data={"title": "Task 2"})
    client.post("/toggle/1")
    res = client.get("/?filter=completed")
    assert res.status_code == 200
    assert "Task 1" in res.text
    assert "Task 2" not in res.text

def test_daily_summary():
    client.post("/add", data={"title": "Task 1"})
    client.post("/add", data={"title": "Task 2"})
    res = client.get("/daily")
    assert res.status_code == 200
    assert "Total: 2" in res.text
```

### ✅ Key Features
- **Zero External CSS/JS**: All styling is inline/embedded, no frameworks required.
- **Shared Storage**: CLI and Web UI read/write the same `tasks.json` file.
- **Graceful Shutdown**: Explicit `SIGINT`/`SIGTERM` handlers print a clean message and exit.
- **Port Conflict Handling**: Catches `OSError` during startup and exits with a clear message.
- **Filtering**: Client-side filter buttons toggle `?filter=all|active|completed` query params.
- **Testing**: Isolated temp storage ensures tests don't pollute your actual task data. Works with `httpx`/`TestClient`.
