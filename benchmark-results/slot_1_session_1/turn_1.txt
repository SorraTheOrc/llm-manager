## `todo.py` — Core logic and CLI

```python
#!/usr/bin/env python3
"""
A simple CLI To-Do application with CRUD operations.

Commands:
    add <title>       Add a new task
    list              List all tasks
    complete <id>     Mark a task as completed
    delete <id>       Delete a task
    help              Show this help message
"""

import argparse
import json
import os
import sys
from pathlib import Path
from typing import Dict, List, Optional


DEFAULT_FILE = "todo.json"
ENV_VAR = "TODO_FILE"


def get_filepath(filepath: Optional[str] = None) -> str:
    """Return the path to the tasks file.

    Use the provided *filepath* if given, else try the environment variable
    ``TODO_FILE``, else fall back to ``todo.json`` in the current directory.
    """
    if filepath:
        return filepath
    return os.environ.get(ENV_VAR, DEFAULT_FILE)


def load_tasks(filepath: str) -> List[Dict]:
    """Load tasks from a JSON file.

    Returns an empty list if the file does not exist or is empty.
    """
    path = Path(filepath)
    if not path.is_file() or path.stat().st_size == 0:
        return []
    with path.open("r", encoding="utf-8") as f:
        return json.load(f)


def save_tasks(tasks: List[Dict], filepath: str) -> None:
    """Save a list of tasks to a JSON file."""
    path = Path(filepath)
    with path.open("w", encoding="utf-8") as f:
        json.dump(tasks, f, indent=2)


def get_next_id(tasks: List[Dict]) -> int:
    """Return the next available auto‑increment ID."""
    if not tasks:
        return 1
    return max(task["id"] for task in tasks) + 1


def add_task(title: str, filepath: Optional[str] = None) -> Dict:
    """Add a new task with the given *title* and return it.

    The file is updated immediately.
    """
    path = get_filepath(filepath)
    tasks = load_tasks(path)
    new_task = {
        "id": get_next_id(tasks),
        "title": title,
        "completed": False,
    }
    tasks.append(new_task)
    save_tasks(tasks, path)
    return new_task


def list_tasks(filepath: Optional[str] = None) -> List[Dict]:
    """Return all tasks from the file."""
    path = get_filepath(filepath)
    return load_tasks(path)


def complete_task(task_id: int, filepath: Optional[str] = None) -> Optional[Dict]:
    """Mark a task as completed.

    Returns the updated task if found, otherwise ``None``.
    """
    path = get_filepath(filepath)
    tasks = load_tasks(path)
    for task in tasks:
        if task["id"] == task_id:
            task["completed"] = True
            save_tasks(tasks, path)
            return task
    return None


def delete_task(task_id: int, filepath: Optional[str] = None) -> bool:
    """Delete a task by ID.

    Returns ``True`` if the task was removed, ``False`` if not found.
    """
    path = get_filepath(filepath)
    tasks = load_tasks(path)
    new_tasks = [t for t in tasks if t["id"] != task_id]
    if len(new_tasks) == len(tasks):
        return False
    save_tasks(new_tasks, path)
    return True


def main() -> None:
    """Parse command‑line arguments and execute the requested action."""
    parser = argparse.ArgumentParser(
        prog="todo",
        description="A simple CLI To-Do application.",
        add_help=False,
    )
    subparsers = parser.add_subparsers(dest="command", required=True)

    # add
    add_parser = subparsers.add_parser("add", help="Add a new task")
    add_parser.add_argument("title", help="Task description")

    # list
    subparsers.add_parser("list", help="List all tasks")

    # complete
    complete_parser = subparsers.add_parser("complete", help="Mark a task as done")
    complete_parser.add_argument("task_id", type=int, help="ID of the task")

    # delete
    delete_parser = subparsers.add_parser("delete", help="Delete a task")
    delete_parser.add_argument("task_id", type=int, help="ID of the task")

    # help
    subparsers.add_parser("help", help="Show this help message")

    args = parser.parse_args()

    if args.command == "help":
        parser.print_help()
        sys.exit(0)

    if args.command == "add":
        task = add_task(args.title)
        print(f"Added: [{task['id']}] {task['title']}")

    elif args.command == "list":
        tasks = list_tasks()
        if not tasks:
            print("No tasks found.")
            return
        print("Tasks:")
        for task in tasks:
            status = "✓" if task["completed"] else "✗"
            print(f"  [{task['id']}] {status} {task['title']}")

    elif args.command == "complete":
        task = complete_task(args.task_id)
        if task is None:
            print(f"Task with id {args.task_id} not found.")
        else:
            print(f"Completed: [{task['id']}] {task['title']}")

    elif args.command == "delete":
        if delete_task(args.task_id):
            print(f"Deleted task {args.task_id}.")
        else:
            print(f"Task with id {args.task_id} not found.")


if __name__ == "__main__":
    main()
```

## `test_todo.py` — Unit tests for all CRUD operations

```python
"""Unit tests for the todo module using a temporary JSON file."""

import json
import os
import tempfile
from pathlib import Path

import pytest

import todo  # the module we are testing


@pytest.fixture
def temp_file():
    """Create a temporary JSON file and provide its path."""
    with tempfile.NamedTemporaryFile(mode="w", suffix=".json", delete=False) as f:
        path = f.name
    yield path
    os.unlink(path)


def test_load_tasks_empty(temp_file):
    """Loading from a non‑existent or empty file returns an empty list."""
    assert todo.load_tasks(temp_file) == []

    # Also test with an existing empty file
    Path(temp_file).write_text("")
    assert todo.load_tasks(temp_file) == []


def test_save_and_load_tasks(temp_file):
    """Saving a list of tasks and loading it back returns the same list."""
    tasks = [
        {"id": 1, "title": "first", "completed": False},
        {"id": 2, "title": "second", "completed": True},
    ]
    todo.save_tasks(tasks, temp_file)
    loaded = todo.load_tasks(temp_file)
    assert loaded == tasks


def test_get_next_id():
    """The next ID is one more than the current maximum ID."""
    tasks = [{"id": 1}, {"id": 3}, {"id": 2}]
    assert todo.get_next_id(tasks) == 4

    assert todo.get_next_id([]) == 1


def test_add_task(temp_file):
    """Adding a task creates an entry with auto‑increment ID and completed=False."""
    task1 = todo.add_task("Buy milk", temp_file)
    assert task1["id"] == 1
    assert task1["title"] == "Buy milk"
    assert task1["completed"] is False

    # Second task gets ID 2
    task2 = todo.add_task("Do laundry", temp_file)
    assert task2["id"] == 2

    # Verify on disk
    tasks = todo.load_tasks(temp_file)
    assert len(tasks) == 2


def test_list_tasks(temp_file):
    """List returns all tasks currently stored."""
    # Initially empty
    assert todo.list_tasks(temp_file) == []

    todo.add_task("First", temp_file)
    todo.add_task("Second", temp_file)
    tasks = todo.list_tasks(temp_file)
    assert len(tasks) == 2


def test_complete_task_found(temp_file):
    """Completing an existing task sets completed to True."""
    todo.add_task("Test task", temp_file)
    updated = todo.complete_task(1, temp_file)
    assert updated is not None
    assert updated["completed"] is True

    # Verify on disk
    tasks = todo.load_tasks(temp_file)
    assert tasks[0]["completed"] is True


def test_complete_task_not_found(temp_file):
    """Completing a non‑existent task returns None."""
    assert todo.complete_task(999, temp_file) is None


def test_delete_task_found(temp_file):
    """Deleting an existing task removes it and returns True."""
    todo.add_task("To delete", temp_file)
    result = todo.delete_task(1, temp_file)
    assert result is True
    assert todo.list_tasks(temp_file) == []


def test_delete_task_not_found(temp_file):
    """Deleting a non‑existent task returns False."""
    assert todo.delete_task(42, temp_file) is False


def test_integration(temp_file):
    """Full workflow: add, list, complete, delete."""
    todo.add_task("Alpha", temp_file)
    todo.add_task("Beta", temp_file)
    todo.add_task("Gamma", temp_file)

    assert len(todo.list_tasks(temp_file)) == 3

    # Complete middle task
    todo.complete_task(2, temp_file)
    tasks = todo.list_tasks(temp_file)
    assert tasks[1]["completed"] is True

    # Delete last task
    todo.delete_task(3, temp_file)
    tasks = todo.list_tasks(temp_file)
    assert len(tasks) == 2
    assert tasks[-1]["id"] == 1  # only Alpha remains with id=1 (note: no re‑ordering)

    # Add another task – id should now be 4 (since max id was 2)
    todo.add_task("Delta", temp_file)
    tasks = todo.list_tasks(temp_file)
    assert tasks[-1]["id"] == 4


def test_env_var_overrides_default(monkeypatch, temp_file):
    """The environment variable TODO_FILE is respected."""
    monkeypatch.setenv("TODO_FILE", temp_file)
    # Now calls without explicit filepath should use the env var
    task = todo.add_task("From env")
    assert task["title"] == "From env"
    assert todo.list_tasks() == [task]
```

## Usage examples

Once both files are in the same directory, you can either run the CLI directly or import the module programmatically.

**1. Command‑line interface**:

```bash
$ python todo.py add "Buy groceries"
Added: [1] Buy groceries

$ python todo.py add "Write tests"
Added: [2] Write tests

$ python todo.py list
Tasks:
  [1] ✗ Buy groceries
  [2] ✗ Write tests

$ python todo.py complete 1
Completed: [1] Buy groceries

$ python todo.py list
Tasks:
  [1] ✓ Buy groceries
  [2] ✗ Write tests

$ python todo.py delete 2
Deleted task 2.

$ python todo.py help
usage: todo [-h] {add,list,complete,delete,help} ...
...
```

**2. Import as a module**:

```python
import todo

todo.add_task("Learn Python")
for task in todo.list_tasks():
    print(task)
```

**3. Running the tests**:

```bash
$ pytest test_todo.py -v
...
13 passed in 0.12s
```
