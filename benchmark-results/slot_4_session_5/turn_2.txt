### `todo.py`
```python
#!/usr/bin/env python3
"""Simple To-Do CLI with Priority & Due Date support."""

import argparse
import json
import sys
from dataclasses import dataclass, asdict
from datetime import date
from typing import List

TASKS_FILE = "tasks.json"


@dataclass
class Task:
    title: str
    completed: bool = False
    priority: str = "medium"  # low, medium, high
    due_date: str = ""        # YYYY-MM-DD or empty string


def load_tasks() -> List[Task]:
    """Load tasks from JSON file with backward-compatible migration."""
    try:
        with open(TASKS_FILE, "r") as f:
            data = json.load(f)
    except FileNotFoundError:
        return []

    tasks = []
    for item in data:
        tasks.append(Task(
            title=item.get("title", ""),
            completed=item.get("completed", False),
            priority=item.get("priority", "medium"),
            due_date=item.get("due_date", "") or ""
        ))
    return tasks


def save_tasks(tasks: List[Task]):
    """Save tasks to JSON file."""
    with open(TASKS_FILE, "w") as f:
        json.dump([asdict(t) for t in tasks], f, indent=2)


def cmd_add(args):
    """Add a new task."""
    tasks = load_tasks()
    tasks.append(Task(
        title=args.title,
        completed=False,
        priority=args.priority,
        due_date=args.due_date or ""
    ))
    save_tasks(tasks)
    print(f"✅ Added: {args.title} (Priority: {args.priority}, Due: {args.due_date or 'None'})")


def cmd_list(args):
    """List tasks with optional filtering and sorting."""
    tasks = load_tasks()

    # Filtering
    if args.priority:
        tasks = [t for t in tasks if t.priority == args.priority]
    
    if args.due_before:
        try:
            cutoff = date.fromisoformat(args.due_before)
            tasks = [
                t for t in tasks 
                if t.due_date and date.fromisoformat(t.due_date) < cutoff
            ]
        except ValueError:
            print("❌ Error: Invalid date format for --due-before. Use YYYY-MM-DD.", file=sys.stderr)
            sys.exit(1)

    # Sorting
    if args.sort == "due_date":
        tasks.sort(key=lambda t: t.due_date or "9999-12-31")
    elif args.sort == "priority":
        priority_order = {"low": 0, "medium": 1, "high": 2}
        tasks.sort(key=lambda t: priority_order.get(t.priority, 1))

    if not tasks:
        print("📭 No tasks found matching criteria.")
        return

    # Table rendering
    print(f"{'ID':<4} {'Title':<25} {'Priority':<10} {'Due Date':<12} {'Done'}")
    print("-" * 60)
    for idx, task in enumerate(tasks, 1):
        status = "✅" if task.completed else "⬜"
        due = task.due_date if task.due_date else "-"
        print(f"{idx:<4} {task.title:<25} {task.priority:<10} {due:<12} {status}")


def main():
    parser = argparse.ArgumentParser(
        description="A simple CLI to-do app with priority and due date support."
    )
    subparsers = parser.add_subparsers(dest="command", help="Available commands")

    # Add command
    add_parser = subparsers.add_parser("add", help="Add a new task")
    add_parser.add_argument("title", help="Task title")
    add_parser.add_argument(
        "--priority", choices=["low", "medium", "high"], default="medium",
        help="Task priority (default: medium)"
    )
    add_parser.add_argument(
        "--due", dest="due_date",
        help="Due date in YYYY-MM-DD format"
    )

    # List command
    list_parser = subparsers.add_parser("list", help="List tasks with filtering and sorting")
    list_parser.add_argument(
        "--priority", choices=["low", "medium", "high"],
        help="Filter by priority level"
    )
    list_parser.add_argument(
        "--due-before",
        help="Filter tasks due before YYYY-MM-DD"
    )
    list_parser.add_argument(
        "--sort", choices=["due_date", "priority"],
        help="Sort tasks by field"
    )

    args = parser.parse_args()
    
    if args.command == "add":
        cmd_add(args)
    elif args.command == "list":
        cmd_list(args)
    else:
        parser.print_help()


if __name__ == "__main__":
    main()
```

### `test_todo.py`
```python
import unittest
import json
import os
import sys
import argparse
from unittest.mock import patch, mock_open
from io import StringIO

# Import functions from todo module
from todo import Task, load_tasks, save_tasks, cmd_list, TASKS_FILE


class TestTodoApp(unittest.TestCase):
    def setUp(self):
        self.test_file = "test_tasks.json"
        # Temporarily override the tasks file path
        import todo
        todo.TASKS_FILE = self.test_file

    def tearDown(self):
        if os.path.exists(self.test_file):
            os.remove(self.test_file)

    def test_backward_compatibility_migration(self):
        """Old files without priority/due_date should default correctly."""
        old_data = [
            {"title": "Buy milk", "completed": False},
            {"title": "Finish report", "completed": True}
        ]
        with open(self.test_file, "w") as f:
            json.dump(old_data, f)

        tasks = load_tasks()
        self.assertEqual(len(tasks), 2)
        self.assertEqual(tasks[0].priority, "medium")
        self.assertEqual(tasks[0].due_date, "")
        self.assertFalse(tasks[0].completed)
        self.assertTrue(tasks[1].completed)

    def test_filter_by_priority(self):
        """list --priority high should only return high priority tasks."""
        tasks = [
            Task("Low task", priority="low"),
            Task("High task 1", priority="high"),
            Task("Med task", priority="medium"),
            Task("High task 2", priority="high")
        ]
        args = argparse.Namespace(priority="high", due_before=None, sort=None)
        
        with patch("todo.save_tasks"), patch("todo.load_tasks", return_value=tasks):
            old_stdout = sys.stdout
            sys.stdout = StringIO()
            cmd_list(args)
            output = sys.stdout.getvalue()
            sys.stdout = old_stdout

        self.assertIn("High task 1", output)
        self.assertIn("High task 2", output)
        self.assertNotIn("Low task", output)
        self.assertNotIn("Med task", output)

    def test_filter_by_due_before(self):
        """list --due-before 2026-01-01 should exclude future tasks."""
        tasks = [
            Task("Early", due_date="2025-01-01"),
            Task("Late", due_date="2026-12-31"),
            Task("Mid", due_date="2025-06-15")
        ]
        args = argparse.Namespace(priority=None, due_before="2026-01-01", sort=None)
        
        with patch("todo.save_tasks"), patch("todo.load_tasks", return_value=tasks):
            old_stdout = sys.stdout
            sys.stdout = StringIO()
            cmd_list(args)
            output = sys.stdout.getvalue()
            sys.stdout = old_stdout

        self.assertIn("Early", output)
        self.assertIn("Mid", output)
        self.assertNotIn("Late", output)

    def test_sort_by_priority(self):
        """list --sort priority should order low < medium < high."""
        tasks = [
            Task("High P", priority="high"),
            Task("Low P", priority="low"),
            Task("Med P", priority="medium")
        ]
        args = argparse.Namespace(priority=None, due_before=None, sort="priority")
        
        with patch("todo.save_tasks"), patch("todo.load_tasks", return_value=tasks):
            old_stdout = sys.stdout
            sys.stdout = StringIO()
            cmd_list(args)
            output = sys.stdout.getvalue()
            sys.stdout = old_stdout

        lines = [l for l in output.strip().split('\n') if l.startswith(('High', 'Low', 'Med'))]
        self.assertEqual(lines[0].split()[1], "High P")
        self.assertEqual(lines[1].split()[1], "Med P")
        self.assertEqual(lines[2].split()[1], "Low P")

    def test_sort_by_due_date(self):
        """list --sort due_date should order by date, leaving empty dates at the end."""
        tasks = [
            Task("Far", due_date="2026-05-01"),
            Task("Soon", due_date="2025-01-10"),
            Task("No Date")
        ]
        args = argparse.Namespace(priority=None, due_before=None, sort="due_date")
        
        with patch("todo.save_tasks"), patch("todo.load_tasks", return_value=tasks):
            old_stdout = sys.stdout
            sys.stdout = StringIO()
            cmd_list(args)
            output = sys.stdout.getvalue()
            sys.stdout = old_stdout

        lines = [l for l in output.strip().split('\n') if l.startswith(('Far', 'Soon', 'No'))]
        self.assertEqual(lines[0].split()[1], "Soon")
        self.assertEqual(lines[1].split()[1], "Far")
        self.assertEqual(lines[2].split()[1], "No Date")

    def test_invalid_due_before_date(self):
        """list --due-before with invalid format should exit with error."""
        args = argparse.Namespace(priority=None, due_before="not-a-date", sort=None)
        
        with patch("todo.load_tasks", return_value=[]):
            with self.assertRaises(SystemExit) as ctx:
                cmd_list(args)
            self.assertEqual(ctx.exception.code, 1)


if __name__ == "__main__":
    unittest.main()
```

### Key Features & Design Decisions

1. **Backward Compatibility**: `load_tasks()` uses `.get()` with sensible defaults. Old JSON files missing `priority` or `due_date` fields will automatically migrate to `medium` priority and empty due dates without crashing.
2. **Filtering Logic**: 
   - `--priority` filters exact matches.
   - `--due-before` parses dates safely and only includes tasks with a valid due date strictly before the cutoff.
3. **Sorting Logic**:
   - `--sort priority` maps `"low": 0, "medium": 1, "high": 2` for correct ascending order.
   - `--sort due_date` uses `"9999-12-31"` as a fallback key so tasks without dates sink to the bottom.
4. **CLI Help**: `argparse` automatically generates structured help text. Run `python todo.py --help` or `python todo.py list --help` to see the documented options.
5. **Testing**: 
   - Uses `unittest` with `mock_open`/`patch` to avoid file I/O side effects.
   - Tests cover migration, filtering, sorting, and error handling.
   - `argparse.Namespace` is used to cleanly pass CLI arguments to command functions.
