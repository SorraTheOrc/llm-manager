### 💾 `todo/core.py`
```python
from dataclasses import dataclass, field
from typing import List, Dict, Optional
import json
import os
from datetime import datetime

@dataclass
class Task:
    id: int
    title: str
    completed: bool = False
    created_at: str = field(default_factory=lambda: datetime.now().isoformat())
    due_date: Optional[str] = None

class TodoStore:
    def __init__(self, db_path: str = "todos.json"):
        self.db_path = db_path
        self.tasks: Dict[int, Task] = {}
        self.next_id = 1
        self.load()

    def load(self):
        if os.path.exists(self.db_path):
            with open(self.db_path, 'r') as f:
                data = json.load(f)
                self.tasks = {int(k): Task(**v) for k, v in data.items()}
                self.next_id = max(self.tasks.keys()) + 1 if self.tasks else 1

    def save(self):
        with open(self.db_path, 'w') as f:
            json.dump({str(k): v.__dict__ for k, v in self.tasks.items()}, f, indent=2)

    def add(self, title: str, due_date: Optional[str] = None) -> Task:
        task = Task(id=self.next_id, title=title, due_date=due_date)
        self.tasks[task.id] = task
        self.next_id += 1
        self.save()
        return task

    def toggle(self, task_id: int) -> Optional[Task]:
        if task_id in self.tasks:
            self.tasks[task_id].completed = not self.tasks[task_id].completed
            self.save()
            return self.tasks[task_id]
        return None

    def delete(self, task_id: int) -> bool:
        if task_id in self.tasks:
            del self.tasks[task_id]
            self.save()
            return True
        return False

    def get_all(self, filter_status: Optional[str] = None) -> List[Task]:
        tasks = list(self.tasks.values())
        if filter_status == "pending":
            tasks = [t for t in tasks if not t.completed]
        elif filter_status == "completed":
            tasks = [t for t in tasks if t.completed]
        return tasks

    def get_daily_summary(self, date_str: Optional[str] = None) -> Dict:
        if date_str is None:
            date_str = datetime.now().strftime("%Y-%m-%d")
        due_today = [t for t in self.tasks.values() if t.due_date == date_str]
        all_pending = [t for t in self.tasks.values() if not t.completed]
        return {
            "date": date_str,
            "due_today": due_today,
            "pending_count": len(all_pending),
            "completed_count": sum(1 for t in self.tasks.values() if t.completed)
        }
```

from fastapi.responses import HTMLResponse, JSONResponse
from todo.core import TodoStore

store = TodoStore()

@app.get("/", response_class=HTMLResponse)
async def index(request: Request):
    status = request.query_params.get("status", "all")
    tasks = store.get_all(filter_status=status if status != "all" else None)
    rows = ""
    for t in tasks:
        cls = "completed" if t.completed else ""
        icon = "✅" if t.completed else "⏳"
        rows += f"""<tr class="{cls}">
            <td>{t.id}</td><td>{t.title}</td><td>{t.due_date or 'N/A'}</td><td>{icon}</td>
            <td class="actions">
                <form action="/toggle/{t.id}" method="POST" style="display:inline"><button type="submit">Toggle</button></form>
                <form action="/delete/{t.id}" method="POST" style="display:inline"><button type="submit" onclick="return confirm('Delete?')">Delete</button></form>
            </td>
        </tr>"""
    empty = '<div class="empty-state">No tasks found. Add one above!</div>' if not tasks else ""
    return TASKS_TEMPLATE.format(
        filter_all="active" if status=="all" else "",
        filter_pending="active" if status=="pending" else "",
        filter_completed="active" if status=="completed" else "",
        rows=rows,
        empty_msg=empty
    )

@app.post("/add")
async def add_task(request: Request):
    try:
        if request.headers.get("content-type", "").startswith("application/json"):
            body = await request.json()
            title = body.get("title")
            due_date = body.get("due_date")
        else:
            form = await request.form()
            title = form.get("title")
            due_date = form.get("due_date")
        if not title:
            return JSONResponse({"status": "error", "message": "Title is required"}, status_code=400)
        store.add(title, due_date)
        return {"status": "ok", "message": "Task added"}
    except Exception as e:
        return JSONResponse({"status": "error", "message": str(e)}, status_code=500)

@app.post("/toggle/{task_id}")
async def toggle_task(task_id: int):
    task = store.toggle(task_id)
    if task:
        return {"status": "ok", "message": f"Toggled task {task_id}"}
    return JSONResponse({"status": "error", "message": "Task not found"}, status_code=404)

@app.post("/delete/{task_id}")
async def delete_task(task_id: int):
    if store.delete(task_id):
        return {"status": "ok", "message": f"Deleted task {task_id}"}
    return JSONResponse({"status": "error", "message": "Task not found"}, status_code=404)

@app.get("/daily", response_class=HTMLResponse)
async def daily_summary():
    from datetime import datetime
    today = datetime.now().strftime("%Y-%m-%d")
    summary = store.get_daily_summary(today)
    tasks_html = ""
    for t in summary["due_today"]:
        cls = "completed" if t.completed else ""
        tasks_html += f"<li class='{cls}'>{t.title} (ID: {t.id})</li>"
    if not tasks_html:
        tasks_html = "<p>No tasks due today.</p>"
    return f"""<!DOCTYPE html>
    <html><head><title>Daily Summary</title>
    <style>body{{font-family:sans-serif;max-width:600px;margin:2rem auto;padding:0 1rem}}.stat{{margin:0.5rem 0;font-size:1.1rem}}</style>
    </head><body>
        <h1>📅 Daily Summary</h1>
        <p class="stat">Date: {summary['date']}</p>
        <p class="stat">Pending: {summary['pending_count']} | Completed: {summary['completed_count']}</p>
        <h2>Due Today:</h2>
        <ul>{tasks_html}</ul>
        <a href="/">← Back to Tasks</a>
    </body></html>"""

### 🖥️ `todo/cli.py`
```python
import click
import uvicorn
from todo.core import TodoStore
from todo.web import app

store = TodoStore()

@click.group()
def cli():
    """Lightweight To-Do CLI & Web UI"""
    pass

@cli.command()
@click.argument("title")
@click.option("--due", "-d", help="Due date (YYYY-MM-DD)")
def add(title, due):
    store.add(title, due)
    click.echo(f"✅ Added: {title}")

@cli.command("list")
@click.option("--status", "-s", type=click.Choice(["pending", "completed"]), help="Filter by status")
def list_tasks(status):
    tasks = store.get_all(filter_status=status)
    if not tasks:
        click.echo("📭 No tasks found.")
        return
    for t in tasks:
        check = "✅" if t.completed else "⏳"
        click.echo(f"{check} [{t.id}] {t.title} (Due: {t.due_date or 'N/A'})")

@cli.command()
@click.argument("task_id", type=int)
def done(task_id):
    task = store.toggle(task_id)
    if task:
        click.echo(f"🔄 Toggled: {task.title}")
    else:
        click.echo(f"❌ Task {task_id} not found.")

@cli.command()
@click.argument("task_id", type=int)
def delete(task_id):
    if store.delete(task_id):
        click.echo(f"🗑️ Deleted task {task_id}")
    else:
        click.echo(f"❌ Task {task_id} not found.")

@cli.command()
@click.option("--date", "-d", help="Date to summarize (YYYY-MM-DD), defaults to today")
def daily(date):
    summary = store.get_daily_summary(date)
    click.echo(f"\n📅 Daily Summary for {summary['date']}")
    click.echo(f"📊 Pending: {summary['pending_count']}, Completed: {summary['completed_count']}")
    if summary['due_today']:
        click.echo("\n📌 Due Today:")
        for t in summary['due_today']:
            click.echo(f"  {'✅' if t.completed else '⏳'} [{t.id}] {t.title}")
    else:
        click.echo("\nNo tasks due today.")
    click.echo()

@cli.command()
@click.option("--port", "-p", default=8080, help="Port to run the web server on")
def serve(port):
    """Start the web UI server"""
    click.echo(f"🚀 Starting Todo Web UI on http://localhost:{port}")
    try:
        uvicorn.run(app, host="0.0.0.0", port=port, log_level="warning")
    except KeyboardInterrupt:
        click.echo("\n👋 Server stopped gracefully.")
    except OSError as e:
        if "Address already in use" in str(e) or "Errno 98" in str(e):
            click.echo(f"❌ Port {port} is already in use. Try a different port or kill the existing process.")
        else:
            click.echo(f"❌ Server error: {e}")
        raise SystemExit(1)

if __name__ == "__main__":
    cli()
```

### 🛠️ Usage Instructions

1. **Install dependencies:**
   ```bash
   pip install -r requirements.txt
   ```

2. **Run CLI commands:**
   ```bash
   python -m todo add "Buy groceries" --due 2023-12-25
   python -m todo list
   python -m todo done 1
   python -m todo daily
   ```

3. **Start Web UI:**
   ```bash
   python -m todo serve --port 8080
   ```
   Open `http://localhost:8080` in your browser. The CLI and web UI share the same `todos.json` file, so they stay perfectly in sync.

4. **Run Tests:**
   ```bash
   pytest tests/test_web.py -v
   ```
