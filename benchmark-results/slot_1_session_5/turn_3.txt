#!/usr/bin/env python3
"""
todo.py - A compact CLI To-Do application with daily/weekly overviews.
"""

import argparse
import json
import os
import sys
import tempfile
from datetime import datetime, timedelta
from collections import defaultdict

DEFAULT_FILE = "tasks.json"


# ================= STORAGE =================
def load_tasks(filepath):
    if not os.path.exists(filepath):
        return []
    with open(filepath, "r") as f:
        return json.load(f)

def save_tasks(tasks, filepath):
    with open(filepath, "w") as f:
        json.dump(tasks, f, indent=2)


# ================= EXISTING COMMANDS =================
def add_task(args):
    tasks = load_tasks(args.file)
    due = datetime.strptime(args.due, "%Y-%m-%d").date() if args.due else None
    new_task = {
        "id": max([t["id"] for t in tasks], default=0) + 1,
        "title": args.title,
        "due": due.isoformat() if due else None,
        "completed": False,
        "completed_at": None
    }
    tasks.append(new_task)
    save_tasks(tasks, args.file)
    print(f"✅ Added: {args.title}")

def list_tasks(args):
    tasks = load_tasks(args.file)
    if not tasks:
        print("📭 No tasks found.")
        return
    for t in tasks:
        status = "✅" if t["completed"] else "⬜"
        due_str = f" (due: {t['due']})" if t['due'] else ""
        print(f"{status} {t['id']}. {t['title']}{due_str}")

def complete_task(args):
    tasks = load_tasks(args.file)
    for t in tasks:
        if t["id"] == args.id:
            t["completed"] = True
            t["completed_at"] = datetime.today().date().isoformat()
            save_tasks(tasks, args.file)
            print(f"🏁 Completed: {t['title']}")
            return
    print("❌ Task not found.")

def delete_task(args):
    tasks = load_tasks(args.file)
    new_tasks = [t for t in tasks if t["id"] != args.id]
    if len(new_tasks) == len(tasks):
        print("❌ Task not found.")
    else:
        save_tasks(new_tasks, args.file)
        print(f"🗑️ Deleted task {args.id}")

def clear_tasks(args):
    save_tasks([], args.file)
    print("🧹 All tasks cleared.")


# ================= NEW COMMANDS =================
def daily_command(args):
    tasks = load_tasks(args.file)
    today = datetime.today().date()
    today_str = today.isoformat()

    # Only consider tasks with a due date
    dated_tasks = [t for t in tasks if t.get("due")]

    due_today = [t for t in dated_tasks if t["due"] == today_str]
    overdue = [t for t in dated_tasks if t["due"] < today_str and not t["completed"]]
    completed_today = [t for t in tasks if t.get("completed_at") and t["completed_at"] == today_str]

    print(f"📅 Daily Summary ({today_str})")
    print(f"✅ Completed today: {len(completed_today)}")
    print(f"⏰ Due today: {len(due_today)}")
    print(f"⚠️ Overdue: {len(overdue)}")
    print("---")

    if due_today:
        print("Tasks due today:")
        for t in due_today:
            status = "✅" if t["completed"] else "⬜"
            print(f"  {status} {t['title']}")
        print("---")

    if overdue:
        print("Overdue tasks:")
        for t in overdue:
            print(f"  ⚠️ {t['title']} (due: {t['due']})")
        print("---")

def week_command(args):
    tasks = load_tasks(args.file)
    today = datetime.today().date()
    week_end = today + timedelta(days=6)
    today_str = today.isoformat()
    end_str = week_end.isoformat()

    # Only consider tasks with a due date in the week
    week_tasks = [
        t for t in tasks
        if t.get("due") and today_str <= t["due"] <= end_str
    ]

    by_date = defaultdict(list)
    for t in week_tasks:
        by_date[t["due"]].append(t)

    print(f"📅 Weekly Overview ({today_str} to {end_str})")
    print(f"📊 Total tasks this week: {len(week_tasks)}")
    print("---")

    if not by_date:
        print("📭 No tasks scheduled this week.")
    else:
        for date_str in sorted(by_date.keys()):
            print(f"📆 {date_str}")
            for t in by_date[date_str]:
                status = "✅" if t["completed"] else "⬜"
                print(f"  {status} {t['title']}")
            print("---")


# ================= CLI SETUP =================
def main():
    parser = argparse.ArgumentParser(description="CLI To-Do Application")
    parser.add_argument("--file", default=DEFAULT_FILE, help="Path to tasks JSON file")
    subparsers = parser.add_subparsers(dest="command", help="Available commands")

    # Add
    p_add = subparsers.add_parser("add", help="Add a new task")
    p_add.add_argument("title", help="Task title")
    p_add.add_argument("--due", default=None, help="Due date (YYYY-MM-DD)")

    # List
    subparsers.add_parser("list", help="List all tasks")

    # Complete
    p_comp = subparsers.add_parser("complete", help="Mark a task as completed")
    p_comp.add_argument("id", type=int, help="Task ID")

    # Delete
    p_del = subparsers.add_parser("delete", help="Delete a task")
    p_del.add_argument("id", type=int, help="Task ID")

    # Clear
    subparsers.add_parser("clear", help="Remove all tasks")

    # Daily
    subparsers.add_parser("daily", help="Show today's summary")

    # Week
    subparsers.add_parser("week", help="Show weekly overview")

    args = parser.parse_args()
    if not args.command:
        parser.print_help()
        sys.exit(1)

    commands = {
        "add": add_task,
        "list": list_tasks,
        "complete": complete_task,
        "delete": delete_task,
        "clear": clear_tasks,
        "daily": daily_command,
        "week": week_command,
    }

    commands[args.command](args)

if __name__ == "__main__":
    main()

### How to Use
1. **Save** the code as `todo.py`

3. **Run commands**:
   ```bash
   python todo.py add "Buy groceries" --due 2024-06-20
   python todo.py add "Finish report"
   python todo.py daily
   python todo.py week
   ```
4. **Run tests**:
   ```bash
   python todo.py test
   ```

- ✅ **Date filtering**: Tasks without a `due` date are strictly excluded from both views.
- ✅ **Compact format**: Clean, emoji-enhanced layout optimized for quick morning scanning.
