```python
#!/usr/bin/env python3
"""
Extended To-Do application with priority and due date.

Usage:
    todo.py add <title> [--priority <level>] [--due-date <YYYY-MM-DD>]
    todo.py list [--priority <level>] [--due-before <YYYY-MM-DD>] [--sort <field>]
    todo.py complete <id>
    todo.py delete <id>
    todo.py help
"""

import argparse
import json
import os
import sys
from datetime import datetime, date
from dataclasses import dataclass, field
from typing import List, Optional, Callable

# --------------- Data Model ---------------

@dataclass
class Task:
    title: str
    completed: bool = False
    priority: str = 'low'          # low, medium, high
    due_date: str = ''             # YYYY-MM-DD or empty
    id: int = 0

    def to_dict(self) -> dict:
        return {
            'id': self.id,
            'title': self.title,
            'completed': self.completed,
            'priority': self.priority,
            'due_date': self.due_date
        }

    @staticmethod
    def from_dict(d: dict) -> 'Task':
        # Migration: supply defaults if fields are missing
        return Task(
            id=d.get('id', 0),
            title=d.get('title', ''),
            completed=d.get('completed', False),
            priority=d.get('priority', 'low'),
            due_date=d.get('due_date', '')
        )

# --------------- Manager ---------------

class TaskManager:
    def __init__(self, filepath: str = 'tasks.json'):
        self.filepath = filepath
        self.tasks: List[Task] = []
        self._load()

    def _load(self):
        if not os.path.exists(self.filepath):
            self.tasks = []
            return
        try:
            with open(self.filepath, 'r') as f:
                data = json.load(f)
            self.tasks = [Task.from_dict(item) for item in data]
        except (json.JSONDecodeError, FileNotFoundError):
            self.tasks = []
        self._assign_ids()

    def _assign_ids(self):
        for i, t in enumerate(self.tasks, start=1):
            t.id = i

    def save(self):
        with open(self.filepath, 'w') as f:
            json.dump([t.to_dict() for t in self.tasks], f, indent=2)

    def add_task(self, title: str, priority: str = 'low', due_date: str = '') -> Task:
        task = Task(
            title=title,
            priority=priority,
            due_date=due_date,
            id=len(self.tasks) + 1
        )
        self.tasks.append(task)
        self._assign_ids()
        self.save()
        return task

    def complete_task(self, task_id: int) -> bool:
        for t in self.tasks:
            if t.id == task_id:
                t.completed = True
                self.save()
                return True
        return False

    def delete_task(self, task_id: int) -> bool:
        original_len = len(self.tasks)
        self.tasks = [t for t in self.tasks if t.id != task_id]
        if len(self.tasks) != original_len:
            self._assign_ids()
            self.save()
            return True
        return False

    def list_tasks(self, priority: Optional[str] = None,
                   due_before: Optional[str] = None,
                   sort_by: Optional[str] = None) -> List[Task]:
        tasks = self.tasks[:]

        # Filter
        if priority:
            tasks = [t for t in tasks if t.priority == priority]
        if due_before:
            try:
                limit = datetime.strptime(due_before, '%Y-%m-%d').date()
            except ValueError:
                print(f"Invalid date: {due_before}. Use YYYY-MM-DD.", file=sys.stderr)
                sys.exit(1)
            tasks = [t for t in tasks if t.due_date and datetime.strptime(t.due_date, '%Y-%m-%d').date() < limit]

        # Sort
        if sort_by:
            if sort_by == 'priority':
                priority_order = {'low': 0, 'medium': 1, 'high': 2}
                tasks.sort(key=lambda t: priority_order.get(t.priority, -1))
            elif sort_by == 'due_date':
                # Tasks without due date appear at the end
                tasks.sort(key=lambda t: (t.due_date == '', t.due_date))
            else:
                print(f"Invalid sort field: {sort_by}. Use 'priority' or 'due_date'.", file=sys.stderr)
                sys.exit(1)

        return tasks

# --------------- CLI ---------------

def format_task(task: Task) -> str:
    status = '[x]' if task.completed else '[ ]'
    return f"{task.id:<4} {status} {task.title:<40} {task.priority:<8} {task.due_date:<12}"

def print_help():
    help_text = """
Usage: todo.py <command> [options]

Commands:
  add <title> [--priority <low|medium|high>] [--due-date <YYYY-MM-DD>]
        Add a new task.
  list [--priority <level>] [--due-before <YYYY-MM-DD>] [--sort <field>]
        List tasks. Field can be 'priority' or 'due_date'.
  complete <id>       Mark task as completed.
  delete <id>         Delete a task.
  help                Show this help message.
"""
    print(help_text.strip())

def main():
    parser = argparse.ArgumentParser(description="To-Do Application", add_help=False)
    parser.add_argument('command', nargs='?', help='Command (add, list, complete, delete, help)')
    parser.add_argument('args', nargs=argparse.REMAINDER)

    # We'll parse known args first to extract command, then handle sub-commands manually
    # because argparse gets messy with subcommands. Simpler: parse command separately.
    known_args, remaining = parser.parse_known_args()
    command = known_args.command

    if not command or command == 'help':
        print_help()
        return

    manager = TaskManager()

    if command == 'add':
        # Usage: add <title> [--priority] [--due-date]
        add_parser = argparse.ArgumentParser()
        add_parser.add_argument('title', nargs='?')
        add_parser.add_argument('--priority', default='low', choices=['low', 'medium', 'high'])
        add_parser.add_argument('--due-date', default='', metavar='YYYY-MM-DD')
        args = add_parser.parse_args(remaining)
        if not args.title:
            print("Error: title is required for add command.", file=sys.stderr)
            sys.exit(1)
        task = manager.add_task(args.title, args.priority, args.due_date)
        print(f"Added task {task.id}: {task.title}")

    elif command == 'list':
        list_parser = argparse.ArgumentParser()
        list_parser.add_argument('--priority', default=None, choices=['low', 'medium', 'high'])
        list_parser.add_argument('--due-before', default=None, metavar='YYYY-MM-DD')
        list_parser.add_argument('--sort', default=None, choices=['priority', 'due_date'])
        args = list_parser.parse_args(remaining)
        tasks = manager.list_tasks(priority=args.priority, due_before=args.due_before, sort_by=args.sort)
        # Print header
        print(f"{'ID':<4} {'Status':<6} {'Title':<40} {'Priority':<8} {'Due Date':<12}")
        print('-' * 70)
        for t in tasks:
            print(format_task(t))

    elif command == 'complete':
        # Usage: complete <id>
        if not remaining:
            print("Error: task id required.", file=sys.stderr)
            sys.exit(1)
        try:
            task_id = int(remaining[0])
        except ValueError:
            print("Error: task id must be an integer.", file=sys.stderr)
            sys.exit(1)
        if manager.complete_task(task_id):
            print(f"Task {task_id} completed.")
        else:
            print(f"Task {task_id} not found.", file=sys.stderr)

    elif command == 'delete':
        # Usage: delete <id>
        if not remaining:
            print("Error: task id required.", file=sys.stderr)
            sys.exit(1)
        try:
            task_id = int(remaining[0])
        except ValueError:
            print("Error: task id must be an integer.", file=sys.stderr)
            sys.exit(1)
        if manager.delete_task(task_id):
            print(f"Task {task_id} deleted.")
        else:
            print(f"Task {task_id} not found.", file=sys.stderr)

    else:
        print(f"Unknown command: {command}", file=sys.stderr)
        print_help()
        sys.exit(1)

if __name__ == '__main__':
    main()
```

```python
import argparse
import sys
from datetime import datetime
from typing import List, Optional

from task import Task, VALID_PRIORITIES, DATE_FORMAT
from storage import load_tasks, save_tasks


def get_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="todo",
        description="A simple to‑do list with priority and due date support."
    )
    sub = parser.add_subparsers(dest="command", required=True)

    # --- add ---
    add_parser = sub.add_parser("add", help="Add a new task")
    add_parser.add_argument("title", help="Task title")
    add_parser.add_argument(
        "-p", "--priority", choices=VALID_PRIORITIES, default="medium",
        help="Priority level (default: medium)"
    )
    add_parser.add_argument(
        "-d", "--due-date", metavar="YYYY-MM-DD", default="",
        help="Due date (format: YYYY-MM-DD)"
    )

    # --- list ---
    list_parser = sub.add_parser("list", help="List tasks (with optional filters and sorting)")
    list_parser.add_argument(
        "--priority", choices=VALID_PRIORITIES,
        help="Filter by priority"
    )
    list_parser.add_argument(
        "--due-before", metavar="DATE",
        help="Show tasks with a due date strictly before this date (YYYY-MM-DD)"
    )
    list_parser.add_argument(
        "--sort", choices=["due_date", "priority"],
        help="Sort by due date (ascending) or priority (low < medium < high)"
    )

    # --- complete ---
    complete_parser = sub.add_parser("complete", help="Mark a task as completed")
    complete_parser.add_argument("index", type=int, help="Task number (from list)")

    return parser


def cmd_add(tasks: List[Task], args: argparse.Namespace) -> None:
    task = Task(
        title=args.title,
        priority=args.priority,
        due_date=args.due_date,
    )
    tasks.append(task)
    save_tasks(tasks)
    print(f'Added: "{task.title}"')


def cmd_list(tasks: List[Task], args: argparse.Namespace) -> None:
    # Filter
    if args.priority:
        tasks = [t for t in tasks if t.priority == args.priority]

    if args.due_before:
        try:
            limit = datetime.strptime(args.due_before, DATE_FORMAT).date()
        except ValueError:
            sys.exit("Error: --due-before must be a valid date (YYYY-MM-DD)")
        tasks = [
            t for t in tasks
            if t.due_date and datetime.strptime(t.due_date, DATE_FORMAT).date() < limit
        ]

    # Sort
    if args.sort == "due_date":
        def sort_key(t: Task) -> str:
            return t.due_date if t.due_date else "9999-12-31"  # tasks without due date go last
        tasks.sort(key=sort_key)
    elif args.sort == "priority":
        order = {"low": 0, "medium": 1, "high": 2}
        tasks.sort(key=lambda t: order[t.priority])

    # Display
    if not tasks:
        print("No tasks.")
        return

    # Header
    print(f"{'#':<4} {'Completed':<10} {'Priority':<8} {'Due Date':<12} Title")
    print("-" * 60)
    for i, task in enumerate(tasks, 1):
        done = "✓" if task.completed else " "
        due = task.due_date if task.due_date else "-"
        print(f"{i:<4} {done:<10} {task.priority:<8} {due:<12} {task.title}")


def cmd_complete(tasks: List[Task], args: argparse.Namespace) -> None:
    if args.index < 1 or args.index > len(tasks):
        sys.exit(f"Error: index {args.index} out of range.")
    tasks[args.index - 1].completed = True
    save_tasks(tasks)
    print(f'Completed: "{tasks[args.index - 1].title}"')


def main() -> None:
    parser = get_parser()
    args = parser.parse_args()

    tasks = load_tasks()

    if args.command == "add":
        cmd_add(tasks, args)
    elif args.command == "list":
        cmd_list(tasks, args)
    elif args.command == "complete":
        cmd_complete(tasks, args)
    else:
        parser.print_help()


if __name__ == "__main__":
    main()
```

# ------- Task tests -------
class TestTask:
    def test_defaults(self):
        t = Task("Test")
        assert t.title == "Test"
        assert t.completed is False
        assert t.priority == "medium"
        assert t.due_date == ""

    def test_to_from_dict(self):
        t = Task("Work", completed=False, priority="high", due_date="2026-08-15")
        d = t.to_dict()
        t2 = Task.from_dict(d)
        assert t == t2

    def test_migration(self):
        # simulate old task without priority and due_date
        data = {"title": "old", "completed": True}
        t = Task.from_dict(data)
        assert t.priority == "medium"
        assert t.due_date == ""
        assert t.completed is True
