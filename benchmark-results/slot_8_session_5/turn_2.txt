### `todo_app.py`
```python
#!/usr/bin/env python3
"""
A CLI To-Do application with priority, due dates, filtering, and sorting.
Supports backward compatibility with older task files via migration on read.
"""

import argparse
import json
import os
import sys
from dataclasses import dataclass, asdict
from typing import List, Optional

TASKS_FILE = "tasks.json"
PRIORITY_ORDER = {"low": 1, "medium": 2, "high": 3}


@dataclass
class Task:
    id: int
    title: str
    completed: bool
    priority: str  # "low", "medium", "high"
    due_date: str  # "YYYY-MM-DD" or empty string

    def to_dict(self) -> dict:
        return asdict(self)

    @classmethod
    def from_dict(cls, data: dict) -> "Task":
        """Migrate old task dicts to the new schema on read."""
        return cls(
            id=data["id"],
            title=data["title"],
            completed=data.get("completed", False),
            priority=data.get("priority", "medium"),
            due_date=data.get("due_date", ""),
        )


def load_tasks() -> List[Task]:
    """Load tasks from JSON, applying backward-compatibility migration."""
    if not os.path.exists(TASKS_FILE):
        return []
    with open(TASKS_FILE, "r", encoding="utf-8") as f:
        data = json.load(f)
    return [Task.from_dict(t) for t in data]


def save_tasks(tasks: List[Task]) -> None:
    """Persist tasks to JSON."""
    with open(TASKS_FILE, "w", encoding="utf-8") as f:
        json.dump([t.to_dict() for t in tasks], f, indent=2)


def add_task(title: str) -> None:
    tasks = load_tasks()
    new_id = max((t.id for t in tasks), default=0) + 1
    task = Task(id=new_id, title=title, completed=False, priority="medium", due_date="")
    tasks.append(task)
    save_tasks(tasks)
    print(f"Added task #{new_id}: {title}")


def complete_task(task_id: int) -> None:
    tasks = load_tasks()
    for t in tasks:
        if t.id == task_id:
            t.completed = True
            save_tasks(tasks)
            print(f"Task #{task_id} marked as completed.")
            return
    print(f"Task #{task_id} not found.")


def delete_task(task_id: int) -> None:
    tasks = load_tasks()
    tasks = [t for t in tasks if t.id != task_id]
    save_tasks(tasks)
    print(f"Task #{task_id} deleted.")


def apply_filters(tasks: List[Task], priority: Optional[str] = None,
                  due_before: Optional[str] = None, include_completed: bool = False) -> List[Task]:
    if not include_completed:
        tasks = [t for t in tasks if not t.completed]
    if priority:
        tasks = [t for t in tasks if t.priority == priority]
    if due_before:
        # Lexicographical comparison works for YYYY-MM-DD format
        tasks = [t for t in tasks if t.due_date and t.due_date < due_before]
    return tasks


def apply_sorting(tasks: List[Task], sort_by: Optional[str] = None) -> List[Task]:
    if sort_by == "due_date":
        # Sort by date ascending; empty dates sort last
        tasks.sort(key=lambda t: (not t.due_date, t.due_date))
    elif sort_by == "priority":
        # Sort by priority level ascending (low -> high)
        tasks.sort(key=lambda t: PRIORITY_ORDER.get(t.priority, 2))
    return tasks


def list_tasks(priority: Optional[str] = None, due_before: Optional[str] = None,
               sort_by: Optional[str] = None, include_completed: bool = False) -> None:
    tasks = load_tasks()
    tasks = apply_filters(tasks, priority, due_before, include_completed)
    tasks = apply_sorting(tasks, sort_by)

    if not tasks:
        print("No tasks found.")
        return

    print(f"{'ID':<4} {'Title':<25} {'Completed':<10} {'Priority':<10} {'Due Date':<12}")
    print("-" * 65)
    for t in tasks:
        status = "Yes" if t.completed else "No"
        print(f"{t.id:<4} {t.title:<25} {status:<10} {t.priority:<10} {t.due_date:<12}")


def main() -> None:
    parser = argparse.ArgumentParser(description="To-Do List CLI")
    subparsers = parser.add_subparsers(dest="command", help="Available commands")

    # add
    add_parser = subparsers.add_parser("add", help="Add a new task")
    add_parser.add_argument("title", help="Task title")

    # complete
    complete_parser = subparsers.add_parser("complete", help="Mark a task as completed")
    complete_parser.add_argument("id", type=int, help="Task ID")

    # list
    list_parser = subparsers.add_parser("list", help="List tasks")
    list_parser.add_argument("--priority", choices=["low", "medium", "high"],
                             help="Filter by priority level")
    list_parser.add_argument("--due-before", help="Filter tasks due before YYYY-MM-DD")
    list_parser.add_argument("--sort", choices=["due_date", "priority"],
                             help="Sort by field (due_date or priority)")
    list_parser.add_argument("--include-completed", action="store_true",
                             help="Include completed tasks in output")

    # delete
    delete_parser = subparsers.add_parser("delete", help="Delete a task")
    delete_parser.add_argument("id", type=int, help="Task ID")

    args = parser.parse_args()
    if not args.command:
        parser.print_help()
        return

    if args.command == "add":
        add_task(args.title)
    elif args.command == "complete":
        complete_task(args.id)
    elif args.command == "list":
        list_tasks(
            priority=args.priority,
            due_before=args.due_before,
            sort_by=args.sort,
            include_completed=args.include_completed,
        )
    elif args.command == "delete":
        delete_task(args.id)


if __name__ == "__main__":
    main()
```

### `test_todo.py`
```python
#!/usr/bin/env python3
import unittest
from todo_app import Task, apply_filters, apply_sorting, PRIORITY_ORDER


class TestTaskMigration(unittest.TestCase):
    def test_migration_on_read(self):
        """Ensure old task files without priority/due_date are migrated safely."""
        old_task = {"id": 1, "title": "Legacy Task", "completed": False}
        task = Task.from_dict(old_task)
        self.assertEqual(task.priority, "medium")
        self.assertEqual(task.due_date, "")


class TestFiltering(unittest.TestCase):
    def setUp(self):
        self.tasks = [
            Task(1, "A", False, "high", "2026-01-01"),
            Task(2, "B", False, "low", "2026-06-01"),
            Task(3, "C", True, "medium", "2025-12-01"),
            Task(4, "D", False, "high", ""),
        ]

    def test_filter_by_priority(self):
        res = apply_filters(self.tasks, priority="high")
        self.assertEqual(len(res), 2)
        self.assertTrue(all(t.priority == "high" for t in res))

    def test_filter_by_due_before(self):
        res = apply_filters(self.tasks, due_before="2026-03-01")
        self.assertEqual(len(res), 1)
        self.assertEqual(res[0].title, "A")

    def test_filter_completed_excluded_by_default(self):
        res = apply_filters(self.tasks, include_completed=False)
        self.assertEqual(len(res), 3)
        self.assertFalse(any(t.completed for t in res))

    def test_filter_completed_included(self):
        res = apply_filters(self.tasks, include_completed=True)
        self.assertEqual(len(res), 4)

    def test_combined_filters(self):
        res = apply_filters(self.tasks, priority="high", due_before="2026-02-01", include_completed=False)
        self.assertEqual(len(res), 1)
        self.assertEqual(res[0].title, "A")


class TestSorting(unittest.TestCase):
    def setUp(self):
        self.tasks = [
            Task(1, "A", False, "low", "2026-01-01"),
            Task(2, "B", False, "high", "2026-03-01"),
            Task(3, "C", False, "medium", "2025-12-01"),
            Task(4, "D", False, "high", ""),
        ]

    def test_sort_by_due_date(self):
        apply_sorting(self.tasks, "due_date")
        dates = [t.due_date for t in self.tasks]
        # Empty dates should sort last
        self.assertEqual(dates, ["2025-12-01", "2026-01-01", "2026-03-01", ""])

    def test_sort_by_priority(self):
        apply_sorting(self.tasks, "priority")
        priorities = [t.priority for t in self.tasks]
        # Ascending order: low -> medium -> high
        self.assertEqual(priorities, ["low", "medium", "high", "high"])

    def test_sort_stability(self):
        """Verify that original order is preserved for equal keys."""
        tasks = [
            Task(1, "X", False, "medium", ""),
            Task(2, "Y", False, "medium", ""),
        ]
        apply_sorting(tasks, "priority")
        self.assertEqual([t.title for t in tasks], ["X", "Y"])


if __name__ == "__main__":
    unittest.main()
```

### Key Design Decisions & Usage

1. **Backward Compatibility**: `Task.from_dict()` uses `.get()` with safe defaults. When reading an old JSON file missing `priority` or `due_date`, they're automatically populated with `"medium"` and `""` respectively. No manual migration script needed.

2. **Date Filtering**: `YYYY-MM-DD` strings sort lexicographically in the same order as chronological dates, making `<` comparison safe without parsing.

3. **Sorting Behavior**:
   - `--sort due_date`: Ascending chronologically. Tasks without due dates appear last.
   - `--sort priority`: Ascending by priority level (`low` → `medium` → `high`). This highlights lighter work first. If you prefer urgent items first, invert the key: `key=lambda t: -PRIORITY_ORDER.get(...)`.

4. **Help Text**: `argparse` auto-generates clean help. Run `python todo_app.py list --help` to see:
   ```
   usage: todo_app.py list [-h] [--priority {low,medium,high}] [--due-before DUE_BEFORE]
                           [--sort {due_date,priority}] [--include-completed]

   optional arguments:
     -h, --help            show this help message and run tests
     --priority {low,medium,high}
                           Filter by priority level
     --due-before DUE_BEFORE
                           Filter tasks due before YYYY-MM-DD
     --sort {due_date,priority}
                           Sort by field (due_date or priority)
     --include-completed   Include completed tasks in output
   ```

5. **Tests**: Run with `python -m unittest test_todo.py -v`. Covers migration, independent/combined filtering, sorting order, empty-date handling, and sort stability.

Save both files in the same directory and run the CLI as `python todo_app.py <command> [args]`. The implementation is modular, type-hinted, and ready for extension.
