### `todo_app.py`
```python
#!/usr/bin/env python3
"""
To-Do List Application with Priority & Due Date Support.

Usage:
    python todo_app.py add <title>
    python todo_app.py list [--priority low|medium|high] [--due-before YYYY-MM-DD] [--sort due_date|priority]
    python todo_app.py done <id>
    python todo_app.py remove <id>
"""

import argparse
import json
import os
import unittest
from datetime import datetime
from typing import List, Dict, Optional, Tuple

TASKS_FILE = "tasks.json"


# ─── Data Persistence & Migration ─────────────────────────────────────────────

def _normalize_task(task: Dict) -> Dict:
    """Ensure backward compatibility by adding missing fields with safe defaults."""
    task.setdefault("priority", "medium")
    task.setdefault("due_date", None)
    if task["due_date"] == "":
        task["due_date"] = None
    if not isinstance(task.get("completed"), bool):
        task["completed"] = False
    return task


def load_tasks() -> List[Dict]:
    """Load tasks from JSON file, applying migration on read."""
    if not os.path.exists(TASKS_FILE):
        return []
    with open(TASKS_FILE, "r", encoding="utf-8") as f:
        raw_tasks = json.load(f)
    return [_normalize_task(t) for t in raw_tasks]


def save_tasks(tasks: List[Dict]) -> None:
    """Persist tasks to JSON file."""
    with open(TASKS_FILE, "w", encoding="utf-8") as f:
        json.dump(tasks, f, indent=2)


# ─── Core Logic (Filter & Sort) ───────────────────────────────────────────────

def filter_tasks(
    tasks: List[Dict],
    priority: Optional[str] = None,
    due_before: Optional[str] = None
) -> Tuple[List[Dict], Optional[str]]:
    """Filter tasks by priority and/or due date."""
    filtered = tasks.copy()

    if priority:
        priority = priority.lower()
        if priority not in ("low", "medium", "high"):
            return [], f"Error: Priority must be low, medium, or high."
        filtered = [t for t in filtered if t["priority"] == priority]

    if due_before:
        try:
            cutoff = datetime.strptime(due_before, "%Y-%m-%d").date()
        except ValueError:
            return [], "Error: Due date must be in YYYY-MM-DD format."
        # Exclude tasks without a due date
        filtered = [
            t for t in filtered
            if t["due_date"] and datetime.strptime(t["due_date"], "%Y-%m-%d").date() < cutoff
        ]

    return filtered, None


def sort_tasks(tasks: List[Dict], sort_by: Optional[str] = None) -> Tuple[List[Dict], Optional[str]]:
    """Sort tasks by due_date or priority."""
    sorted_list = tasks.copy()

    if sort_by == "due_date":
        # Sort ascending by date; tasks without due dates go last
        sorted_list.sort(key=lambda t: (t["due_date"] is None, t["due_date"] or ""))
    elif sort_by == "priority":
        priority_order = {"high": 0, "medium": 1, "low": 2}
        sorted_list.sort(key=lambda t: priority_order.get(t["priority"], 1))
    elif sort_by:
        return [], f"Error: Unknown sort option '{sort_by}'. Use 'due_date' or 'priority'."

    return sorted_list, None


# ─── CLI Commands ─────────────────────────────────────────────────────────────

def add_task(title: str) -> None:
    tasks = load_tasks()
    new_id = max((t["id"] for t in tasks), default=0) + 1
    tasks.append({
        "id": new_id,
        "title": title,
        "completed": False,
        "priority": "medium",
        "due_date": None
    })
    save_tasks(tasks)
    print(f"Added task #{new_id}: {title}")


def list_tasks(priority: Optional[str] = None, due_before: Optional[str] = None, sort_by: Optional[str] = None) -> None:
    tasks = load_tasks()

    filtered, err = filter_tasks(tasks, priority=priority, due_before=due_before)
    if err:
        print(err)
        return

    sorted_tasks, err = sort_tasks(filtered, sort_by=sort_by)
    if err:
        print(err)
        return

    print(f"{'ID':<5} {'Title':<30} {'Status':<10} {'Priority':<10} {'Due Date':<12}")
    print("-" * 70)
    if not sorted_tasks:
        print("No tasks found.")
        return

    for t in sorted_tasks:
        status = "Done" if t["completed"] else "Pending"
        print(f"{t['id']:<5} {t['title']:<30} {status:<10} {t['priority']:<10} {t['due_date'] or 'None':<12}")


def complete_task(task_id: int) -> None:
    tasks = load_tasks()
    for t in tasks:
        if t["id"] == task_id:
            t["completed"] = True
            save_tasks(tasks)
            print(f"Task #{task_id} marked as completed.")
            return
    print(f"Task #{task_id} not found.")


def remove_task(task_id: int) -> None:
    tasks = load_tasks()
    tasks = [t for t in tasks if t["id"] != task_id]
    save_tasks(tasks)
    print(f"Task #{task_id} removed.")


# ─── Argument Parser ──────────────────────────────────────────────────────────

def parse_args():
    parser = argparse.ArgumentParser(
        description="To-Do List Application with Priority & Due Date Support"
    )
    subparsers = parser.add_subparsers(dest="command", help="Available commands")

    add_parser = subparsers.add_parser("add", help="Add a new task")
    add_parser.add_argument("title", help="Task title")

    list_parser = subparsers.add_parser("list", help="List tasks with optional filters and sorting")
    list_parser.add_argument("--priority", choices=["low", "medium", "high"],
                             help="Filter by priority level")
    list_parser.add_argument("--due-before", help="Filter tasks due before YYYY-MM-DD")
    list_parser.add_argument("--sort", choices=["due_date", "priority"],
                             help="Sort results by field")

    done_parser = subparsers.add_parser("done", help="Mark a task as completed")
    done_parser.add_argument("id", type=int, help="Task ID")

    remove_parser = subparsers.add_parser("remove", help="Remove a task")
    remove_parser.add_argument("id", type=int, help="Task ID")

    return parser.parse_args()


def main():
    args = parse_args()
    if args.command == "add":
        add_task(args.title)
    elif args.command == "list":
        list_tasks(priority=args.priority, due_before=args.due_before, sort_by=args.sort)
    elif args.command == "done":
        complete_task(args.id)
    elif args.command == "remove":
        remove_task(args.id)
    else:
        parse_args().parse_args(["--help"])


# ─── Tests ────────────────────────────────────────────────────────────────────

class TestTodoApp(unittest.TestCase):
    def setUp(self):
        self.sample_tasks = [
            {"id": 1, "title": "Buy groceries", "completed": False, "priority": "medium", "due_date": "2024-12-01"},
            {"id": 2, "title": "Pay bills", "completed": False, "priority": "high", "due_date": "2024-11-15"},
            {"id": 3, "title": "Read book", "completed": False, "priority": "low", "due_date": None},
            {"id": 4, "title": "Fix bug", "completed": True, "priority": "high", "due_date": "2024-10-20"},
        ]

    def test_filter_priority(self):
        filtered, err = filter_tasks(self.sample_tasks, priority="high")
        self.assertIsNone(err)
        self.assertEqual(len(filtered), 2)
        self.assertTrue(all(t["priority"] == "high" for t in filtered))

    def test_filter_due_before(self):
        filtered, err = filter_tasks(self.sample_tasks, due_before="2024-11-20")
        self.assertIsNone(err)
        # Should include Nov 15 and Oct 20, exclude Dec 01 and None
        self.assertEqual(len(filtered), 2)
        self.assertTrue(all(t["due_date"] and datetime.strptime(t["due_date"], "%Y-%m-%d").date() < datetime.strptime("2024-11-20", "%Y-%m-%d").date() for t in filtered))

    def test_filter_no_due_date_excluded(self):
        filtered, _ = filter_tasks(self.sample_tasks, due_before="2025-01-01")
        # Task 3 has no due date, should be excluded
        self.assertFalse(any(t["id"] == 3 for t in filtered))

    def test_sort_due_date(self):
        sorted_tasks, err = sort_tasks(self.sample_tasks, sort_by="due_date")
        self.assertIsNone(err)
        # Expected order: 2024-10-20, 2024-11-15, 2024-12-01, None
        self.assertEqual([t["id"] for t in sorted_tasks], [4, 2, 1, 3])

    def test_sort_priority(self):
        sorted_tasks, err = sort_tasks(self.sample_tasks, sort_by="priority")
        self.assertIsNone(err)
        # Expected order: high (2,4), medium (1), low (3)
        self.assertEqual([t["id"] for t in sorted_tasks], [2, 4, 1, 3])

    def test_migration_missing_fields(self):
        old_format = [{"id": 1, "title": "Legacy task", "completed": 1}]
        migrated = _normalize_task(old_format[0])
        self.assertEqual(migrated["priority"], "medium")
        self.assertIsNone(migrated["due_date"])
        self.assertTrue(isinstance(migrated["completed"], bool))
        self.assertTrue(migrated["completed"])

    def test_invalid_inputs(self):
        _, err = filter_tasks([], priority="urgent")
        self.assertIn("Error", err)
        _, err = sort_tasks([], sort_by="title")
        self.assertIn("Error", err)


if __name__ == "__main__":
    # Run tests if executed directly without arguments
    if len(__import__("sys").argv) == 1:
        unittest.main(argv=[""], exit=False, verbosity=2)
    else:
        main()
```

### Key Features & Design Decisions

1. **Backward Compatibility**: The `_normalize_task()` function runs on every `load_tasks()` call. It safely injects `priority` (defaults to `"medium"`) and `due_date` (defaults to `None`) if they're missing from older JSON files. Empty strings are converted to `None`.
2. **Pure Filter/Sort Functions**: `filter_tasks()` and `sort_tasks()` are decoupled from I/O and UI. They return `(data, error)` tuples, making them easily testable and reusable.
3. **Date Handling**: 
   - `--due-before` strictly compares against `YYYY-MM-DD` and excludes tasks without a due date (standard behavior for date filters).
   - `--sort due_date` sorts ascending chronologically, pushing tasks without due dates to the bottom.
4. **Priority Sorting**: Uses a numeric mapping (`high:0, medium:1, low:2`) for consistent ordering.
5. **CLI Help**: Updated `argparse` help text clearly documents all new flags and their expected formats.
6. **Testing**: Includes a comprehensive `unittest` suite covering:
   - Priority filtering
   - Due date filtering & exclusion of null dates
   - Sorting by date and priority
   - Migration of legacy task formats
   - Invalid input handling
