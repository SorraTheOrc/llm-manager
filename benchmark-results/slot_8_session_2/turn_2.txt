We'll extend a simple Python To‑Do application with support for priority and due dates. The code is structured as a command‑line tool (`todo.py`) with a `TaskManager` class that handles JSON persistence, including migrating old‑format tasks on read. We also add comprehensive tests (`test_todo.py`) for the new filtering and sorting features.

## Key Extension Points

- Each task now has: `title`, `completed` (bool), `priority` (low/medium/high), `due_date` (YYYY-MM-DD or empty string).
- `list` command shows an extra “Priority” and “Due Date” column.
- `list` supports:
  - `--priority high` filter
  - `--due-before YYYY-MM-DD` filter
  - `--sort due_date` or `--sort priority` (empty due dates sort last; priority maps to numeric order low<medium<highest)
- Backward compatibility: when reading an existing file, missing fields are filled with defaults (`completed=False`, `priority='low'`, `due_date=''`).
- `help` text is updated to document the new options.

## File Structure

The implementation consists of two files:

```text
todo.py          # Main CLI application
test_todo.py     # Pytest tests for filtering and sorting
```

#!/usr/bin/env python3
"""To-Do CLI application with priority and due date support."""

import argparse
import json
import os
import sys
from datetime import datetime, date


# Default priority and due_date for migrated / new tasks
DEFAULT_PRIORITY = 'low'
DEFAULT_DUE_DATE = ''

PRIORITY_ORDER = {'low': 0, 'medium': 1, 'high': 2}
DATE_FORMAT = '%Y-%m-%d'

TASKS_FILE = os.path.join(os.path.dirname(__file__) or '.', 'todos.json')


class TaskManager:
    """Handles loading, saving, and querying tasks."""

    def __init__(self, filepath: str = TASKS_FILE):
        self.filepath = filepath
        self.tasks = self._load()

    def _load(self) -> list:
        try:
            with open(self.filepath, 'r') as f:
                data = json.load(f)
        except FileNotFoundError:
            return []
        except json.JSONDecodeError:
            print("Warning: corrupted task file. Starting fresh.")
            return []

        self._migrate(data)
        return data

    @staticmethod
    def _migrate(tasks: list):
        """Ensure all tasks have the required fields (backward compatibility)."""
        for task in tasks:
            if 'completed' not in task:
                task['completed'] = False
            if 'priority' not in task:
                task['priority'] = DEFAULT_PRIORITY
            if 'due_date' not in task:
                task['due_date'] = DEFAULT_DUE_DATE

    def _save(self):
        with open(self.filepath, 'w') as f:
            json.dump(self.tasks, f, indent=2, ensure_ascii=False)

    def add(self, title: str, priority: str = DEFAULT_PRIORITY,
            due_date: str = DEFAULT_DUE_DATE):
        task = {
            'title': title,
            'completed': False,
            'priority': priority,
            'due_date': due_date
        }
        self.tasks.append(task)
        self._save()
        return len(self.tasks)  # return task ID (1‑based)

    def list_tasks(self, priority: str = None, due_before: str = None,
                   sort_by: str = None) -> list:
        tasks = self.tasks[:]

        # Filter by priority
        if priority:
            tasks = [t for t in tasks if t['priority'] == priority]

        # Filter by due_before
        if due_before:
            try:
                due_before_dt = datetime.strptime(due_before, DATE_FORMAT).date()
            except ValueError:
                print(f"Invalid due-before date: {due_before}. Use YYYY-MM-DD.")
                sys.exit(1)
            tasks = [t for t in tasks if t['due_date'] and
                     datetime.strptime(t['due_date'], DATE_FORMAT).date() <= due_before_dt]

        # Sort
        if sort_by == 'due_date':
            tasks.sort(key=lambda t: t['due_date'] if t['due_date'] else '9999-12-31')
        elif sort_by == 'priority':
            tasks.sort(key=lambda t: PRIORITY_ORDER.get(t['priority'], 0))

        return tasks

    def complete(self, task_id: int) -> bool:
        if 1 <= task_id <= len(self.tasks):
            self.tasks[task_id - 1]['completed'] = True
            self._save()
            return True
        return False

    def delete(self, task_id: int) -> bool:
        if 1 <= task_id <= len(self.tasks):
            self.tasks.pop(task_id - 1)
            self._save()
            return True
        return False


def format_task_list(tasks: list) -> str:
    """Return a string containing a formatted table of tasks."""
    if not tasks:
        return "No tasks found."

    lines = []
    # Header
    lines.append(
        f"{'ID':<4} {'Completed':<12} {'Priority':<10} {'Due Date':<12} {'Title'}"
    )
    lines.append('-' * 80)

    for i, task in enumerate(tasks, start=1):
        completed = '✓' if task['completed'] else '✗'
        priority = task['priority'].capitalize()
        due_date = task['due_date'] if task['due_date'] else '-'
        title = task['title']
        lines.append(f"{i:<4} {completed:<12} {priority:<10} {due_date:<12} {title}")

    return '\n'.join(lines)


def main():
    parser = argparse.ArgumentParser(
        description="Simple To-Do application with priority and due dates.",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
Examples:
  todo.py add "Buy milk"
  todo.py add "Write report" --priority high --due-date 2026-08-15
  todo.py list
  todo.py list --priority high
  todo.py list --due-before 2026-08-01
  todo.py list --sort due_date
  todo.py complete 1
  todo.py delete 2
        """
    )
    subparsers = parser.add_subparsers(dest='command', help='Available commands')

    # Add
    add_parser = subparsers.add_parser('add', help='Add a new task')
    add_parser.add_argument('title', help='Task title')
    add_parser.add_argument('--priority', choices=['low', 'medium', 'high'],
                            default=DEFAULT_PRIORITY, help='Task priority (default: low)')
    add_parser.add_argument('--due-date', metavar='YYYY-MM-DD',
                            default=DEFAULT_DUE_DATE, help='Due date (optional)')

    # List
    list_parser = subparsers.add_parser('list', help='List tasks (with filtering and sorting)')
    list_parser.add_argument('--priority', choices=['low', 'medium', 'high'],
                             help='Filter by priority')
    list_parser.add_argument('--due-before', metavar='YYYY-MM-DD',
                             help='Show tasks due on or before a date')
    list_parser.add_argument('--sort', choices=['due_date', 'priority'],
                             help='Sort tasks (unsorted by default)')

    # Complete
    complete_parser = subparsers.add_parser('complete', help='Mark a task as completed')
    complete_parser.add_argument('task_id', type=int, help='Task ID (from list)')

    # Delete
    delete_parser = subparsers.add_parser('delete', help='Delete a task')
    delete_parser.add_argument('task_id', type=int, help='Task ID (from list)')

    args = parser.parse_args()

    manager = TaskManager()

    if args.command == 'add':
        manager.add(args.title, args.priority, args.due_date)
        print(f"Task added (ID {len(manager.tasks)}).")

    elif args.command == 'list':
        tasks = manager.list_tasks(args.priority, args.due_before, args.sort)
        print(format_task_list(tasks))

    elif args.command == 'complete':
        if manager.complete(args.task_id):
            print(f"Task {args.task_id} marked completed.")
        else:
            print(f"Error: no task with ID {args.task_id}.")

    elif args.command == 'delete':
        if manager.delete(args.task_id):
            print(f"Task {args.task_id} deleted.")
        else:
            print(f"Error: no task with ID {args.task_id}.")

    else:
        parser.print_help()


if __name__ == '__main__':
    main()

## 1. Implementation: `todo.py`

```python
import json
import os
import argparse
from datetime import datetime

DATA_FILE = "tasks.json"
PRIORITIES = ["low", "medium", "high"]
PRIORITY_RANK = {"low": 0, "medium": 1, "high": 2, "": -1}
EMPTY_DATE_SORT = "9999-12-31"


def validate_date(date_text):
    """Return True if date_text is empty or a valid YYYY-MM-DD string."""
    if not date_text:
        return True
    try:
        datetime.strptime(date_text, "%Y-%m-%d")
        return True
    except ValueError:
        return False


# ------------------------------------------------------------------
# Data persistence
# ------------------------------------------------------------------
def load_tasks():
    """Load tasks from the JSON file. Migrate legacy tasks automatically."""
    if not os.path.exists(DATA_FILE):
        return []
    with open(DATA_FILE, "r") as f:
        tasks = json.load(f)
    for task in tasks:
        # Backward compatibility: ensure new fields exist
        task.setdefault("priority", "")
        task.setdefault("due_date", "")
    return tasks


def save_tasks(tasks):
    """Write tasks to the JSON file."""
    with open(DATA_FILE, "w") as f:
        json.dump(tasks, f, indent=2)


# ------------------------------------------------------------------
# Filtering & sorting logic  (pure functions – easy to test)
# ------------------------------------------------------------------
def filter_tasks(tasks, priority=None, due_before=None):
    """
    Return a filtered copy of the tasks list.
      - priority : only tasks with this priority (string)
      - due_before : only tasks whose due_date is < given date
                     (tasks without a due_date are excluded)
    """
    filtered = list(tasks)
    if priority:
        filtered = [t for t in filtered if t.get("priority") == priority]
    if due_before:
        filtered = [
            t for t in filtered
            if t.get("due_date") and t["due_date"] < due_before
        ]
    return filtered


def sort_tasks(tasks, sort_key=None):
    """
    Return a sorted copy of the tasks list.
    Supported keys: 'priority', 'due_date', 'title', 'completed'.
    """
    if sort_key is None:
        return list(tasks)

    sorted_list = list(tasks)
    if sort_key == "priority":
        sorted_list.sort(key=lambda t: PRIORITY_RANK.get(t.get("priority", ""), -1))
    elif sort_key == "due_date":
        sorted_list.sort(key=lambda t: t.get("due_date") if t.get("due_date") else EMPTY_DATE_SORT)
    elif sort_key == "title":
        sorted_list.sort(key=lambda t: t.get("title", ""))
    elif sort_key == "completed":
        sorted_list.sort(key=lambda t: t.get("completed", False))
    return sorted_list


# ------------------------------------------------------------------
# Display
# ------------------------------------------------------------------
def display_tasks(tasks):
    """Print a formatted table of tasks."""
    if not tasks:
        print("No tasks to show.")
        return

    print(f"{'ID':<4} {'Title':<30} {'Status':<12} {'Priority':<10} {'Due Date':<12}")
    print("-" * 68)
    for i, task in enumerate(tasks, start=1):
        status = "Done" if task["completed"] else "Pending"
        priority = task["priority"] if task["priority"] else "-"
        due_date = task["due_date"] if task["due_date"] else "-"
        # Truncate long titles for clean formatting
        title = task["title"][:30]
        print(f"{i:<4} {title:<30} {status:<12} {priority:<10} {due_date:<12}")


# ------------------------------------------------------------------
# Command handlers
# ------------------------------------------------------------------
def cmd_add(args):
    """Handle `add` command."""
    if args.due_date and not validate_date(args.due_date):
        print("Error: Invalid due date format. Use YYYY-MM-DD.")
        return
    tasks = load_tasks()
    task = {
        "title": args.title,
        "completed": False,
        "priority": args.priority if args.priority else "",
        "due_date": args.due_date if args.due_date else "",
    }
    tasks.append(task)
    save_tasks(tasks)
    print(f"Added task: '{task['title']}'")


def cmd_list(args):
    """Handle `list` command (with filtering/sorting)."""
    if args.due_before and not validate_date(args.due_before):
        print("Error: Invalid --due-before date format. Use YYYY-MM-DD.")
        return
    tasks = load_tasks()
    tasks = filter_tasks(tasks, priority=args.priority, due_before=args.due_before)
    tasks = sort_tasks(tasks, sort_key=args.sort)
    display_tasks(tasks)


def cmd_done(args):
    """Handle `done` / `complete` command."""
    tasks = load_tasks()
    try:
        idx = int(args.id) - 1
        task = tasks[idx]
        task["completed"] = not task["completed"]
        save_tasks(tasks)
        status = "done" if task["completed"] else "pending"
        print(f"Marked task '{task['title']}' as {status}")
    except (IndexError, ValueError):
        print("Error: Invalid task ID.")


def cmd_remove(args):
    """Handle `remove` / `rm` command."""
    tasks = load_tasks()
    try:
        idx = int(args.id) - 1
        task = tasks.pop(idx)
        save_tasks(tasks)
        print(f"Removed task '{task['title']}'")
    except (IndexError, ValueError):
        print("Error: Invalid task ID.")


# ------------------------------------------------------------------
# Argument parsing
# ------------------------------------------------------------------
def create_parser():
    parser = argparse.ArgumentParser(
        description="Simple To-Do application with priority and due dates."
    )
    subparsers = parser.add_subparsers(dest="command", title="Commands")

    # add
    add_parser = subparsers.add_parser("add", help="Add a new task")
    add_parser.add_argument("title", help="Title of the task")
    add_parser.add_argument(
        "--priority", choices=PRIORITIES,
        help="Priority of the task (low / medium / high)"
    )
    add_parser.add_argument(
        "--due-date", metavar="YYYY-MM-DD",
        help="Due date of the task (format: YYYY-MM-DD)"
    )

    # list
    list_parser = subparsers.add_parser(
        "list", aliases=["ls"],
        help="List tasks with optional filtering and sorting"
    )
    list_parser.add_argument(
        "--priority", choices=PRIORITIES,
        help="Filter by priority level"
    )
    list_parser.add_argument(
        "--due-before", metavar="YYYY-MM-DD",
        help="Filter tasks with a due date before this date"
    )
    list_parser.add_argument(
        "--sort", choices=["priority", "due_date", "title", "completed"],
        help="Sort tasks by the given field"
    )

    # done / complete
    done_parser = subparsers.add_parser(
        "done", aliases=["complete"],
        help="Toggle task completion"
    )
    done_parser.add_argument("id", help="ID of the task (from 'list' output)")

    # remove
    remove_parser = subparsers.add_parser(
        "remove", aliases=["rm"],
        help="Remove a task"
    )
    remove_parser.add_argument("id", help="ID of the task (from 'list' output)")

    return parser


def main():
    parser = create_parser()
    args = parser.parse_args()

    if args.command == "add":
        cmd_add(args)
    elif args.command in ("list", "ls"):
        cmd_list(args)
    elif args.command in ("done", "complete"):
        cmd_done(args)
    elif args.command in ("remove", "rm"):
        cmd_remove(args)
    else:
        parser.print_help()


if __name__ == "__main__":
    main()

        result = filter_tasks(tasks, priority=None, due_before="2026-08-01")
        result = sort_tasks(result, sort_key="priority")
        self.assertEqual(len(result), 1)
        self.assertEqual(result[0]["title"], "Call plumber")


if __name__ == "__main__":
    unittest.main()
```

## 4. Key Design Decisions

| Requirement                 | Implementation                                                                 |
|-----------------------------|---------------------------------------------------------------------------------|
| **Backward compatibility**  | `load_tasks()` calls `setdefault("priority", "")` and `setdefault("due_date", "")` for every loaded task. |
| **Filtering**               | `filter_tasks()` returns a new list; empty due dates are excluded from `--due-before`. |
| **Sorting**                 | `sort_tasks()` uses a default value for empty dates (`9999-12-31`) and a rank dict for priority. |
| **Testability**             | Filtering and sorting are pure functions with no file I/O.                      |
| **Input validation**        | `validate_date()` checks the YYYY-MM-DD format for new CLI arguments.           |

Everything is self-contained in standard Python 3, requires no external dependencies, and can be run with `python todo.py` or tested with `python -m unittest test_todo.py -v`. The migration on read is completely transparent to the user—legacy files are upgraded seamlessly the first time they are read.
