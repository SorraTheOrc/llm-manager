### `todo_app.py`

```python
"""A simple CLI To-Do application with JSON storage and full CRUD operations."""

from __future__ import annotations

import argparse
import json
import sys
from dataclasses import dataclass, asdict
from pathlib import Path
from typing import Optional

DEFAULT_TODO_FILE = Path("todo.json")


@dataclass
class Task:
    """Represents a single to-do task."""
    id: int
    title: str
    completed: bool = False

    def to_dict(self) -> dict:
        """Convert task to a dictionary for JSON serialization."""
        return asdict(self)

    @classmethod
    def from_dict(cls, data: dict) -> Task:
        """Create a Task instance from a dictionary."""
        return cls(**data)


class TodoStorage:
    """Handles persistent storage and CRUD operations for tasks."""

    def __init__(self, file_path: Path) -> None:
        self.file_path = file_path
        self.tasks: list[Task] = []
        self._load()

    def _load(self) -> None:
        """Load tasks from the JSON file."""
        if self.file_path.exists():
            try:
                data = json.loads(self.file_path.read_text())
                self.tasks = [Task.from_dict(t) for t in data]
            except (json.JSONDecodeError, KeyError) as e:
                print(f"Warning: Corrupted storage file. Starting fresh. ({e})")
                self.tasks = []
        else:
            self.tasks = []

    def save(self) -> None:
        """Persist current tasks to the JSON file."""
        self.file_path.parent.mkdir(parents=True, exist_ok=True)
        self.file_path.write_text(json.dumps([t.to_dict() for t in self.tasks], indent=2))

    def add(self, title: str) -> Task:
        """Add a new task. Raises ValueError if title is empty."""
        if not (title := title.strip()):
            raise ValueError("Task title cannot be empty")
        
        new_id = max((t.id for t in self.tasks), default=0) + 1
        task = Task(id=new_id, title=title)
        self.tasks.append(task)
        self.save()
        return task

    def list_tasks(self, completed: Optional[bool] = None) -> list[Task]:
        """List tasks, optionally filtered by completion status."""
        if completed is not None:
            return [t for t in self.tasks if t.completed == completed]
        return list(self.tasks)

    def complete(self, task_id: int) -> Optional[Task]:
        """Mark a task as completed. Returns the task or None if not found."""
        for task in self.tasks:
            if task.id == task_id:
                task.completed = True
                self.save()
                return task
        return None

    def delete(self, task_id: int) -> Optional[Task]:
        """Delete a task by ID. Returns the task or None if not found."""
        for i, task in enumerate(self.tasks):
            if task.id == task_id:
                removed = self.tasks.pop(i)
                self.save()
                return removed
        return None


def create_parser() -> argparse.ArgumentParser:
    """Create and configure the argument parser."""
    parser = argparse.ArgumentParser(
        prog="todo",
        description="A simple CLI To-Do application using JSON storage."
    )
    parser.add_argument(
        "--file", type=Path, default=DEFAULT_TODO_FILE,
        help="Path to the JSON storage file (default: todo.json)"
    )
    subparsers = parser.add_subparsers(dest="command", required=True, help="Available commands")

    subparsers.add_parser("add", help="Add a new task")
    subparsers.add_parser("list", help="List tasks")
    subparsers.add_parser("complete", help="Mark a task as completed")
    subparsers.add_parser("delete", help="Delete a task")
    subparsers.add_parser("help", help="Show usage information")

    # Add specific arguments to subparsers
    p_add = subparsers.add_parser("add", help="Add a new task")
    p_add.add_argument("title", type=str, help="Task title")

    p_list = subparsers.add_parser("list", help="List tasks")
    p_list.add_argument("--completed", action="store_true", help="Show only completed tasks")
    p_list.add_argument("--pending", action="store_true", help="Show only pending tasks")

    p_complete = subparsers.add_parser("complete", help="Mark a task as completed")
    p_complete.add_argument("id", type=int, help="Task ID")

    p_delete = subparsers.add_parser("delete", help="Delete a task")
    p_delete.add_argument("id", type=int, help="Task ID")

    return parser


def run(args: Optional[list[str]] = None, storage_file: Optional[Path] = None) -> int:
    """Execute CLI commands. Returns exit code (0 for success, 1 for error)."""
    if storage_file is None:
        storage_file = DEFAULT_TODO_FILE

    parser = create_parser()
    parsed = parser.parse_args(args)
    
    # Allow CLI --file argument to override default or test-provided path
    if parsed.file != DEFAULT_TODO_FILE:
        storage_file = parsed.file

    storage = TodoStorage(storage_file)

    try:
        match parsed.command:
            case "add":
                task = storage.add(parsed.title)
                print(f"Added task #{task.id}: {task.title}")
                
            case "list":
                if parsed.completed and parsed.pending:
                    print("Error: Cannot use --completed and --pending together.")
                    return 1
                    
                if parsed.completed:
                    tasks = storage.list_tasks(completed=True)
                elif parsed.pending:
                    tasks = storage.list_tasks(completed=False)
                else:
                    tasks = storage.list_tasks()

                if not tasks:
                    print("No tasks found.")
                else:
                    for t in tasks:
                        status = "[x]" if t.completed else "[ ]"
                        print(f"{status} {t.id}: {t.title}")
                        
            case "complete":
                task = storage.complete(parsed.id)
                if task:
                    print(f"Task #{task.id} marked as completed.")
                else:
                    print(f"Error: Task #{parsed.id} not found.")
                    return 1
                    
            case "delete":
                task = storage.delete(parsed.id)
                if task:
                    print(f"Deleted task #{task.id}: {task.title}")
                else:
                    print(f"Error: Task #{parsed.id} not found.")
                    return 1
                    
            case "help":
                parser.print_help()
                
    except ValueError as e:
        print(f"Error: {e}")
        return 1
    except Exception as e:
        print(f"Unexpected error: {e}")
        return 1
        
    return 0


if __name__ == "__main__":
    sys.exit(run())
```

### How to Use & Test

1. **Save the files** in a project directory:
   ```
   your_project/
   ├── todo_app.py
   └── tests/
       └── test_todo_app.py
   ```

2. **Run the tests**:
   ```bash
   pip install pytest
   pytest tests/ -v
   ```

3. **Run the CLI**:
   ```bash
   python -m todo_app help
   python -m todo_app add "Buy milk"
   python -m todo_app list
   python -m todo_app complete 1
   python -m todo_app list --completed
   python -m todo_app delete 1
   ```

4. **Import as a module**:
   ```python
   from todo_app import TodoStorage, Task, run
   from pathlib import Path

   storage = TodoStorage(Path("my_tasks.json"))
   task = storage.add("Learn Python")
   print(task.id)  # 1
   ```

### Key Design Decisions
- **Dataclass for Tasks**: Provides immutable-like structure, automatic `__init__`, and clean serialization via `asdict()`.
- **Auto-increment ID**: Computed from existing tasks (`max(...) + 1`) to avoid gaps after deletions, keeping IDs stable.
- **Atomic-ish Storage**: Writes complete JSON on every mutation. For a CLI app, this is fast and prevents corruption better than partial writes.
- **Strict Input Validation**: Empty titles raise `ValueError`. Non-existent IDs return `None` and trigger user-friendly error messages.
- **Testability**: `run()` accepts `storage_file` to bypass file I/O during tests. `TodoStorage` is fully isolated and mockable.
- **Python 3.12+ Features**: Uses `match/case`, modern type hints, and `pathlib` for robust file handling.
