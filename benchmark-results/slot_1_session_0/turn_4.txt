## 2. Data Layer (`todo_app/db.py`)

```python
import json
from datetime import datetime
from pathlib import Path

DATA_DIR = Path.home() / ".todo"
DATA_FILE = DATA_DIR / "tasks.json"


def _load_tasks():
    if not DATA_FILE.exists():
        return []
    try:
        with open(DATA_FILE, "r") as f:
            return json.load(f)
    except (json.JSONDecodeError, OSError):
        return []


def _save_tasks(tasks):
    DATA_DIR.mkdir(parents=True, exist_ok=True)
    with open(DATA_FILE, "w") as f:
        json.dump(tasks, f, indent=2, ensure_ascii=False)


def _next_id(tasks):
    if not tasks:
        return 1
    return max(t["id"] for t in tasks) + 1


def add_task(description):
    tasks = _load_tasks()
    task = {
        "id": _next_id(tasks),
        "description": description,
        "done": False,
        "created_at": datetime.now().isoformat(),
        "completed_at": None,
    }
    tasks.append(task)
    _save_tasks(tasks)
    return task


def list_tasks():
    return _load_tasks()


def toggle_task(task_id):
    tasks = _load_tasks()
    for task in tasks:
        if task["id"] == task_id:
            task["done"] = not task["done"]
            task["completed_at"] = datetime.now().isoformat() if task["done"] else None
            _save_tasks(tasks)
            return task
    return None


def delete_task(task_id):
    tasks = _load_tasks()
    new_tasks = [t for t in tasks if t["id"] != task_id]
    if len(tasks) == len(new_tasks):
        return False
    _save_tasks(new_tasks)
    return True


def get_daily_summary():
    tasks = _load_tasks()
    today = datetime.now().date().isoformat()
    pending = [t for t in tasks if not t["done"]]
    completed_today = [
        t for t in tasks
        if t["done"] and t.get("completed_at", "").startswith(today)
    ]
    return pending + completed_today
```

## 3. CLI Interface (`todo_app/cli.py`)

```python
import sys
import argparse


def main():
    parser = argparse.ArgumentParser(prog="todo")
    sub = parser.add_subparsers(dest="command")

    # add
    p_add = sub.add_parser("add", help="Add a new task")
    p_add.add_argument("description", type=str)

    # list
    sub.add_parser("list", help="List all tasks")

    # toggle
    p_tog = sub.add_parser("toggle", help="Toggle a task")
    p_tog.add_argument("task_id", type=int)

    # delete
    p_del = sub.add_parser("delete", help="Delete a task")
    p_del.add_argument("task_id", type=int)

    # daily
    sub.add_parser("daily", help="Daily summary")

    # serve
    p_serve = sub.add_parser("serve", help="Start web server")
    p_serve.add_argument("--port", type=int, default=8080)

    args = parser.parse_args()

    from . import db

    if args.command == "add":
        task = db.add_task(args.description)
        print(f"Added task: {task['description']} (ID: {task['id']})")

    elif args.command == "list":
        tasks = db.list_tasks()
        if not tasks:
            print("No tasks.")
            return
        for t in tasks:
            status = "[x]" if t["done"] else "[ ]"
            print(f"{status} {t['id']}: {t['description']}")

    elif args.command == "toggle":
        task = db.toggle_task(args.task_id)
        if task:
            state = "completed" if task["done"] else "pending"
            print(f"Task {task['id']} is now {state}.")
        else:
            print(f"Task {args.task_id} not found.")

    elif args.command == "delete":
        if db.delete_task(args.task_id):
            print(f"Deleted task {args.task_id}.")
        else:
            print(f"Task {args.task_id} not found.")

    elif args.command == "daily":
        tasks = db.get_daily_summary()
        pending = [t for t in tasks if not t["done"]]
        completed = [t for t in tasks if t["done"]]
        print("Pending tasks:")
        for t in pending:
            print(f"  {t['id']}: {t['description']}")
        print("Completed today:")
        for t in completed:
            print(f"  {t['id']}: {t['description']}")

    elif args.command == "serve":
        try:
            from . import web
            web.run(port=args.port)
        except KeyboardInterrupt:
            print("\nServer stopped.")
        except OSError as e:
            print(f"Error: {e}", file=sys.stderr)
            sys.exit(1)

    else:
        parser.print_help()


if __name__ == "__main__":
    main()
```
