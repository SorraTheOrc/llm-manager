import argparse
import json
import os
import sys
from datetime import datetime
from typing import Any, Dict, List, Optional

# ---------- Constants ----------
TASK_FILE = "tasks.json"
PRIORITY_ORDER = {"low": 0, "medium": 1, "high": 2}
DEFAULT_PRIORITY = "medium"
DATE_FORMAT = "%Y-%m-%d"


# ---------- Task helpers ----------
def migrate_task(task: Dict[str, Any]) -> Dict[str, Any]:
    """Ensure all required keys exist (backward compatibility)."""
    if "priority" not in task:
        task["priority"] = DEFAULT_PRIORITY
    if "due_date" not in task:
        task["due_date"] = ""
    if "completed" not in task:
        task["completed"] = False
    if "id" not in task:
        # Assign a new id if missing (should not happen in normal use)
        # For safety, we give a temporary placeholder; the caller should fix it.
        task["id"] = -1
    return task


def load_tasks() -> List[Dict[str, Any]]:
    """Load tasks from file, migrating old entries."""
    if not os.path.exists(TASK_FILE):
        return []
    try:
        with open(TASK_FILE, "r") as f:
            tasks: List[Dict[str, Any]] = json.load(f)
    except (json.JSONDecodeError, IOError):
        print(f"Error reading {TASK_FILE}. Starting with empty list.", file=sys.stderr)
        return []
    # Migrate each task
    for task in tasks:
        migrate_task(task)
    return tasks


def save_tasks(tasks: List[Dict[str, Any]]) -> None:
    """Save tasks to file."""
    # Ensure all tasks are migrated before saving
    for task in tasks:
        migrate_task(task)
    with open(TASK_FILE, "w") as f:
        json.dump(tasks, f, indent=2)


def generate_id(tasks: List[Dict[str, Any]]) -> int:
    """Return a new unique id."""
    if not tasks:
        return 1
    return max(task["id"] for task in tasks) + 1


# ---------- CLI commands ----------
def cmd_add(args: argparse.Namespace) -> None:
    """Add a new task."""
    tasks = load_tasks()
    new_id = generate_id(tasks)
    task = {
        "id": new_id,
        "title": args.title,
        "completed": False,
        "priority": args.priority if args.priority else DEFAULT_PRIORITY,
        "due_date": args.due if args.due else "",
    }
    tasks.append(task)
    save_tasks(tasks)
    print(f"Added task {new_id}: {args.title}")


def cmd_list(args: argparse.Namespace) -> None:

    tasks = load_tasks()

    # Filtering
    if args.priority:
        tasks = [t for t in tasks if t["priority"] == args.priority]
    if args.due_before:
        try:
            cutoff = datetime.strptime(args.due_before, DATE_FORMAT).date()
        except ValueError:
            print(f"Invalid date format. Use YYYY-MM-DD.", file=sys.stderr)
            sys.exit(1)
        tasks = [t for t in tasks if t["due_date"] and
                 datetime.strptime(t["due_date"], DATE_FORMAT).date() < cutoff]

    # Sorting
    if args.sort:
        reverse = False
        field = args.sort
        # due_date sorting: empty dates go to end
        if field == "due_date":
            def sort_key(t):
                if not t["due_date"]:
                    return "9999-99-99"  # far future
                return t["due_date"]
            tasks.sort(key=sort_key)
        elif field == "priority":
            tasks.sort(key=lambda t: PRIORITY_ORDER.get(t["priority"], 1))
        elif field == "id":
            tasks.sort(key=lambda t: t["id"])
        elif field == "title":
            tasks.sort(key=lambda t: t["title"])
        else:
            print(f"Unknown sort field: {field}", file=sys.stderr)
            sys.exit(1)
    else:
        # Default sorting by id
        tasks.sort(key=lambda t: t["id"])

    # Display
    if not tasks:
        print("No tasks found.")
        return

    # Format: ID, title, completed, priority, due_date
    header = f"{'ID':<4} {'Title':<30} {'Done':<6} {'Priority':<8} {'Due Date':<12}"
    print(header)
    print("-" * 60)

        print(f"{t['id']:<4} {t['title'][:30]:<30} {done:<6} {t['priority']:<8} {due:<12}")

class Task:
    """Represents a single to-do task."""

    def __init__(self, title: str, completed: bool = False,
                 priority: str = "low", due_date: str = "", task_id: int = None):
        self.id = task_id
        self.title = title
        self.completed = completed
        self.priority = priority if priority in VALID_PRIORITIES else "low"
        self.due_date = due_date  # empty string if not set

    def to_dict(self) -> Dict[str, Any]:
        return {
            "id": self.id,
            "title": self.title,
            "completed": self.completed,
            "priority": self.priority,
            "due_date": self.due_date
        }

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> 'Task':
        # Migration: add defaults for missing fields
        if "priority" not in data:
            data["priority"] = "low"
        if "due_date" not in data:
            data["due_date"] = ""
        return cls(
            title=data["title"],
            completed=data.get("completed", False),
            priority=data.get("priority", "low"),
            due_date=data.get("due_date", ""),
            task_id=data.get("id")
        )

    def __repr__(self):
        return f"Task({self.id}, {self.title!r}, done={self.completed}, prio={self.priority}, due={self.due_date})"


class TaskManager:
    """Handles loading, saving, and querying tasks."""

    def __init__(self, filepath: str = TASK_FILE):
        self.filepath = filepath
        self._tasks: List[Task] = []
        self._next_id = 1
        self.load()

    def load(self):
        """Load tasks from file, with migration for old format."""
        if not os.path.exists(self.filepath):
            self._tasks = []
            self._next_id = 1
            return

        try:
            with open(self.filepath, "r") as f:
                data = json.load(f)
        except (json.JSONDecodeError, IOError):
            print("Warning: corrupted task file, starting fresh.", file=sys.stderr)
            self._tasks = []
            self._next_id = 1
            return

        # Support both list of dicts and dict with 'tasks' key (flexible)
        if isinstance(data, list):
            task_list = data
        elif isinstance(data, dict) and "tasks" in data:
            task_list = data["tasks"]
        else:
            task_list = []

        self._tasks = [Task.from_dict(item) for item in task_list]
        if self._tasks:
            self._next_id = max(t.id for t in self._tasks) + 1
        else:
            self._next_id = 1

    def save(self):
        """Persist tasks to file."""
        data = {
            "tasks": [t.to_dict() for t in self._tasks]
        }
        with open(self.filepath, "w") as f:
            json.dump(data, f, indent=2)

    def add(self, title: str, priority: str = "low", due_date: str = "") -> Task:
        """Add a new task and save."""
        task = Task(
            title=title,
            completed=False,
            priority=priority,
            due_date=due_date,
            task_id=self._next_id
        )
        self._next_id += 1
        self._tasks.append(task)
        self.save()
        return task

    def get_all(self) -> List[Task]:
        return self._tasks

    def find_by_id(self, task_id: int) -> Optional[Task]:
        for t in self._tasks:
            if t.id == task_id:
                return t
        return None

    def complete(self, task_id: int) -> bool:
        task = self.find_by_id(task_id)
        if not task:
            return False
        task.completed = True
        self.save()
        return True

    def delete(self, task_id: int) -> bool:
        for i, t in enumerate(self._tasks):
            if t.id == task_id:
                del self._tasks[i]
                self.save()
                return True
        return False

    def filter(self, priority: Optional[str] = None,
               due_before: Optional[date] = None) -> List[Task]:
        """Return filtered tasks."""
        result = self._tasks[:]
        if priority:
            result = [t for t in result if t.priority == priority]
        if due_before:
            result = [
                t for t in result
                if t.due_date and self._parse_date(t.due_date) and
                   self._parse_date(t.due_date) < due_before
            ]
        return result

    def sort_tasks(self, tasks: List[Task],
                   sort_by: Optional[str] = None,
                   reverse: bool = False) -> List[Task]:
        """Return a new sorted list of tasks."""
        if sort_by == "priority":
            def key_fn(t):
                return PRIORITY_ORDER.get(t.priority, -1)
        elif sort_by == "due_date":
            def key_fn(t):
                # Put empty due dates at the end when ascending
                if not t.due_date:
                    return date.max if not reverse else date.min
                d = self._parse_date(t.due_date)
                return d if d else (date.min if reverse else date.max)
        elif sort_by == "id":
            key_fn = lambda t: t.id
        elif sort_by == "title":
            key_fn = lambda t: t.title.lower()
        else:
            # default sort by id
            key_fn = lambda t: t.id

        return sorted(tasks, key=key_fn, reverse=reverse)

    @staticmethod
    def _parse_date(d_str: str) -> Optional[date]:
        if not d_str:
            return None
        try:
            return datetime.strptime(d_str, DATE_FORMAT).date()
        except ValueError:

    # Sorting
    if args.sort:
        if args.sort == "priority":
            tasks.sort(key=lambda t: PRIORITY_ORDER.get(t.priority, -1))
        elif args.sort == "due_date":
            # Empty due dates go to the end, then sort by date
            def due_sort_key(t):
                if not t.due_date:
                    return (1, "")  # (is_empty, date) – empty after non‑empty
                return (0, t.due_date)
            tasks.sort(key=due_sort_key)
        else:
            print(f"Warning: Unknown sort key '{args.sort}' – ignored.")

    # Print table
    if not tasks:
        print("No tasks.")
        return
    print_header()
    for task in tasks:
        print(format_task(task))

def cmd_complete(args):
    tasks = load_tasks()
    for task in tasks:
        if task.id == args.id:
            task.completed = True
            save_tasks(tasks)
            print(f"Task #{task.id} marked as completed.")
            return
    print(f"Task #{args.id} not found.")

def cmd_delete(args):
    tasks = load_tasks()
    new_tasks = [t for t in tasks if t.id != args.id]
    if len(new_tasks) == len(tasks):
        print(f"Task #{args.id} not found.")
    else:
        save_tasks(new_tasks)
        print(f"Task #{args.id} deleted.")
