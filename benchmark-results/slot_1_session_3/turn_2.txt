### `todo.py`

```python
#!/usr/bin/env python3
"""
To-Do List Application with Priority and Due Date Support.

Features:
- Tasks have title, completed status, priority (low/medium/high), and due_date.
- List command supports filtering by priority and due date.
- List command supports sorting by priority and due date.
- Automatic migration of legacy tasks on read.
"""

import argparse
import json
import os
import sys
from datetime import datetime
from typing import List, Optional

DATA_FILE = "tasks.json"

PRIORITY_ORDER = {"high": 0, "medium": 1, "low": 2}
VALID_PRIORITIES = {"low", "medium", "high"}


class TodoManager:
    def __init__(self, data_file: str = DATA_FILE):
        self.data_file = data_file
        self.tasks: List[dict] = []
        self._load_tasks()

    def _load_tasks(self):
        """Load tasks from JSON file and migrate if necessary."""
        if os.path.exists(self.data_file):
            with open(self.data_file, "r") as f:
                raw_tasks = json.load(f)
                self.tasks = [self._migrate_task(t) for t in raw_tasks]
        else:
            self.tasks = []

    def _migrate_task(self, task: dict) -> dict:
        """Add missing fields to legacy tasks."""
        if "priority" not in task:
            task["priority"] = "low"
        if "due_date" not in task:
            task["due_date"] = ""
        return task

    def _save_tasks(self):
        """Save tasks to JSON file."""
        with open(self.data_file, "w") as f:
            json.dump(self.tasks, f, indent=2)

    def add_task(self, title: str, priority: str = "low", due_date: str = ""):
        """Add a new task."""
        if priority not in VALID_PRIORITIES:
            print(f"Error: Invalid priority '{priority}'. Use low, medium, or high.")
            sys.exit(1)
        
        if due_date and not self._is_valid_date(due_date):
            print(f"Error: Invalid date format '{due_date}'. Use YYYY-MM-DD.")
            sys.exit(1)

        task = {
            "title": title,
            "completed": False,
            "priority": priority,
            "due_date": due_date,
        }
        self.tasks.append(task)
        self._save_tasks()
        print(f"Added task: {title}")

    def complete_task(self, index: int):
        """Mark a task as completed."""
        if 0 <= index < len(self.tasks):
            self.tasks[index]["completed"] = True
            self._save_tasks()
            print(f"Completed task: {self.tasks[index]['title']}")
        else:
            print("Error: Invalid task index.")

    def list_tasks(
        self,
        filter_priority: Optional[str] = None,
        filter_due_before: Optional[str] = None,
        sort_by: Optional[str] = None,
    ):
        """List tasks with optional filtering and sorting."""
        filtered = self.tasks[:]

        # Filter by priority
        if filter_priority:
            if filter_priority not in VALID_PRIORITIES:
                print(f"Error: Invalid priority '{filter_priority}'.")
                sys.exit(1)
            filtered = [t for t in filtered if t["priority"] == filter_priority]

        # Filter by due date
        if filter_due_before:
            try:
                cutoff = datetime.strptime(filter_due_before, "%Y-%m-%d").date()
            except ValueError:
                print(f"Error: Invalid date format '{filter_due_before}'. Use YYYY-MM-DD.")
                sys.exit(1)
            filtered = [
                t
                for t in filtered
                if t["due_date"]
                and datetime.strptime(t["due_date"], "%Y-%m-%d").date() < cutoff
            ]

        # Sort
        if sort_by == "due_date":
            filtered.sort(
                key=lambda t: (
                    t["due_date"] == "",
                    t["due_date"],
                )
            )
        elif sort_by == "priority":
            filtered.sort(key=lambda t: PRIORITY_ORDER.get(t["priority"], 2))

        self._print_tasks(filtered)

    def _print_tasks(self, tasks: List[dict]):
        """Print tasks in a formatted table."""
        if not tasks:
            print("No tasks found.")
            return

        print(f"{'Index':<6}{'Title':<25}{'Status':<10}{'Priority':<10}{'Due Date':<12}")
        print("-" * 63)
        for i, task in enumerate(tasks):
            status = "Done" if task["completed"] else "Todo"
            due = task["due_date"] if task["due_date"] else "-"
            print(f"{i:<6}{task['title']:<25}{status:<10}{task['priority']:<10}{due:<12}")

    @staticmethod
    def _is_valid_date(date_str: str) -> bool:
        """Check if string is a valid YYYY-MM-DD date."""
        try:
            datetime.strptime(date_str, "%Y-%m-%d")
            return True
        except ValueError:
            return False


def main():
    parser = argparse.ArgumentParser(
        description="To-Do List Application",
        formatter_class=argparse.RawDescriptionHelpFormatter,
    )
    subparsers = parser.add_subparsers(dest="command", help="Available commands")

    # Add task
    add_parser = subparsers.add_parser("add", help="Add a new task")
    add_parser.add_argument("title", help="Task title")
    add_parser.add_argument(
        "--priority",
        choices=["low", "medium", "high"],
        default="low",
        help="Task priority (default: low)",
    )
    add_parser.add_argument(
        "--due", help="Due date in YYYY-MM-DD format", default=""
    )

    # Complete task
    complete_parser = subparsers.add_parser("complete", help="Mark a task as done")
    complete_parser.add_argument("index", type=int, help="Task index")

    # List tasks
    list_parser = subparsers.add_parser(
        "list", help="List tasks with filtering and sorting"
    )
    list_parser.add_argument(
        "--priority",
        choices=["low", "medium", "high"],
        help="Filter by priority",
    )
    list_parser.add_argument(
        "--due-before",
        help="Filter tasks due before date (YYYY-MM-DD)",
    )
    list_parser.add_argument(
        "--sort",
        choices=["due_date", "priority"],
        help="Sort tasks by field",
    )

    args = parser.parse_args()

    if not args.command:
        parser.print_help()
        sys.exit(1)

    manager = TodoManager()

    if args.command == "add":
        manager.add_task(args.title, args.priority, args.due)
    elif args.command == "complete":
        manager.complete_task(args.index)
    elif args.command == "list":
        manager.list_tasks(
            filter_priority=args.priority,
            filter_due_before=args.due_before,
            sort_by=args.sort,
        )


if __name__ == "__main__":
    main()
```

### Key Improvements and Features

1.  **Data Model**:
    *   Tasks now include `priority` (low/medium/high) and `due_date` (YYYY-MM-DD or empty).
    *   Validation ensures only valid priorities and date formats are accepted.

2.  **Migration**:
    *   The `_migrate_task` method automatically adds missing fields to legacy tasks when loading, ensuring backward compatibility.

3.  **List Command Enhancements**:
    *   **Columns**: Displays Index, Title, Status, Priority, and Due Date.
    *   **Filtering**:
        *   `--priority`: Filters tasks by priority level.
        *   `--due-before`: Filters tasks due before a specified date. Empty due dates are excluded.
    *   **Sorting**:
        *   `--sort priority`: Sorts by priority (High → Medium → Low).
        *   `--sort due_date`: Sorts by due date, with tasks having no due date appearing last.

4.  **Help Text**:
    *   Updated `argparse` help text to document all new options and commands.

5.  **Tests**:
    *   Comprehensive unit tests cover adding tasks, migration, filtering by priority and due date, and sorting by priority and due date.

6.  **Code Quality**:
    *   Clean separation of concerns with a `TodoManager` class.
    *   Input validation and error handling.
    *   Configurable data file path.
