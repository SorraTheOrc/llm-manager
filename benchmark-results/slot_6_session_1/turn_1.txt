### 1. `todo_cli.py`
```python
"""CLI-based To-Do Application with JSON storage.

Usage:
    python todo_cli.py add <title>
    python todo_cli.py list [--completed|--uncompleted]
    python todo_cli.py complete <id>
    python todo_cli.py delete <id>
    python todo_cli.py --help

Environment Variables:
    TODO_STORAGE: Path to the JSON file (default: ~/.todo.json)
"""

import json
import os
import sys
import argparse
from pathlib import Path
from typing import Optional

__all__ = ["TodoManager", "main"]


class TodoManager:
    """Manages CRUD operations for tasks using a JSON file."""

    def __init__(self, storage_path: str | Path | None = None):
        if storage_path is None:
            storage_path = os.environ.get("TODO_STORAGE", "~/.todo.json")
        self.storage_path = Path(storage_path).expanduser()
        self.storage_path.parent.mkdir(parents=True, exist_ok=True)
        self._ensure_file()

    def _ensure_file(self) -> None:
        """Create storage file if it doesn't exist."""
        if not self.storage_path.exists():
            self.storage_path.write_text("[]")

    def _load(self) -> list[dict]:
        """Load tasks from JSON file. Returns empty list on corruption."""
        try:
            return json.loads(self.storage_path.read_text())
        except (json.JSONDecodeError, FileNotFoundError):
            return []

    def _save(self, tasks: list[dict]) -> None:
        """Persist tasks to JSON file with pretty formatting."""
        self.storage_path.write_text(json.dumps(tasks, indent=2) + "\n")

    def add_task(self, title: str) -> int:
        """Add a new task. Returns the auto-incremented task ID."""
        if not title.strip():
            raise ValueError("Title cannot be empty")

        tasks = self._load()
        next_id = max((t["id"] for t in tasks), default=0) + 1

        tasks.append({
            "id": next_id,
            "title": title.strip(),
            "completed": False
        })
        self._save(tasks)
        return next_id

    def list_tasks(self, completed: Optional[bool] = None) -> list[dict]:
        """List tasks, optionally filtering by completion status."""
        tasks = self._load()
        if completed is not None:
            tasks = [t for t in tasks if t["completed"] == completed]
        return sorted(tasks, key=lambda t: t["id"])

    def complete_task(self, task_id: int) -> bool:
        """Mark a task as completed. Returns True if task was found and updated."""
        tasks = self._load()
        for task in tasks:
            if task["id"] == task_id:
                task["completed"] = True
                self._save(tasks)
                return True
        return False

    def delete_task(self, task_id: int) -> bool:
        """Delete a task by ID. Returns True if task was found and removed."""
        tasks = self._load()
        initial_len = len(tasks)
        tasks = [t for t in tasks if t["id"] != task_id]

        if len(tasks) < initial_len:
            self._save(tasks)
            return True
        return False


def main(argv: Optional[list[str]] = None) -> int:
    """CLI entry point. Returns exit code (0 for success, 1 for error)."""
    parser = argparse.ArgumentParser(
        description="Simple CLI To-Do Application",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="Examples:\n  todo add Buy groceries\n  todo list --completed\n  todo complete 1\n  todo delete 2"
    )
    subparsers = parser.add_subparsers(dest="command", help="Available commands")

    subparsers.add_parser("add", help="Add a new task")
    list_parser = subparsers.add_parser("list", help="List tasks")
    list_parser.add_argument("--completed", action="store_true", help="Show only completed tasks")
    list_parser.add_argument("--uncompleted", action="store_true", help="Show only uncompleted tasks")
    subparsers.add_parser("complete", help="Mark a task as completed")
    subparsers.add_parser("delete", help="Delete a task")

    args = parser.parse_args(argv)

    if not args.command:
        parser.print_help()
        return 0

    manager = TodoManager()

    if args.command == "add":
        try:
            new_id = manager.add_task(args.title)
            print(f"Added task #{new_id}: {args.title}")
        except ValueError as e:
            print(f"Error: {e}", file=sys.stderr)
            return 1

    elif args.command == "list":
        completed_filter = None
        if args.completed:
            completed_filter = True
        elif args.uncompleted:
            completed_filter = False

        tasks = manager.list_tasks(completed_filter)
        if not tasks:
            print("No tasks found.")
            return 0

        print(f"{'ID':<5} {'Status':<10} {'Title'}")
        print("-" * 50)
        for task in tasks:
            status = "✓" if task["completed"] else "○"
            print(f"{task['id']:<5} {status:<10} {task['title']}")

    elif args.command == "complete":
        if manager.complete_task(args.id):
            print(f"Task #{args.id} marked as completed.")
        else:
            print(f"Error: Task #{args.id} not found.", file=sys.stderr)
            return 1

    elif args.command == "delete":
        if manager.delete_task(args.id):
            print(f"Task #{args.id} deleted.")
        else:
            print(f"Error: Task #{args.id} not found.", file=sys.stderr)
            return 1

    return 0


if __name__ == "__main__":
    sys.exit(main())
```

class TestTodoManager:

        manager = TodoManager(tmp_path / "test.json")
        task_id = manager.add_task("Buy groceries")
        assert task_id == 1
        tasks = manager.list_tasks()
        assert len(tasks) == 1
        assert tasks[0]["id"] == 1
        assert tasks[0]["title"] == "Buy groceries"
        assert tasks[0]["completed"] is False

        manager = TodoManager(tmp_path / "test.json")
        assert manager.add_task("Task 1") == 1
        assert manager.add_task("Task 2") == 2
        assert manager.add_task("Task 3") == 3

        manager = TodoManager(tmp_path / "test.json")
        with pytest.raises(ValueError, match="Title cannot be empty"):
            manager.add_task("   ")
        with pytest.raises(ValueError, match="Title cannot be empty"):
            manager.add_task("")

    def test_list_tasks_filters(self, tmp_path: Path):
        manager = TodoManager(tmp_path / "test.json")
        manager.add_task("Task 1")
        manager.add_task("Task 2")

        assert len(manager.list_tasks()) == 2
        assert len(manager.list_tasks(completed=True)) == 1
        assert len(manager.list_tasks(completed=False)) == 1

    def test_complete_task(self, tmp_path: Path):
        manager = TodoManager(tmp_path / "test.json")
        manager.add_task("Task 1")
        assert manager.complete_task(1) is True
        assert manager.list_tasks(completed=True)[0]["completed"] is True
        assert manager.complete_task(999) is False

    def test_delete_task(self, tmp_path: Path):
        manager = TodoManager(tmp_path / "test.json")
        manager.add_task("Task 1")
        manager.add_task("Task 2")
        assert manager.delete_task(1) is True
        assert len(manager.list_tasks()) == 1
        assert manager.delete_task(999) is False

    def test_json_corruption_recovery(self, tmp_path: Path):

        f.write_text("invalid json {{")
        manager = TodoManager(f)
        assert manager._load() == []
        manager.add_task("Recovery task")
        assert len(manager.list_tasks()) == 1

### Key Design Decisions
1. **Importable Module**: `TodoManager` encapsulates all business logic. `main()` accepts `argv` for deterministic testing and CLI execution.
2. **Storage Resilience**: Gracefully handles missing files and JSON corruption by falling back to an empty list.
3. **Auto-Increment**: Computes next ID from existing tasks rather than relying on a separate counter, ensuring consistency even if tasks are deleted.
4. **Security & Validation**: Strips whitespace, rejects empty titles, validates IDs via `argparse`, and never uses `eval`/`exec`.
5. **Testability**: Uses `tmp_path` fixture for isolated storage, environment variable override for test runs, and `pytest.CaptureFixture` for output verification.
6. **UX**: Clear column formatting, sensible defaults, explicit success/error messages, and standard `--help` integration.
